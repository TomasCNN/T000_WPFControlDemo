private class PropertyToGenerate
{
    public string Name { get; set; }
    public string Type { get; set; }
    public bool IsEnum { get; set; } // 新增：是否为枚举类型
    public string EnumUnderlyingType { get; set; } // 新增：枚举底层类型
    public ushort Address { get; set; }
    public int RegisterType { get; set; }
    public int Length { get; set; }
    public int Endianness { get; set; }
    public double ScaleFactor { get; set; }
    public int BitPosition { get; set; }
    public int StringEncoding { get; set; }
}

private static ClassToGenerate GetClassDeclaration(GeneratorSyntaxContext context)
{
    var classDeclaration = (ClassDeclarationSyntax)context.Node;
    var symbol = context.SemanticModel.GetDeclaredSymbol(classDeclaration);

    if (symbol == null)
        return null;

    var hasAttribute = symbol.GetAttributes()
        .Any(a => a.AttributeClass?.Name == "GenerateModbusParserAttribute");

    if (!hasAttribute)
        return null;

    var properties = new List<PropertyToGenerate>();
    foreach (var property in symbol.GetMembers().OfType<IPropertySymbol>())
    {
        var attribute = property.GetAttributes()
            .FirstOrDefault(a => a.AttributeClass?.Name == "ModbusRegisterAttribute");

        if (attribute == null)
            continue;

        var address = (ushort)attribute.ConstructorArguments[0].Value;
        var type = (int)attribute.ConstructorArguments[1].Value;
        var length = (int)attribute.ConstructorArguments[2].Value;
        var endianness = (int)attribute.ConstructorArguments[3].Value;
        var scaleFactor = (double)attribute.ConstructorArguments[4].Value;
        var bitPosition = (int)attribute.ConstructorArguments[5].Value;
        var stringEncoding = (int)attribute.ConstructorArguments[6].Value;
        var enumUnderlyingType = attribute.ConstructorArguments[7].Value as ITypeSymbol;

        bool isEnum = property.Type.TypeKind == TypeKind.Enum;
        string enumUnderlyingTypeStr = null;

        // 编译时错误检查
        if (isEnum)
        {
            // 如果用户没有指定底层类型，自动获取枚举的实际底层类型
            if (enumUnderlyingType == null)
            {
                enumUnderlyingType = ((INamedTypeSymbol)property.Type).EnumUnderlyingType;
            }
            // 验证用户指定的底层类型是否与枚举实际底层类型匹配
            else if (!SymbolEqualityComparer.Default.Equals(
                enumUnderlyingType, 
                ((INamedTypeSymbol)property.Type).EnumUnderlyingType))
            {
                context.ReportDiagnostic(Diagnostic.Create(
                    new DiagnosticDescriptor(
                        id: "MODBUS002",
                        title: "枚举底层类型不匹配",
                        messageFormat: "属性 '{0}' 的枚举底层类型应为 '{1}'，但指定了 '{2}'",
                        category: "ModbusGenerator",
                        defaultSeverity: DiagnosticSeverity.Error,
                        isEnabledByDefault: true),
                    property.Locations.FirstOrDefault(),
                    property.Name,
                    ((INamedTypeSymbol)property.Type).EnumUnderlyingType.Name,
                    enumUnderlyingType.Name));
            }

            enumUnderlyingTypeStr = enumUnderlyingType.ToDisplayString();

            // 验证长度是否与底层类型匹配
            int expectedLength = enumUnderlyingType.SpecialType switch
            {
                SpecialType.System_Byte or SpecialType.System_SByte => 1,
                SpecialType.System_Int16 or SpecialType.System_UInt16 => 1,
                SpecialType.System_Int32 or SpecialType.System_UInt32 => 2,
                SpecialType.System_Int64 or SpecialType.System_UInt64 => 4,
                _ => -1
            };

            if (expectedLength != length)
            {
                context.ReportDiagnostic(Diagnostic.Create(
                    new DiagnosticDescriptor(
                        id: "MODBUS003",
                        title: "枚举长度不匹配",
                        messageFormat: "枚举类型 '{0}' 的底层类型是 '{1}'，需要长度 {2}，但指定了 {3}",
                        category: "ModbusGenerator",
                        defaultSeverity: DiagnosticSeverity.Error,
                        isEnabledByDefault: true),
                    property.Locations.FirstOrDefault(),
                    property.Type.Name,
                    enumUnderlyingType.Name,
                    expectedLength,
                    length));
            }
        }
        // 字符串类型必须指定长度>0
        else if (property.Type.SpecialType == SpecialType.System_String && length <= 0)
        {
            context.ReportDiagnostic(Diagnostic.Create(
                new DiagnosticDescriptor(
                    id: "MODBUS001",
                    title: "无效的字符串长度",
                    messageFormat: "字符串类型属性 '{0}' 必须指定大于0的Length参数",
                    category: "ModbusGenerator",
                    defaultSeverity: DiagnosticSeverity.Error,
                    isEnabledByDefault: true),
                property.Locations.FirstOrDefault(),
                property.Name));
        }

        properties.Add(new PropertyToGenerate
        {
            Name = property.Name,
            Type = property.Type.ToDisplayString(),
            IsEnum = isEnum,
            EnumUnderlyingType = enumUnderlyingTypeStr,
            Address = address,
            RegisterType = type,
            Length = length,
            Endianness = endianness,
            ScaleFactor = scaleFactor,
            BitPosition = bitPosition,
            StringEncoding = stringEncoding
        });
    }

    return new ClassToGenerate
    {
        Namespace = symbol.ContainingNamespace.ToDisplayString(),
        Name = symbol.Name,
        Properties = properties
    };
}

private static void GenerateEnumRead(StringBuilder sb, PropertyToGenerate prop, string registerArray)
{
    sb.AppendLine($"            if ({registerArray}.Length > {prop.Address + prop.Length - 1})");
    sb.AppendLine("            {");

    // 根据枚举底层类型生成相应的读取代码
    switch (prop.EnumUnderlyingType)
    {
        case "byte":
        case "sbyte":
            sb.AppendLine($"                byte value = (byte){registerArray}[{prop.Address}];");
            break;
        case "short":
            sb.AppendLine($"                short value = (short){registerArray}[{prop.Address}];");
            break;
        case "ushort":
            sb.AppendLine($"                ushort value = {registerArray}[{prop.Address}];");
            break;
        case "int":
            GenerateInt32ReadForEnum(sb, prop, registerArray, "int");
            break;
        case "uint":
            GenerateUInt32ReadForEnum(sb, prop, registerArray, "uint");
            break;
        case "long":
            GenerateInt64ReadForEnum(sb, prop, registerArray, "long");
            break;
        case "ulong":
            GenerateUInt64ReadForEnum(sb, prop, registerArray, "ulong");
            break;
    }

    sb.AppendLine($"                target.{prop.Name} = ({prop.Type})value;");
    sb.AppendLine("            }");
}

    {
        case Endianness.BigEndian:

            break;
        case Endianness.LittleEndian:
            sb.AppendLine($"                {typeName} value = ({typeName})(({registerArray}[{prop.Address + 1}] << 16) | {registerArray}[{prop.Address}]);");
            break;

    sb.AppendLine("#nullable enable");
    sb.AppendLine();

    sb.AppendLine("    {");

    sb.AppendLine();
    sb.AppendLine($"        static {classToGenerate.Name}Config()");
    sb.AppendLine("        {");

    // 计算保持寄存器范围
    var holdingRegisters = classToGenerate.Properties.Where(p => p.RegisterType == (int)RegisterType.HoldingRegister).ToList();
    ushort holdingStart = holdingRegisters.Any() ? holdingRegisters.Min(p => p.Address) : (ushort)0;
    ushort holdingEnd = holdingRegisters.Any() ? holdingRegisters.Max(p => (ushort)(p.Address + p.Length - 1)) : (ushort)0;
    ushort holdingLength = holdingRegisters.Any() ? (ushort)(holdingEnd - holdingStart + 1) : (ushort)0;

    ushort inputStart = inputRegisters.Any() ? inputRegisters.Min(p => p.Address) : (ushort)0;
    ushort inputEnd = inputRegisters.Any() ? inputRegisters.Max(p => (ushort)(p.Address + p.Length - 1)) : (ushort)0;

    sb.AppendLine($"                coilsLength: {coilLength},");
    sb.AppendLine($"                discreteInputsStartAddress: {discreteStart},");
    sb.AppendLine($"                discreteInputsLength: {discreteLength});");
    sb.AppendLine("        }");
    sb.AppendLine();
    sb.AppendLine($"        public static ModbusCommunicationConfig Instance => _config;");
    sb.AppendLine();

    sb.AppendLine("        {");

    sb.AppendLine("            }");

    sb.AppendLine("            if (_config.InputRegistersLength > 0)");
    sb.AppendLine("            {");
    sb.AppendLine($"                inputRegisters = master.ReadInputRegisters(slaveId, _config.InputRegistersStartAddress, _config.InputRegistersLength);");
    sb.AppendLine("            }");
    sb.AppendLine();

    sb.AppendLine("            if (_config.CoilsLength > 0)");
    sb.AppendLine("            {");
    sb.AppendLine($"                coils = master.ReadCoils(slaveId, _config.CoilsStartAddress, _config.CoilsLength);");
    sb.AppendLine("            }");
    sb.AppendLine();

    sb.AppendLine("            if (_config.DiscreteInputsLength > 0)");
    sb.AppendLine("            {");
    sb.AppendLine($"                discreteInputs = master.ReadDiscreteInputs(slaveId, _config.DiscreteInputsStartAddress, _config.DiscreteInputsLength);");
    sb.AppendLine("            }");

    sb.AppendLine();

    sb.AppendLine("        }");

    sb.AppendLine("        {");
    sb.AppendLine("            if (data == null)");
    sb.AppendLine("                throw new ArgumentNullException(nameof(data));");
    sb.AppendLine();
    sb.AppendLine("            if (_config.HoldingRegistersLength > 0)");
    sb.AppendLine("            {");
    sb.AppendLine($"                var holdingRegisters = new ushort[_config.HoldingRegistersLength];");
    sb.AppendLine($"                {classToGenerate.Name}Parser.WriteToRegisters(data, holdingRegisters);");
    sb.AppendLine($"                master.WriteMultipleRegisters(slaveId, _config.HoldingRegistersStartAddress, holdingRegisters);");
    sb.AppendLine("            }");
    sb.AppendLine("        }");
    sb.AppendLine("    }");
    sb.AppendLine("}");

    return sb.ToString();
}

## 四、新增功能详解

### 4.1 枚举类型支持

- **自动检测底层类型**：无需手动指定，源生成器会自动检测枚举的实际底层类型
- **支持所有枚举底层类型**：`byte`、`sbyte`、`short`、`ushort`、`int`、`uint`、`long`、`ulong`
- **完整的字节序处理**：支持所有 4 种字节序格式
- **编译时验证**：
  - 验证用户指定的底层类型是否与枚举实际底层类型匹配
  - 验证寄存器长度是否与底层类型匹配
  - 不匹配时直接编译报错，避免运行时错误

### 4.2 自动生成通信配置类

生成的`{ClassName}Config`类提供以下功能：

- **自动计算寄存器范围**：自动计算每种寄存器类型的起始地址和长度
- **内置批量读写方法**：`ReadAll()`和`WriteAll()`方法，一行代码完成所有数据的读写
- **最优通信策略**：自动合并连续的寄存器，减少网络往返次数
- **零配置**：无需手动编写任何通信代码，完全基于数据模型自动生成

### 4.3 寄存器映射表增强

- 新增`EnumUnderlyingType`属性，显示枚举的底层类型
- 完善的调试信息输出
- 支持按寄存器类型查询

## 六、最佳实践总结

1. **优先使用枚举**：对于状态、代码等有限集合的值，始终使用枚举类型，提高代码可读性和类型安全性
2. **让源生成器自动检测枚举底层类型**：不要手动指定`enumUnderlyingType`参数，除非有特殊需求
3. **使用自动生成的通信方法**：`ReadAll()`和`WriteAll()`方法已经优化到极致，无需手动编写通信代码
4. **利用编译时检查**：源生成器会在编译时发现大多数错误，充分利用这一特性
