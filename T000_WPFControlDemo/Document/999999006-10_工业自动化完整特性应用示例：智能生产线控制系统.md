## 三、第二步：实现设备驱动自动注册框架

使用`DeviceDriverAttribute`实现设备驱动的自动发现和注册，添加新驱动时无需修改任何现有代码。

csharp:

```c#
using System;
using System.Collections.Generic;
using System.Linq;
using System.Reflection;
using IndustrialAutomation.Core.Attributes;

namespace IndustrialAutomation.Core.Devices
{
    /// <summary>
    /// 设备驱动接口
    /// </summary>
    public interface IDeviceDriver
    {
        string DeviceId { get; set; }
        bool IsConnected { get; }
        
        bool Connect(string connectionString);
        void Disconnect();
        object ReadDataPoint(string dataPointId);
        void WriteDataPoint(string dataPointId, object value);
    }

    /// <summary>
    /// 设备驱动工厂
    /// </summary>
    public static class DeviceDriverFactory
    {
        private static readonly Dictionary<string, Type> _driverRegistry = new();

        static DeviceDriverFactory()
        {
            // 自动扫描所有程序集，注册标记了DeviceDriverAttribute的驱动
            RegisterDriversFromAssembly(Assembly.GetExecutingAssembly());
        }

        /// <summary>
        /// 从指定程序集注册驱动
        /// </summary>
        public static void RegisterDriversFromAssembly(Assembly assembly)
        {
            var driverTypes = assembly.GetTypes()
                .Where(t => typeof(IDeviceDriver).IsAssignableFrom(t) 
                           && !t.IsAbstract 
                           && !t.IsInterface);

            foreach (var type in driverTypes)
            {
                var attribute = type.GetCustomAttribute<DeviceDriverAttribute>();
                if (attribute != null)
                {
                    foreach (var model in attribute.SupportedModels)
                    {
                        string key = $"{attribute.Manufacturer}:{model}".ToLowerInvariant();
                        _driverRegistry[key] = type;
                        Console.WriteLine($"注册驱动: {attribute.Manufacturer} {model} -> {type.Name}");
                    }
                }
            }
        }

        /// <summary>
        /// 创建设备驱动实例
        /// </summary>
        public static IDeviceDriver CreateDriver(string manufacturer, string model, string deviceId)
        {
            string key = $"{manufacturer}:{model}".ToLowerInvariant();
            if (_driverRegistry.TryGetValue(key, out var driverType))
            {
                var driver = (IDeviceDriver)Activator.CreateInstance(driverType);
                driver.DeviceId = deviceId;
                return driver;
            }

            throw new NotSupportedException($"不支持的设备: {manufacturer} {model}");
        }

        /// <summary>
        /// 获取所有已注册的驱动信息
        /// </summary>
        public static List<DriverInfo> GetRegisteredDrivers()
        {
            return _driverRegistry.Select(kv => 
            {
                var attribute = kv.Value.GetCustomAttribute<DeviceDriverAttribute>();
                return new DriverInfo
                {
                    DeviceType = attribute.DeviceType,
                    Manufacturer = attribute.Manufacturer,
                    Model = kv.Key.Split(':')[1],
                    Protocol = attribute.Protocol,
                    Version = attribute.Version
                };
            }).ToList();
        }
    }

    public class DriverInfo
    {
        public string DeviceType { get; set; }
        public string Manufacturer { get; set; }
        public string Model { get; set; }
        public string Protocol { get; set; }
        public string Version { get; set; }
    }
}
```

## 四、第三步：实现统一数据点映射框架

使用`DataPointAttribute`实现统一的数据点管理，自动处理数据验证、死区过滤和报警检测。

csharp:

```c#
using System;
using System.Collections.Generic;
using System.Linq;
using System.Reflection;
using IndustrialAutomation.Core.Attributes;

namespace IndustrialAutomation.Core.Data
{
    /// <summary>
    /// 数据点定义
    /// </summary>
    public class DataPointDefinition
    {
        public string Id { get; set; }
        public string Name { get; set; }
        public DataPointType DataType { get; set; }
        public string Unit { get; set; }
        public DataAccessMode AccessMode { get; set; }
        public double Deadband { get; set; }
        public int HistoryRetentionDays { get; set; }
        public double AlarmHigh { get; set; }
        public double AlarmLow { get; set; }
        public PropertyInfo PropertyInfo { get; set; }
        
        public object LastValue { get; set; }
        public DateTime LastUpdateTime { get; set; }
    }

    /// <summary>
    /// 数据点管理器
    /// </summary>
    public class DataPointManager
    {
        private readonly Dictionary<string, DataPointDefinition> _dataPoints = new();
        private readonly object _lock = new();

        /// <summary>
        /// 从数据模型类注册数据点
        /// </summary>
        public void RegisterDataPoints<T>() where T : class
        {
            var properties = typeof(T).GetProperties(BindingFlags.Public | BindingFlags.Instance);
            
            foreach (var prop in properties)
            {
                var attribute = prop.GetCustomAttribute<DataPointAttribute>();
                if (attribute == null) continue;

                var dp = new DataPointDefinition
                {
                    Id = attribute.Id,
                    Name = attribute.Name,
                    DataType = attribute.DataType,
                    Unit = attribute.Unit,
                    AccessMode = attribute.AccessMode,
                    Deadband = attribute.Deadband,
                    HistoryRetentionDays = attribute.HistoryRetentionDays,
                    AlarmHigh = attribute.AlarmHigh,
                    AlarmLow = attribute.AlarmLow,
                    PropertyInfo = prop
                };

                _dataPoints[dp.Id] = dp;
                Console.WriteLine($"注册数据点: {dp.Id} ({dp.Name})");
            }
        }

        /// <summary>
        /// 更新数据点值
        /// </summary>
        public bool UpdateDataPoint<T>(T dataModel, string dataPointId, object value) where T : class
        {
            lock (_lock)
            {
                if (!_dataPoints.TryGetValue(dataPointId, out var dp))
                    return false;

                if (dp.AccessMode == DataAccessMode.WriteOnly)
                    return false;

                // 应用死区过滤
                if (dp.Deadband > 0 && dp.LastValue != null)
                {
                    try
                    {
                        double currentValue = Convert.ToDouble(dp.LastValue);
                        double newValue = Convert.ToDouble(value);
                        
                        if (Math.Abs(newValue - currentValue) < dp.Deadband)
                            return false; // 变化在死区内，不更新
                    }
                    catch
                    {
                        // 非数值类型跳过死区检查
                    }
                }

                // 类型转换
                object convertedValue;
                try
                {
                    convertedValue = Convert.ChangeType(value, dp.PropertyInfo.PropertyType);
                }
                catch
                {
                    return false;
                }

                // 更新值
                dp.PropertyInfo.SetValue(dataModel, convertedValue);
                dp.LastValue = convertedValue;
                dp.LastUpdateTime = DateTime.Now;

                // 报警检测
                CheckAlarm(dp, convertedValue);

                return true;
            }
        }

        /// <summary>
        /// 检查报警条件
        /// </summary>
        private void CheckAlarm(DataPointDefinition dp, object value)
        {
            if (double.IsNaN(dp.AlarmHigh) && double.IsNaN(dp.AlarmLow))
                return;

            try
            {
                double numValue = Convert.ToDouble(value);
                
                if (!double.IsNaN(dp.AlarmHigh) && numValue > dp.AlarmHigh)
                {
                    OnAlarmTriggered(dp, AlarmSeverity.Error, $"{dp.Name} 超过上限: {numValue}{dp.Unit} > {dp.AlarmHigh}{dp.Unit}");
                }
                else if (!double.IsNaN(dp.AlarmLow) && numValue < dp.AlarmLow)
                {
                    OnAlarmTriggered(dp, AlarmSeverity.Error, $"{dp.Name} 低于下限: {numValue}{dp.Unit} < {dp.AlarmLow}{dp.Unit}");
                }
            }
            catch
            {
                // 非数值类型跳过报警检查
            }
        }

        /// <summary>
        /// 报警触发事件
        /// </summary>
        public event Action<DataPointDefinition, AlarmSeverity, string> AlarmTriggered;

        protected virtual void OnAlarmTriggered(DataPointDefinition dp, AlarmSeverity severity, string message)
        {
            AlarmTriggered?.Invoke(dp, severity, message);
        }

        /// <summary>
        /// 获取所有数据点
        /// </summary>
        public IEnumerable<DataPointDefinition> GetAllDataPoints()
        {
            return _dataPoints.Values;
        }
    }
}
```

## 五、第四步：实现 PLC 通信配置与自动解析

结合`PlcDataModelAttribute`和`ModbusRegisterAttribute`，实现 PLC 数据模型的自动配置和高性能解析。

csharp:

```c#
using System;
using System.Linq;
using System.Net.Sockets;
using System.Reflection;
using Modbus.Device;
using IndustrialAutomation.Core.Attributes;

namespace IndustrialAutomation.Core.Plc
{
    /// <summary>
    /// PLC通信基类
    /// </summary>
    public abstract class PlcCommunicatorBase<T> where T : class, new()
    {
        protected readonly T _dataModel = new();
        protected ModbusIpMaster _modbusMaster;
        protected TcpClient _tcpClient;
        protected readonly PlcDataModelAttribute _config;

        protected PlcCommunicatorBase()
        {
            // 自动获取PLC配置特性
            _config = typeof(T).GetCustomAttribute<PlcDataModelAttribute>();
            if (_config == null)
                throw new InvalidOperationException($"类 {typeof(T).Name} 必须标记PlcDataModelAttribute");
        }

        /// <summary>
        /// 连接到PLC
        /// </summary>
        public virtual bool Connect(string ipAddress, int port = 502)
        {
            try
            {
                _tcpClient = new TcpClient();
                _tcpClient.Connect(ipAddress, port);
                _modbusMaster = ModbusIpMaster.CreateIp(_tcpClient);
                _modbusMaster.Transport.ReadTimeout = _config.TimeoutMs;
                _modbusMaster.Transport.Retries = _config.RetryCount;
                
                Console.WriteLine($"PLC连接成功: {ipAddress}:{port}");
                return true;
            }
            catch (Exception ex)
            {
                Console.WriteLine($"PLC连接失败: {ex.Message}");
                return false;
            }
        }

        /// <summary>
        /// 断开连接
        /// </summary>
        public virtual void Disconnect()
        {
            _modbusMaster?.Dispose();
            _tcpClient?.Close();
            Console.WriteLine("PLC连接已断开");
        }

        /// <summary>
        /// 读取所有数据
        /// </summary>
        public abstract T ReadAll();

        /// <summary>
        /// 写入所有数据
        /// </summary>
        public abstract void WriteAll(T data);

        /// <summary>
        /// 获取数据模型
        /// </summary>
        public T DataModel => _dataModel;
    }

    #region 生产线PLC数据模型
    /// <summary>
    /// 生产线PLC数据模型
    /// </summary>
    [PlcDataModel(DefaultSlaveId = 1, TimeoutMs = 500, RetryCount = 2)]
    public class ProductionLinePlcData
    {
        #region 状态数据
        [DataPoint("line.status", "生产线状态", DataPointType.Enum, 
            accessMode: DataAccessMode.ReadOnly, alarmHigh: 3)]
        [ModbusRegister(0x0000, RegisterType.HoldingRegister, length: 1)]
        public LineStatus Status { get; set; }

        [DataPoint("line.speed", "生产线速度", DataPointType.Float, 
            unit: "m/min", accessMode: DataAccessMode.ReadWrite, deadband: 0.1)]
        [ModbusRegister(0x0001, RegisterType.HoldingRegister, scaleFactor: 0.1)]
        public float Speed { get; set; }

        [DataPoint("line.production_count", "生产数量", DataPointType.Integer, 
            accessMode: DataAccessMode.ReadOnly, historyRetentionDays: 365)]
        [ModbusRegister(0x0002, RegisterType.HoldingRegister, length: 2)]
        public int ProductionCount { get; set; }

        [DataPoint("line.defect_count", "不良品数量", DataPointType.Integer, 
            accessMode: DataAccessMode.ReadOnly, historyRetentionDays: 365)]
        [ModbusRegister(0x0004, RegisterType.HoldingRegister, length: 2)]
        public int DefectCount { get; set; }
        #endregion

        #region 温度数据
        [DataPoint("temp.zone1", "一区温度", DataPointType.Float, 
            unit: "°C", accessMode: DataAccessMode.ReadOnly, 
            deadband: 0.5, alarmHigh: 80.0, alarmLow: 20.0)]
        [ModbusRegister(0x0010, RegisterType.InputRegister, scaleFactor: 0.1)]
        public float TemperatureZone1 { get; set; }

        [DataPoint("temp.zone2", "二区温度", DataPointType.Float, 
            unit: "°C", accessMode: DataAccessMode.ReadOnly, 
            deadband: 0.5, alarmHigh: 80.0, alarmLow: 20.0)]
        [ModbusRegister(0x0011, RegisterType.InputRegister, scaleFactor: 0.1)]
        public float TemperatureZone2 { get; set; }
        #endregion

        #region 控制信号
        [DataPoint("control.start", "启动信号", DataPointType.Boolean, 
            accessMode: DataAccessMode.WriteOnly)]
        [ModbusRegister(0x0020, RegisterType.Coil)]
        public bool StartSignal { get; set; }

        [DataPoint("control.stop", "停止信号", DataPointType.Boolean, 
            accessMode: DataAccessMode.WriteOnly)]
        [ModbusRegister(0x0021, RegisterType.Coil)]
        public bool StopSignal { get; set; }

        [DataPoint("control.reset", "复位信号", DataPointType.Boolean, 
            accessMode: DataAccessMode.WriteOnly)]
        [ModbusRegister(0x0022, RegisterType.Coil)]
        public bool ResetSignal { get; set; }
        #endregion

        #region 报警数据
        [DataPoint("alarm.active", "活动报警", DataPointType.Boolean, 
            accessMode: DataAccessMode.ReadOnly)]
        [ModbusRegister(0x0030, RegisterType.DiscreteInput)]
        public bool AlarmActive { get; set; }

        [DataPoint("alarm.code", "报警代码", DataPointType.Integer, 
            accessMode: DataAccessMode.ReadOnly)]
        [ModbusRegister(0x0031, RegisterType.InputRegister)]
        public int AlarmCode { get; set; }
        #endregion
    }

    public enum LineStatus : ushort
    {
        Idle = 0,
        Running = 1,
        Paused = 2,
        Error = 3,
        EmergencyStop = 4
    }
    #endregion

    #region 生产线PLC通信实现
    /// <summary>
    /// 生产线PLC通信器
    /// </summary>
    public class ProductionLinePlcCommunicator : PlcCommunicatorBase<ProductionLinePlcData>
    {
        public override ProductionLinePlcData ReadAll()
        {
            if (_modbusMaster == null || !_tcpClient.Connected)
                throw new InvalidOperationException("PLC未连接");

            // 批量读取所有寄存器
            var holdingRegisters = _modbusMaster.ReadHoldingRegisters(_config.DefaultSlaveId, 0, 6);
            var inputRegisters = _modbusMaster.ReadInputRegisters(_config.DefaultSlaveId, 0, 13);
            var coils = _modbusMaster.ReadCoils(_config.DefaultSlaveId, 0, 3);
            var discreteInputs = _modbusMaster.ReadDiscreteInputs(_config.DefaultSlaveId, 0, 1);

            // 解析数据
            _dataModel.Status = (LineStatus)holdingRegisters[0];
            _dataModel.Speed = holdingRegisters[1] * 0.1f;
            _dataModel.ProductionCount = (holdingRegisters[2] << 16) | holdingRegisters[3];
            _dataModel.DefectCount = (holdingRegisters[4] << 16) | holdingRegisters[5];

            _dataModel.TemperatureZone1 = inputRegisters[0] * 0.1f;
            _dataModel.TemperatureZone2 = inputRegisters[1] * 0.1f;

            _dataModel.StartSignal = coils[0];
            _dataModel.StopSignal = coils[1];
            _dataModel.ResetSignal = coils[2];

            _dataModel.AlarmActive = discreteInputs[0];
            _dataModel.AlarmCode = inputRegisters[17];

            return _dataModel;
        }

        public override void WriteAll(ProductionLinePlcData data)
        {
            if (_modbusMaster == null || !_tcpClient.Connected)
                throw new InvalidOperationException("PLC未连接");

            // 写入保持寄存器
            var holdingRegisters = new ushort[6];
            holdingRegisters[0] = (ushort)data.Status;
            holdingRegisters[1] = (ushort)(data.Speed * 10);
            holdingRegisters[2] = (ushort)(data.ProductionCount >> 16);
            holdingRegisters[3] = (ushort)(data.ProductionCount & 0xFFFF);
            holdingRegisters[4] = (ushort)(data.DefectCount >> 16);
            holdingRegisters[5] = (ushort)(data.DefectCount & 0xFFFF);

            _modbusMaster.WriteMultipleRegisters(_config.DefaultSlaveId, 0, holdingRegisters);

            // 写入线圈
            var coils = new[] { data.StartSignal, data.StopSignal, data.ResetSignal };
            _modbusMaster.WriteMultipleCoils(_config.DefaultSlaveId, 0, coils);
        }
    }

}
```

## 七、第六步：实现实时任务调度器

使用`RealTimeTaskAttribute`实现基于优先级的实时任务调度，确保关键任务按时执行。

csharp:

```c#
using System;
using System.Collections.Generic;
using System.Linq;
using System.Reflection;
using System.Threading;
using System.Threading.Tasks;
using IndustrialAutomation.Core.Attributes;

namespace IndustrialAutomation.Core.Scheduling
{
    /// <summary>
    /// 实时任务调度器
    /// </summary>
    public class RealTimeTaskScheduler
    {
        private readonly List<TaskDefinition> _tasks = new();
        private readonly CancellationTokenSource _cts = new();
        private bool _isRunning;

        /// <summary>
        /// 注册对象中的所有实时任务
        /// </summary>
        public void RegisterTasks(object target)
        {
            var methods = target.GetType().GetMethods(BindingFlags.Public | BindingFlags.Instance);
            
            foreach (var method in methods)
            {
                var attribute = method.GetCustomAttribute<RealTimeTaskAttribute>();
                if (attribute == null) continue;

                _tasks.Add(new TaskDefinition
                {
                    Target = target,
                    Method = method,
                    Attribute = attribute
                });

                Console.WriteLine($"注册实时任务: {method.Name} (周期: {attribute.PeriodMs}ms, 优先级: {attribute.Priority})");
            }
        }

        /// <summary>
        /// 启动所有任务
        /// </summary>
        public void Start()
        {
            if (_isRunning) return;

            _isRunning = true;
            Console.WriteLine("实时任务调度器启动");

            // 按优先级排序任务
            var sortedTasks = _tasks.OrderByDescending(t => t.Attribute.Priority).ToList();

            foreach (var task in sortedTasks)
            {
                _ = RunTaskAsync(task, _cts.Token);
            }
        }

        /// <summary>
        /// 停止所有任务
        /// </summary>
        public void Stop()
        {
            if (!_isRunning) return;

            _cts.Cancel();
            _isRunning = false;
            Console.WriteLine("实时任务调度器停止");
        }

        private async Task RunTaskAsync(TaskDefinition task, CancellationToken cancellationToken)
        {
            while (!cancellationToken.IsCancellationRequested)
            {
                var startTime = DateTime.UtcNow;

                try
                {
                    // 执行任务
                    task.Method.Invoke(task.Target, null);
                }
                catch (Exception ex)
                {
                    Console.WriteLine($"任务 {task.Method.Name} 执行失败: {ex.Message}");
                    
                    if (task.Attribute.IsCritical)
                    {
                        Console.WriteLine("关键任务失败，触发紧急停止!");
                        EmergencyStop?.Invoke();
                        return;
                    }
                }

                // 计算下一次执行时间
                var elapsed = DateTime.UtcNow - startTime;
                var delay = task.Attribute.PeriodMs - elapsed.TotalMilliseconds;

                if (delay > 0)
                {
                    await Task.Delay((int)delay, cancellationToken);
                }
                else
                {
                    Console.WriteLine($"任务 {task.Method.Name} 超时 {Math.Abs(delay):F0}ms");
                }
            }
        }

        /// <summary>
        /// 紧急停止事件
        /// </summary>
        public event Action EmergencyStop;

        private class TaskDefinition
        {
            public object Target { get; set; }
            public MethodInfo Method { get; set; }
            public RealTimeTaskAttribute Attribute { get; set; }
        }
    }
}
```

## 八、第七步：上层业务逻辑集成

展示如何将所有组件集成到实际的生产线控制系统中。

csharp:

```c#
using System;
using System.Linq;
using IndustrialAutomation.Core.Data;
using IndustrialAutomation.Core.Devices;
using IndustrialAutomation.Core.Plc;
using IndustrialAutomation.Core.Scheduling;
using IndustrialAutomation.Core.Attributes;

namespace IndustrialAutomation.ProductionLine
{
    /// <summary>
    /// 生产线控制器
    /// </summary>
    public class ProductionLineController
    {
        private readonly ProductionLinePlcCommunicator _plcCommunicator;
        private readonly DataPointManager _dataPointManager;
        private readonly RealTimeTaskScheduler _taskScheduler;
        private readonly IDeviceDriver _camera;

        public ProductionLineController()
        {
            // 初始化组件
            _plcCommunicator = new ProductionLinePlcCommunicator();
            _dataPointManager = new DataPointManager();
            _taskScheduler = new RealTimeTaskScheduler();

            // 注册数据点
            _dataPointManager.RegisterDataPoints<ProductionLinePlcData>();
            
            // 注册报警处理
            _dataPointManager.AlarmTriggered += OnAlarmTriggered;

            // 创建设备驱动
            _camera = DeviceDriverFactory.CreateDriver("IK", "E200", "Camera1");

            // 注册实时任务
            _taskScheduler.RegisterTasks(this);
            _taskScheduler.EmergencyStop += OnEmergencyStop;
        }

        /// <summary>
        /// 启动生产线
        /// </summary>
        public bool Start(string plcIpAddress, string cameraIpAddress)
        {
            Console.WriteLine("正在启动生产线...");

            // 连接PLC
            if (!_plcCommunicator.Connect(plcIpAddress))
                return false;

            // 连接相机
            if (!_camera.Connect(cameraIpAddress))
            {
                _plcCommunicator.Disconnect();
                return false;
            }

            // 启动实时任务
            _taskScheduler.Start();

            Console.WriteLine("生产线启动成功!");
            return true;
        }

        /// <summary>
        /// 停止生产线
        /// </summary>
        public void Stop()
        {
            Console.WriteLine("正在停止生产线...");

            _taskScheduler.Stop();
            _camera.Disconnect();
            _plcCommunicator.Disconnect();

            Console.WriteLine("生产线已停止");
        }

        #region 实时任务
        /// <summary>
        /// 安全监控任务(最高优先级，10ms执行一次)
        /// </summary>
        [RealTimeTask(periodMs: 10, priority: 99, deadlineMs: 8, isCritical: true)]
        public void SafetyMonitorTask()
        {
            var data = _plcCommunicator.ReadAll();
            
            if (data.Status == LineStatus.EmergencyStop)
            {
                Console.WriteLine("检测到紧急停止信号!");
                OnEmergencyStop();
            }
        }

        /// <summary>
        /// PLC数据采集任务(高优先级，50ms执行一次)
        /// </summary>
        [RealTimeTask(periodMs: 50, priority: 80, deadlineMs: 40)]
        public void PlcDataAcquisitionTask()
        {
            var data = _plcCommunicator.ReadAll();
            
            // 更新数据点
            _dataPointManager.UpdateDataPoint(data, "line.status", data.Status);
            _dataPointManager.UpdateDataPoint(data, "line.speed", data.Speed);
            _dataPointManager.UpdateDataPoint(data, "line.production_count", data.ProductionCount);
            _dataPointManager.UpdateDataPoint(data, "line.defect_count", data.DefectCount);
            _dataPointManager.UpdateDataPoint(data, "temp.zone1", data.TemperatureZone1);
            _dataPointManager.UpdateDataPoint(data, "temp.zone2", data.TemperatureZone2);
            _dataPointManager.UpdateDataPoint(data, "alarm.active", data.AlarmActive);
            _dataPointManager.UpdateDataPoint(data, "alarm.code", data.AlarmCode);
        }

        /// <summary>
        /// 状态上报任务(中优先级，1000ms执行一次)
        /// </summary>
        [RealTimeTask(periodMs: 1000, priority: 50, deadlineMs: 500)]
        public void StatusReportTask()
        {
            var data = _plcCommunicator.DataModel;
            var yield = data.ProductionCount > 0 
                ? (1 - (double)data.DefectCount / data.ProductionCount) * 100 
                : 100;

            Console.WriteLine($"[{DateTime.Now:HH:mm:ss}] 状态: {data.Status,-12} 速度: {data.Speed:F1}m/min 产量: {data.ProductionCount} 良率: {yield:F1}%");
        }
        #endregion

        #region 事件处理
        private void OnAlarmTriggered(DataPointDefinition dp, AlarmSeverity severity, string message)
        {
            Console.WriteLine($"[{severity}] {message}");
            
            if (severity == AlarmSeverity.Critical)
            {
                // 触发紧急停止
                var data = _plcCommunicator.DataModel;
                data.StopSignal = true;
                _plcCommunicator.WriteAll(data);
            }
        }

        private void OnEmergencyStop()
        {
            Console.WriteLine("执行紧急停止!");
            Stop();
        }
        #endregion

        #region 控制方法
        public void StartProduction()
        {
            var data = _plcCommunicator.DataModel;
            data.StartSignal = true;
            _plcCommunicator.WriteAll(data);
            Console.WriteLine("发送启动信号");
        }

        public void StopProduction()
        {
            var data = _plcCommunicator.DataModel;
            data.StopSignal = true;
            _plcCommunicator.WriteAll(data);
            Console.WriteLine("发送停止信号");
        }

        public void ResetAlarm()
        {
            var data = _plcCommunicator.DataModel;
            data.ResetSignal = true;
            _plcCommunicator.WriteAll(data);
            Console.WriteLine("发送复位信号");
        }
        #endregion
    }

    /// <summary>
    /// 程序入口
    /// </summary>
    class Program
    {
        static void Main(string[] args)
        {
            Console.WriteLine("=== 智能生产线控制系统 ===");
            Console.WriteLine();

            // 显示已注册的驱动
            Console.WriteLine("已注册的设备驱动:");
            foreach (var driver in DeviceDriverFactory.GetRegisteredDrivers())
            {
                Console.WriteLine($"  {driver.Manufacturer} {driver.Model} - {driver.DeviceType} (v{driver.Version})");
            }
            Console.WriteLine();

            // 创建生产线控制器
            var controller = new ProductionLineController();

            try
            {
                // 启动生产线
                if (controller.Start("192.168.1.100", "192.168.1.101"))
                {
                    controller.StartProduction();

                    Console.WriteLine();
                    Console.WriteLine("按任意键停止生产线...");
                    Console.ReadKey();

                    controller.StopProduction();
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine($"系统错误: {ex.Message}");
            }
            finally
            {
                controller.Stop();
            }

            Console.WriteLine();
            Console.WriteLine("程序结束");
        }
    }
}
```

## 十、核心优势与扩展方向

### 10.1 特性驱动架构的核心优势

1. **声明式编程**：所有配置都通过特性声明，代码简洁易读
2. **高度可扩展**：添加新设备、新数据点或新任务时，只需添加特性标记
3. **自动发现**：驱动和任务自动注册，无需手动修改工厂代码
4. **集中管理**：所有元数据集中定义，便于维护和修改
5. **编译时检查**：特性参数错误在编译时即可发现
6. **高性能**：结合源生成器可以实现零反射的高性能解析

### 10.2 扩展方向

1. **添加源生成器**：为 PLC 数据模型自动生成解析代码，替代手动编写的 ReadAll/WriteAll 方法
2. **实现 HMI 自动生成**：根据数据点特性自动生成 HMI 界面
3. **添加历史数据存储**：根据数据点的 HistoryRetentionDays 属性自动配置历史数据存储
4. **实现 OPC UA 服务器**：自动将数据点暴露为 OPC UA 节点
5. **添加配方管理**：使用特性定义配方参数，自动实现配方的保存和加载

这个架构已经在多个实际工业项目中得到验证，能够显著提高开发效率和系统可维护性，特别适合设备种类多、协议复杂的工业自动化场景。
