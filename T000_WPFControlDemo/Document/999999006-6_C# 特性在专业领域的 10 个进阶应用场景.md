## 三、实时系统任务调度与优先级管理

    public int MaxRetries { get; }

        int timeoutMs = 1000, int maxRetries = 0, bool isCritical = false)
    {

        DeadlineMs = deadlineMs;
        TimeoutMs = timeoutMs;

// 实时任务调度器
public class RealTimeTaskScheduler
{
    private readonly List<TaskDefinition> _tasks = new();
    private readonly CancellationTokenSource _cts = new();

    public void RegisterTasks(object target)
    {
        var methods = target.GetType().GetMethods(BindingFlags.Public | BindingFlags.Instance);
        foreach (var method in methods)
        {
            var attr = method.GetCustomAttribute<RealTimeTaskAttribute>();
            if (attr != null)
            {
                _tasks.Add(new TaskDefinition
                {
                    Target = target,
                    Method = method,
                    Attribute = attr
                });
            }
        }
    }

    public void Start()
    {
        foreach (var task in _tasks.OrderByDescending(t => t.Attribute.Priority))
        {
            _ = RunTaskAsync(task, _cts.Token);
        }
    }

    private async Task RunTaskAsync(TaskDefinition task, CancellationToken cancellationToken)
    {
        while (!cancellationToken.IsCancellationRequested)
        {
            var startTime = DateTime.UtcNow;
            
            try
            {
                using var cts = new CancellationTokenSource(task.Attribute.TimeoutMs);
                using var linkedCts = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken, cts.Token);
                
                await Task.Run(() => task.Method.Invoke(task.Target, null), linkedCts.Token);
            }
            catch (OperationCanceledException)
            {
                Console.WriteLine($"任务 {task.Method.Name} 执行超时");
            }
            catch (Exception ex)
            {
                Console.WriteLine($"任务 {task.Method.Name} 执行失败：{ex.Message}");
                
                if (task.Attribute.IsCritical)
                {
                    // 关键任务失败，触发系统紧急停止
                    EmergencyStop();
                    return;
                }
            }

            var elapsed = DateTime.UtcNow - startTime;
            var delay = task.Attribute.PeriodMs - elapsed.TotalMilliseconds;
            
            if (delay > 0)
            {
                await Task.Delay((int)delay, cancellationToken);
            }
            else
            {
                Console.WriteLine($"任务 {task.Method.Name} 超出周期时间 {Math.Abs(delay)}ms");
            }
        }
    }

    private void EmergencyStop()
    {
        _cts.Cancel();
        // 执行紧急停止逻辑
    }

    private class TaskDefinition
    {
        public object Target { get; set; }
        public MethodInfo Method { get; set; }
        public RealTimeTaskAttribute Attribute { get; set; }
    }
}

public class DataPointManager
{
    private readonly Dictionary<string, DataPointDefinition> _dataPoints = new();

    public void RegisterDataPoints<T>()
    {
        var properties = typeof(T).GetProperties(BindingFlags.Public | BindingFlags.Instance);
        foreach (var prop in properties)
        {
            var attr = prop.GetCustomAttribute<DataPointAttribute>();
            if (attr != null)
            {
                _dataPoints[attr.Id] = new DataPointDefinition
                {
                    Id = attr.Id,
                    Name = attr.Name,
                    Type = attr.Type,
                    Unit = attr.Unit,
                    IsReadOnly = attr.IsReadOnly,
                    Deadband = attr.Deadband,
                    HistoryRetentionDays = attr.HistoryRetentionDays,
                    PropertyInfo = prop
                };
            }
        }
    }

    public void UpdateDataPoint<T>(T dataModel, string dataPointId, object value)
    {
        if (_dataPoints.TryGetValue(dataPointId, out var dp))
        {
            if (dp.IsReadOnly)
                throw new InvalidOperationException($"数据点 {dataPointId} 是只读的");

            // 应用死区过滤
            if (dp.Type == DataPointType.Float && dp.Deadband > 0)
            {
                var currentValue = (double)dp.PropertyInfo.GetValue(dataModel);
                if (Math.Abs((double)value - currentValue) < dp.Deadband)
                {
                    return; // 变化在死区内，不更新
                }
            }

            dp.PropertyInfo.SetValue(dataModel, Convert.ChangeType(value, dp.PropertyInfo.PropertyType));
        }
    }
}

public class DataPointDefinition
{
    public string Id { get; set; }
    public string Name { get; set; }
    public DataPointType Type { get; set; }
    public string Unit { get; set; }
    public bool IsReadOnly { get; set; }
    public double Deadband { get; set; }
    public int HistoryRetentionDays { get; set; }
    public PropertyInfo PropertyInfo { get; set; }
}

    public List<BenchmarkResult> RunBenchmarks<T>() where T : new()
    {
        var instance = new T();
        var results = new List<BenchmarkResult>();

        var methods = typeof(T).GetMethods(BindingFlags.Public | BindingFlags.Instance)
            .Select(m => new
            {
                Method = m,
                Attr = m.GetCustomAttribute<PerformanceBenchmarkAttribute>()
            })
            .Where(x => x.Attr != null);

        foreach (var item in methods)
        {
            Console.WriteLine($"正在运行基准测试：{item.Attr.Name}");

            // 预热
            for (int i = 0; i < item.Attr.WarmupIterations; i++)
            {
                item.Method.Invoke(instance, null);
            }

            // 正式测试
            var times = new List<long>();
            for (int i = 0; i < item.Attr.TestIterations; i++)
            {
                var sw = Stopwatch.StartNew();
                item.Method.Invoke(instance, null);
                sw.Stop();
                times.Add(sw.ElapsedTicks);
            }

            // 计算统计数据
            var averageMs = times.Average(t => t * 1000.0 / Stopwatch.Frequency);
            var minMs = times.Min(t => t * 1000.0 / Stopwatch.Frequency);
            var maxMs = times.Max(t => t * 1000.0 / Stopwatch.Frequency);
            var stdDevMs = Math.Sqrt(times.Average(t => Math.Pow(t * 1000.0 / Stopwatch.Frequency - averageMs, 2)));

            var result = new BenchmarkResult
            {
                Name = item.Attr.Name,
                AverageTimeMs = averageMs,
                MinTimeMs = minMs,
                MaxTimeMs = maxMs,
                StdDevMs = stdDevMs,
                MaxAllowedTimeMs = item.Attr.MaxAllowedTimeMs,
                Passed = averageMs <= item.Attr.MaxAllowedTimeMs
            };

            results.Add(result);

            Console.WriteLine($"  平均时间：{averageMs:F3}ms (允许最大：{item.Attr.MaxAllowedTimeMs:F3}ms)");
            Console.WriteLine($"  结果：{(result.Passed ? "通过" : "失败")}");
        }

        return results;
    }
}

public class CommandLineParser
{
    public T Parse<T>(string[] args) where T : new()
    {
        var result = new T();
        var properties = typeof(T).GetProperties(BindingFlags.Public | BindingFlags.Instance)
            .Select(p => new
            {
                Property = p,
                Attr = p.GetCustomAttribute<CommandLineArgumentAttribute>()
            })
            .Where(x => x.Attr != null)
            .ToDictionary(x => x.Attr.LongName);

        // 设置默认值
        foreach (var item in properties.Values)
        {
            if (item.Attr.DefaultValue != null)
            {
                item.Property.SetValue(result, Convert.ChangeType(item.Attr.DefaultValue, item.Property.PropertyType));
            }
        }

        // 解析参数
        for (int i = 0; i < args.Length; i++)
        {
            var arg = args[i];
            if (arg.StartsWith("--"))
            {
                var longName = arg.Substring(2);
                if (properties.TryGetValue(longName, out var item))
                {
                    if (item.Property.PropertyType == typeof(bool))
                    {
                        item.Property.SetValue(result, true);
                    }
                    else if (i + 1 < args.Length)
                    {
                        var value = args[++i];
                        item.Property.SetValue(result, Convert.ChangeType(value, item.Property.PropertyType));
                    }
                }
            }
            else if (arg.StartsWith("-"))
            {
                var shortName = arg.Substring(1);
                var item = properties.Values.FirstOrDefault(x => x.Attr.ShortName == shortName);
                if (item != null)
                {
                    if (item.Property.PropertyType == typeof(bool))
                    {
                        item.Property.SetValue(result, true);
                    }
                    else if (i + 1 < args.Length)
                    {
                        var value = args[++i];
                        item.Property.SetValue(result, Convert.ChangeType(value, item.Property.PropertyType));
                    }
                }
            }
        }

        // 验证必填参数
        foreach (var item in properties.Values)
        {
            if (item.Attr.IsRequired && item.Property.GetValue(result) == null)
            {
                throw new ArgumentException($"参数 --{item.Attr.LongName} 是必填的");
            }
        }

        return result;
    }

    public void ShowHelp<T>()
    {
        Console.WriteLine("用法：");
        Console.WriteLine("  程序名 [选项]");
        Console.WriteLine();
        Console.WriteLine("选项：");

        var properties = typeof(T).GetProperties(BindingFlags.Public | BindingFlags.Instance)
            .Select(p => p.GetCustomAttribute<CommandLineArgumentAttribute>())
            .Where(a => a != null);

        foreach (var attr in properties)
        {
            Console.WriteLine($"  -{attr.ShortName}, --{attr.LongName,-15} {attr.Description}");
            if (attr.DefaultValue != null)
            {
                Console.WriteLine($"                        默认值：{attr.DefaultValue}");
            }
            if (attr.IsRequired)
            {
                Console.WriteLine($"                        必填");
            }
        }
    }
}

    private TState _currentState;

    {
        _currentState = initialState;
        LoadTransitions();
    }

    private void LoadTransitions()
    {
        var stateType = typeof(TState);
        foreach (TState state in Enum.GetValues(stateType))
        {
            var field = stateType.GetField(state.ToString());
            var attributes = field.GetCustomAttributes<StateTransitionAttribute>();
            
            foreach (var attr in attributes)
            {
                var fromState = attr.FromState == null ? default : (TState)Enum.Parse(stateType, attr.FromState.ToString());
                var toState = (TState)Enum.Parse(stateType, attr.ToState.ToString());
                
                if (!_transitions.ContainsKey(fromState))
                {
                    _transitions[fromState] = new List<Transition>();
                }
                
                _transitions[fromState].Add(new Transition
                {
                    ToState = toState,
                    Trigger = attr.Trigger,
                    GuardCondition = attr.GuardCondition,
                    Action = attr.Action
                });
            }
        }
    }

    public bool CanFire(TTrigger trigger)
    {
        return _transitions.TryGetValue(_currentState, out var transitions) &&
               transitions.Any(t => t.Trigger == trigger.ToString());
    }

        if (!_transitions.TryGetValue(_currentState, out var transitions))
        {
