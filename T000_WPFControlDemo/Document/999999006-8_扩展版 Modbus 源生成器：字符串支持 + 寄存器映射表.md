```c#
private static void GenerateStringRead(StringBuilder sb, PropertyToGenerate prop, string registerArray)
{
    sb.AppendLine($"            if ({registerArray}.Length > {prop.Address + prop.Length - 1})");
    sb.AppendLine("            {");
    sb.AppendLine($"                var bytes = new byte[{prop.Length * 2}];");
    sb.AppendLine($"                for (int i = 0; i < {prop.Length}; i++)");
    sb.AppendLine("                {");
    
    // 根据字节序处理每个寄存器的高低字节
    if ((Endianness)prop.Endianness == Endianness.BigEndian)
    {
        sb.AppendLine($"                    bytes[i * 2] = (byte)({registerArray}[{prop.Address} + i] >> 8);");
        sb.AppendLine($"                    bytes[i * 2 + 1] = (byte)({registerArray}[{prop.Address} + i] & 0xFF);");
    }
    else // LittleEndian
    {
        sb.AppendLine($"                    bytes[i * 2] = (byte)({registerArray}[{prop.Address} + i] & 0xFF);");
        sb.AppendLine($"                    bytes[i * 2 + 1] = (byte)({registerArray}[{prop.Address} + i] >> 8);");
    }
    
    sb.AppendLine("                }");
    
    // 根据编码转换为字符串
    if ((StringEncoding)prop.StringEncoding == StringEncoding.ASCII)
    {
        sb.AppendLine($"                var str = System.Text.Encoding.ASCII.GetString(bytes);");
    }
    else // UTF8
    {
        sb.AppendLine($"                var str = System.Text.Encoding.UTF8.GetString(bytes);");
    }
    
    // 自动截断到第一个空字符
    sb.AppendLine("                var nullIndex = str.IndexOf('\\0');");
    sb.AppendLine("                target.{prop.Name} = nullIndex >= 0 ? str.Substring(0, nullIndex) : str;");
    sb.AppendLine("            }");
}
```

```c#
private static void Execute(ClassToGenerate classToGenerate, SourceProductionContext context)

// 新增：生成寄存器映射表类
private static string GenerateRegisterMapClass(ClassToGenerate classToGenerate)
{
    var sb = new StringBuilder();
    sb.AppendLine("// <auto-generated/>");
    sb.AppendLine("#nullable enable");
    sb.AppendLine();
    sb.AppendLine("using System;");
    sb.AppendLine("using System.Collections.Generic;");
    sb.AppendLine("using System.Collections.ObjectModel;");
    sb.AppendLine("using Modbus.Attributes;");
    sb.AppendLine();
    sb.AppendLine($"namespace {classToGenerate.Namespace}");
    sb.AppendLine("{");
    sb.AppendLine($"    public static class {classToGenerate.Name}Map");
    sb.AppendLine("    {");
    sb.AppendLine($"        private static readonly ReadOnlyDictionary<string, ModbusRegisterInfo> _registerMap;");
    sb.AppendLine();
    sb.AppendLine($"        static {classToGenerate.Name}Map()");
    sb.AppendLine("        {");
    sb.AppendLine($"            var dictionary = new Dictionary<string, ModbusRegisterInfo>({classToGenerate.Properties.Count}, StringComparer.Ordinal);");
    sb.AppendLine();

    foreach (var prop in classToGenerate.Properties)
    {
        sb.AppendLine($"            dictionary.Add(\"{prop.Name}\", new ModbusRegisterInfo(");
        sb.AppendLine($"                address: {prop.Address},");
        sb.AppendLine($"                registerType: (RegisterType){prop.RegisterType},");
        sb.AppendLine($"                length: {prop.Length},");
        sb.AppendLine($"                endianness: (Endianness){prop.Endianness},");
        sb.AppendLine($"                scaleFactor: {prop.ScaleFactor},");
        sb.AppendLine($"                bitPosition: {prop.BitPosition},");
        sb.AppendLine($"                stringEncoding: (StringEncoding){prop.StringEncoding},");
        sb.AppendLine($"                propertyType: typeof({prop.Type}),");
        sb.AppendLine($"                propertyName: \"{prop.Name}\"));");
        sb.AppendLine();
    }

    sb.AppendLine($"            _registerMap = new ReadOnlyDictionary<string, ModbusRegisterInfo>(dictionary);");
    sb.AppendLine("        }");
    sb.AppendLine();
    sb.AppendLine($"        public static IReadOnlyDictionary<string, ModbusRegisterInfo> RegisterMap => _registerMap;");
    sb.AppendLine();
    sb.AppendLine($"        public static ModbusRegisterInfo? GetRegisterInfo(string propertyName)");
    sb.AppendLine("        {");
    sb.AppendLine($"            _registerMap.TryGetValue(propertyName, out var info);");
    sb.AppendLine($"            return info;");
    sb.AppendLine("        }");
    sb.AppendLine();
    sb.AppendLine($"        public static IEnumerable<ModbusRegisterInfo> GetRegistersByType(RegisterType type)");
    sb.AppendLine("        {");
    sb.AppendLine($"            foreach (var info in _registerMap.Values)");
    sb.AppendLine("            {");
    sb.AppendLine($"                if (info.RegisterType == type)");
    sb.AppendLine($"                    yield return info;");
    sb.AppendLine("            }");
    sb.AppendLine("        }");
    sb.AppendLine("    }");
    sb.AppendLine("}");

    return sb.ToString();
}
```

### 3.2 使用生成的解析器和映射表

csharp:

```c#
public class ProductionLineCommunicator
{
    private readonly ModbusIpMaster _modbusMaster;

    // ... 构造函数和其他方法 ...

    public ProductionLineData ReadData()
    {
        var data = new ProductionLineData();

        // 使用映射表自动计算需要读取的寄存器范围
        var holdingRegistersToRead = ProductionLineDataMap.GetRegistersByType(RegisterType.HoldingRegister);
        ushort startAddress = holdingRegistersToRead.Min(r => r.Address);
        ushort endAddress = holdingRegistersToRead.Max(r => r.Address + r.Length - 1);
        ushort length = (ushort)(endAddress - startAddress + 1);

        // 一次性读取所有需要的寄存器
        var holdingRegisters = _modbusMaster.ReadHoldingRegisters(1, startAddress, length);
        
        // 调整数组索引（因为我们从startAddress开始读取）
        var adjustedRegisters = new ushort[endAddress + 1];
        Array.Copy(holdingRegisters, 0, adjustedRegisters, startAddress, holdingRegisters.Length);

        // 使用生成的解析器填充数据模型
        ProductionLineDataParser.ReadFromRegisters(data, adjustedRegisters, Array.Empty<ushort>(), Array.Empty<bool>(), Array.Empty<bool>());

        return data;
    }

    public void WriteData(ProductionLineData data)
    {
        // 使用映射表获取最大地址，预分配足够大的数组
        var maxAddress = ProductionLineDataMap.RegisterMap.Values
            .Where(r => r.RegisterType == RegisterType.HoldingRegister)
            .Max(r => r.Address + r.Length - 1);
        
        var holdingRegisters = new ushort[maxAddress + 1];

        // 使用生成的解析器将数据写入寄存器数组
        ProductionLineDataParser.WriteToRegisters(data, holdingRegisters);

        // 批量写入所有寄存器
        _modbusMaster.WriteMultipleRegisters(1, 0, holdingRegisters);
    }

    public void PrintRegisterMap()
    {
        Console.WriteLine("PLC寄存器映射表：");
        Console.WriteLine("==================");
        
        foreach (var info in ProductionLineDataMap.RegisterMap.Values.OrderBy(r => r.Address))
        {
            Console.WriteLine($"{info.PropertyName,-15} | {info.RegisterType,-15} | 0x{info.Address:X4} | 长度: {info.Length,2}");
        }
    }
}
```

## 四、新增功能说明

### 4.1 字符串类型支持

- **多编码支持**：支持 ASCII 和 UTF-8 两种常见编码
- **自动字节序处理**：根据特性指定的字节序正确解析字符串
- **自动截断**：自动截断到第一个空字符 ('\0')，符合 PLC 字符串的常见约定
- **自动填充**：写入时自动用空字符填充不足的长度
- **编译时检查**：字符串类型属性必须指定大于 0 的 Length 参数，否则编译报错

### 4.2 寄存器地址映射表

生成的`{ClassName}Map`类提供以下功能：

- **完整的元数据访问**：获取每个属性对应的所有寄存器信息
- **按类型查询**：快速获取指定类型的所有寄存器
- **只读字典**：线程安全的只读访问
- **调试友好**：`ModbusRegisterInfo`重写了`ToString()`方法，方便调试和日志输出

### 4.3 自动批量读取优化

使用映射表可以自动计算需要读取的寄存器范围，实现最优的批量读取策略：

- 自动找到最小起始地址和最大结束地址
- 一次性读取所有需要的寄存器，减少网络往返
- 自动调整数组索引，无需手动计算偏移量

## 六、最佳实践

1. **字符串长度选择**：根据 PLC 实际定义的字符串长度设置`Length`参数，不要过大
2. **编码选择**：优先使用 ASCII 编码，因为大多数 PLC 默认使用 ASCII；只有在需要中文等特殊字符时才使用 UTF-8
3. **批量读取**：始终使用映射表计算需要读取的寄存器范围，一次性读取所有数据
4. **重用数据模型**：在热路径中重用数据模型实例，避免频繁创建和销毁对象
5. **空值处理**：写入字符串时，`null`会被自动转换为空字符串，无需额外处理
