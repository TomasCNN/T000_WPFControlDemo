### 2.2 PDF 导出核心实现

csharp:

```c#
// IndustrialAlarmLog.xaml.cs
using System.IO;
using iTextSharp.text;
using iTextSharp.text.pdf;
using System.Windows.Documents;

public partial class IndustrialAlarmLog : UserControl
{
    // 公司信息配置（可根据实际情况修改）
    private const string CompanyName = "XX自动化科技有限公司";
    private const string SystemName = "工业生产线监控系统";

    /// <summary>
    /// 导出报警记录到PDF（工业合规格式）
    /// </summary>
    private void ExportToPdf(IEnumerable<AlarmRecord> records = null)
    {
        var exportRecords = records ?? _alarmRecords;
        if (!exportRecords.Any())
        {
            MessageBox.Show("没有可导出的报警记录", "提示", MessageBoxButton.OK, MessageBoxImage.Information);
            return;
        }

        var saveDialog = new SaveFileDialog
        {
            Filter = "PDF文件(*.pdf)|*.pdf",
            FileName = $"报警日志_{DateTime.Now:yyyyMMddHHmmss}"
        };

        if (saveDialog.ShowDialog() != true)
            return;

        try
        {
            // 创建PDF文档
            var document = new Document(PageSize.A4, 50, 50, 50, 50);
            PdfWriter.GetInstance(document, new FileStream(saveDialog.FileName, FileMode.Create));
            document.Open();

            // 1. 添加页眉
            var headerFont = FontFactory.GetFont(FontFactory.HELVETICA_BOLD, 16, BaseColor.BLACK);
            var subHeaderFont = FontFactory.GetFont(FontFactory.HELVETICA, 10, BaseColor.GRAY);
            
            var headerTable = new PdfPTable(1);
            headerTable.WidthPercentage = 100;
            headerTable.DefaultCell.Border = Rectangle.NO_BORDER;
            
            headerTable.AddCell(new Paragraph(CompanyName, headerFont) { Alignment = Element.ALIGN_CENTER });
            headerTable.AddCell(new Paragraph(SystemName, subHeaderFont) { Alignment = Element.ALIGN_CENTER });
            headerTable.AddCell(new Paragraph($"报警日志导出时间：{DateTime.Now:yyyy-MM-dd HH:mm:ss}", subHeaderFont) { Alignment = Element.ALIGN_CENTER });
            headerTable.AddCell(new Paragraph(" ")); // 空行
            
            document.Add(headerTable);

            // 2. 添加统计信息
            var stats = GenerateStatistics(exportRecords);
            var statsFont = FontFactory.GetFont(FontFactory.HELVETICA, 10, BaseColor.BLACK);
            
            var statsTable = new PdfPTable(4);
            statsTable.WidthPercentage = 100;
            statsTable.DefaultCell.Padding = 5;
            statsTable.DefaultCell.HorizontalAlignment = Element.ALIGN_CENTER;
            
            statsTable.AddCell(new Phrase("信息", FontFactory.GetFont(FontFactory.HELVETICA_BOLD, 10, BaseColor.BLUE)));
            statsTable.AddCell(new Phrase("警告", FontFactory.GetFont(FontFactory.HELVETICA_BOLD, 10, BaseColor.ORANGE)));
            statsTable.AddCell(new Phrase("错误", FontFactory.GetFont(FontFactory.HELVETICA_BOLD, 10, BaseColor.RED)));
            statsTable.AddCell(new Phrase("严重", FontFactory.GetFont(FontFactory.HELVETICA_BOLD, 10, BaseColor.WHITE, BaseColor.RED)));
            
            statsTable.AddCell(stats.InfoCount.ToString());
            statsTable.AddCell(stats.WarningCount.ToString());
            statsTable.AddCell(stats.ErrorCount.ToString());
            statsTable.AddCell(stats.CriticalCount.ToString());
            
            document.Add(statsTable);
            document.Add(new Paragraph(" ")); // 空行

            // 3. 添加报警记录表格
            var table = new PdfPTable(5);
            table.WidthPercentage = 100;
            table.SetWidths(new float[] { 8, 25, 12, 15, 40 });

            // 表头
            var headerCellFont = FontFactory.GetFont(FontFactory.HELVETICA_BOLD, 10, BaseColor.WHITE);
            var headerBgColor = new BaseColor(51, 122, 183); // 工业蓝
            
            AddPdfCell(table, "序号", headerCellFont, headerBgColor, Element.ALIGN_CENTER);
            AddPdfCell(table, "报警时间", headerCellFont, headerBgColor, Element.ALIGN_CENTER);
            AddPdfCell(table, "报警级别", headerCellFont, headerBgColor, Element.ALIGN_CENTER);
            AddPdfCell(table, "操作员", headerCellFont, headerBgColor, Element.ALIGN_CENTER);
            AddPdfCell(table, "报警信息", headerCellFont, headerBgColor, Element.ALIGN_CENTER);

            // 数据行
            var cellFont = FontFactory.GetFont(FontFactory.HELVETICA, 9, BaseColor.BLACK);
            int rowNum = 1;
            
            foreach (var record in exportRecords)
            {
                BaseColor rowColor = BaseColor.WHITE;
                BaseColor fontColor = BaseColor.BLACK;

                // 根据报警级别设置颜色
                switch (record.Level)
                {
                    case AlarmLevel.Warning:
                        fontColor = BaseColor.ORANGE;
                        break;
                    case AlarmLevel.Error:
                        fontColor = BaseColor.RED;
                        break;
                    case AlarmLevel.Critical:
                        rowColor = BaseColor.RED;
                        fontColor = BaseColor.WHITE;
                        break;
                }

                AddPdfCell(table, rowNum.ToString(), cellFont, rowColor, Element.ALIGN_CENTER, fontColor);
                AddPdfCell(table, record.Timestamp.ToString("yyyy-MM-dd HH:mm:ss.fff"), cellFont, rowColor, Element.ALIGN_LEFT, fontColor);
                AddPdfCell(table, record.Level.ToString(), cellFont, rowColor, Element.ALIGN_CENTER, fontColor);
                AddPdfCell(table, record.OperatorName, cellFont, rowColor, Element.ALIGN_CENTER, fontColor);
                AddPdfCell(table, record.Message, cellFont, rowColor, Element.ALIGN_LEFT, fontColor);
                
                rowNum++;
            }

            document.Add(table);

            // 4. 添加页脚
            var footerFont = FontFactory.GetFont(FontFactory.HELVETICA, 8, BaseColor.GRAY);
            var footer = new Paragraph($"第 {document.PageNumber} 页 | 导出时间：{DateTime.Now:yyyy-MM-dd HH:mm:ss}", footerFont);
            footer.Alignment = Element.ALIGN_RIGHT;
            document.Add(footer);

            document.Close();

            MessageBox.Show($"成功导出 {exportRecords.Count()} 条记录！\n文件：{saveDialog.FileName}", 
                "导出成功", MessageBoxButton.OK, MessageBoxImage.Information);
        }
        catch (Exception ex)
        {
            MessageBox.Show($"导出失败：{ex.Message}", "错误", MessageBoxButton.OK, MessageBoxImage.Error);
        }
    }

    // PDF单元格辅助方法
    private void AddPdfCell(PdfPTable table, string text, Font font, BaseColor bgColor, int alignment, BaseColor fontColor = null)
    {
        var cell = new PdfPCell(new Phrase(text, font));
        cell.BackgroundColor = bgColor;
        cell.HorizontalAlignment = alignment;
        cell.VerticalAlignment = Element.ALIGN_MIDDLE;
        cell.Padding = 5;
        
        if (fontColor != null)
        {
            ((Phrase)cell.Phrase).Font.Color = fontColor;
        }
        
        table.AddCell(cell);
    }

    // 重载GenerateStatistics方法，支持指定记录集
    private AlarmStatistics GenerateStatistics(IEnumerable<AlarmRecord> records)
    {
        var recordList = records.ToList();
        return new AlarmStatistics
        {
            TotalCount = recordList.Count,
            InfoCount = recordList.Count(r => r.Level == AlarmLevel.Info),
            WarningCount = recordList.Count(r => r.Level == AlarmLevel.Warning),
            ErrorCount = recordList.Count(r => r.Level == AlarmLevel.Error),
            CriticalCount = recordList.Count(r => r.Level == AlarmLevel.Critical),
            StartTime = recordList.Any() ? recordList.Min(r => r.Timestamp) : DateTime.Now,
            EndTime = recordList.Any() ? recordList.Max(r => r.Timestamp) : DateTime.Now
        };
    }

    private void BtnExportPdf_Click(object sender, RoutedEventArgs e)
    {
        ExportToPdf();
    }
}
```

## 三、功能二：实时报警声音提示

工业系统中，不同级别的报警需要**不同的声音提示**，严重报警应持续播放直到操作员确认。

### 3.1 添加报警声音资源

1. 在项目中创建`Sounds`文件夹
2. 添加以下 WAV 声音文件（建议使用系统自带或免费版权声音）：
   - `info.wav`：信息提示音（柔和短音）
   - `warning.wav`：警告提示音（中等长度）
   - `error.wav`：错误报警音（较长）
   - `critical.wav`：严重报警音（急促重复）
3. 将所有声音文件的**生成操作**设置为 "嵌入的资源"

### 3.2 声音报警管理器实现

csharp:

```c#
// Services/AlarmSoundManager.cs
using System;
using System.IO;
using System.Media;
using System.Reflection;
using System.Threading.Tasks;

namespace WpfApp.Services
{
    /// <summary>
    /// 报警声音管理器
    /// </summary>
    public class AlarmSoundManager
    {
        private static readonly Lazy<AlarmSoundManager> _instance = new(() => new AlarmSoundManager());
        public static AlarmSoundManager Instance => _instance.Value;

        private readonly SoundPlayer _infoPlayer;
        private readonly SoundPlayer _warningPlayer;
        private readonly SoundPlayer _errorPlayer;
        private readonly SoundPlayer _criticalPlayer;

        public bool SoundEnabled { get; set; } = true;

        private AlarmSoundManager()
        {
            // 从嵌入资源加载声音
            _infoPlayer = LoadSound("info.wav");
            _warningPlayer = LoadSound("warning.wav");
            _errorPlayer = LoadSound("error.wav");
            _criticalPlayer = LoadSound("critical.wav");
        }

        private SoundPlayer LoadSound(string fileName)
        {
            try
            {
                var assembly = Assembly.GetExecutingAssembly();
                var resourceName = $"WpfApp.Sounds.{fileName}";
                
                using var stream = assembly.GetManifestResourceStream(resourceName);
                if (stream == null)
                    return null;

                return new SoundPlayer(stream);
            }
            catch
            {
                return null;
            }
        }

        /// <summary>
        /// 播放对应级别的报警声音
        /// </summary>
        public void PlayAlarmSound(AlarmLevel level)
        {
            if (!SoundEnabled)
                return;

            Task.Run(() =>
            {
                try
                {
                    switch (level)
                    {
                        case AlarmLevel.Info:
                            _infoPlayer?.Play();
                            break;
                        case AlarmLevel.Warning:
                            _warningPlayer?.Play();
                            break;
                        case AlarmLevel.Error:
                            _errorPlayer?.Play();
                            break;
                        case AlarmLevel.Critical:
                            // 严重报警重复播放3次
                            for (int i = 0; i < 3; i++)
                            {
                                _criticalPlayer?.PlaySync();
                                Task.Delay(500).Wait();
                            }
                            break;
                    }
                }
                catch
                {
                    // 声音播放失败不影响系统运行
                }
            });
        }

        /// <summary>
        /// 停止所有声音
        /// </summary>
        public void StopAllSounds()
        {
            _infoPlayer?.Stop();
            _warningPlayer?.Stop();
            _errorPlayer?.Stop();
            _criticalPlayer?.Stop();
        }
    }
}
```

## 四、功能三：历史日志查询（SQLite 持久化存储）

历史日志查询需要**持久化存储**和**高效的分页查询**能力，我们使用轻量级的 SQLite 数据库，无需安装额外服务，适合工业设备部署。

### 4.1 数据库上下文与初始化

csharp:

```c#
// Data/AlarmLogDbContext.cs
using System;
using System.Collections.Generic;
using System.Data;
using Microsoft.Data.Sqlite;
using Dapper;

namespace WpfApp.Data
{
    /// <summary>
    /// 报警日志数据库上下文
    /// </summary>
    public class AlarmLogDbContext
    {
        private readonly string _connectionString;

        public AlarmLogDbContext(string dbPath = "AlarmLogs.db")
        {
            _connectionString = $"Data Source={dbPath}";
            InitializeDatabase();
        }

        private void InitializeDatabase()
        {
            using var connection = new SqliteConnection(_connectionString);
            connection.Open();

            // 创建报警日志表
            connection.Execute(@"
                CREATE TABLE IF NOT EXISTS AlarmLogs (
                    Id INTEGER PRIMARY KEY AUTOINCREMENT,
                    Level INTEGER NOT NULL,
                    Message TEXT NOT NULL,
                    OperatorName TEXT NOT NULL,
                    Timestamp DATETIME NOT NULL
                );

                CREATE INDEX IF NOT EXISTS IX_AlarmLogs_Timestamp ON AlarmLogs(Timestamp);
                CREATE INDEX IF NOT EXISTS IX_AlarmLogs_Level ON AlarmLogs(Level);
            ");
        }

        /// <summary>
        /// 保存报警记录
        /// </summary>
        public void SaveAlarm(AlarmRecord record)
        {
            using var connection = new SqliteConnection(_connectionString);
            connection.Execute(@"
                INSERT INTO AlarmLogs (Level, Message, OperatorName, Timestamp)
                VALUES (@Level, @Message, @OperatorName, @Timestamp)",
                new { record.Level, record.Message, record.OperatorName, record.Timestamp });
        }

        /// <summary>
        /// 分页查询报警记录
        /// </summary>
        public (List<AlarmRecord> Records, int TotalCount) QueryAlarms(
            DateTime? startTime = null,
            DateTime? endTime = null,
            AlarmLevel? level = null,
            string operatorName = null,
            string keyword = null,
            int pageNumber = 1,
            int pageSize = 50)
        {
            using var connection = new SqliteConnection(_connectionString);
            
            // 构建查询条件
            var conditions = new List<string>();
            var parameters = new DynamicParameters();

            if (startTime.HasValue)
            {
                conditions.Add("Timestamp >= @StartTime");
                parameters.Add("@StartTime", startTime.Value);
            }

            if (endTime.HasValue)
            {
                conditions.Add("Timestamp <= @EndTime");
                parameters.Add("@EndTime", endTime.Value);
            }

            if (level.HasValue)
            {
                conditions.Add("Level = @Level");
                parameters.Add("@Level", level.Value);
            }

            if (!string.IsNullOrEmpty(operatorName))
            {
                conditions.Add("OperatorName LIKE @OperatorName");
                parameters.Add("@OperatorName", $"%{operatorName}%");
            }

            if (!string.IsNullOrEmpty(keyword))
            {
                conditions.Add("Message LIKE @Keyword");
                parameters.Add("@Keyword", $"%{keyword}%");
            }

            string whereClause = conditions.Any() 
                ? $"WHERE {string.Join(" AND ", conditions)}" 
                : string.Empty;

            // 查询总数
            int totalCount = connection.ExecuteScalar<int>(
                $"SELECT COUNT(*) FROM AlarmLogs {whereClause}", parameters);

            // 查询分页数据
            var records = connection.Query<AlarmRecord>(
                $@"SELECT * FROM AlarmLogs {whereClause}
                   ORDER BY Timestamp DESC
                   LIMIT @PageSize OFFSET @Offset",
                new 
                { 
                    PageSize = pageSize, 
                    Offset = (pageNumber - 1) * pageSize,
                    parameters
                }).ToList();

            return (records, totalCount);
        }

        /// <summary>
        /// 清理过期日志
        /// </summary>
        public int CleanupOldLogs(int retentionDays)
        {
            using var connection = new SqliteConnection(_connectionString);
            var cutoffDate = DateTime.Now.AddDays(-retentionDays);
            
            return connection.Execute(
                "DELETE FROM AlarmLogs WHERE Timestamp < @CutoffDate",
                new { CutoffDate = cutoffDate });
        }
    }
```

#### 代码部分 (`HistoryLogQueryWindow.xaml.cs`)

csharp:

```c#
using System;
using System.Linq;
using System.Windows;
using WpfApp.Data;

namespace WpfApp
{
    public partial class HistoryLogQueryWindow : Window
    {
        private readonly AlarmLogDbContext _dbContext;
        private int _currentPage = 1;
        private int _totalPages = 0;
        private const int PageSize = 50;

        public HistoryLogQueryWindow()
        {
            InitializeComponent();
            _dbContext = new AlarmLogDbContext();
            
            // 默认查询最近7天
            dpStartTime.SelectedDate = DateTime.Now.AddDays(-7);
            dpEndTime.SelectedDate = DateTime.Now;
        }

        private void BtnQuery_Click(object sender, RoutedEventArgs e)
        {
            _currentPage = 1;
            ExecuteQuery();
        }

        private void ExecuteQuery()
        {
            try
            {
                // 解析查询条件
                DateTime? startTime = dpStartTime.SelectedDate;
                DateTime? endTime = dpEndTime.SelectedDate?.AddDays(1).AddTicks(-1); // 包含当天最后一刻
                
                AlarmLevel? level = null;
                if (cboLevel.SelectedItem is ComboBoxItem selectedLevel && selectedLevel.Content.ToString() != "全部")
                {
                    if (Enum.TryParse(selectedLevel.Content.ToString(), out AlarmLevel parsedLevel))
                    {
                        level = parsedLevel;
                    }
                }

                string operatorName = txtOperator.Text.Trim();
                string keyword = txtKeyword.Text.Trim();

                // 执行查询
                var (records, totalCount) = _dbContext.QueryAlarms(
                    startTime, endTime, level, operatorName, keyword, _currentPage, PageSize);

                // 更新UI
                dgResults.ItemsSource = records;
                _totalPages = (int)Math.Ceiling((double)totalCount / PageSize);
                
                txtPageInfo.Text = $"第 {_currentPage} / {_totalPages} 页，共 {totalCount} 条记录";
                
                // 更新分页按钮状态
                btnPrevPage.IsEnabled = _currentPage > 1;
                btnNextPage.IsEnabled = _currentPage < _totalPages;
            }
            catch (Exception ex)
            {
                MessageBox.Show($"查询失败：{ex.Message}", "错误", MessageBoxButton.OK, MessageBoxImage.Error);
            }
        }

        private void BtnReset_Click(object sender, RoutedEventArgs e)
        {
            dpStartTime.SelectedDate = DateTime.Now.AddDays(-7);
            dpEndTime.SelectedDate = DateTime.Now;
            cboLevel.SelectedIndex = 0;
            txtOperator.Clear();
            txtKeyword.Clear();
            dgResults.ItemsSource = null;
            txtPageInfo.Text = string.Empty;
        }

        private void BtnPrevPage_Click(object sender, RoutedEventArgs e)
        {
            if (_currentPage > 1)
            {
                _currentPage--;
                ExecuteQuery();
            }
        }

        private void BtnNextPage_Click(object sender, RoutedEventArgs e)
        {
            if (_currentPage < _totalPages)
            {
                _currentPage++;
                ExecuteQuery();
            }
        }

        private void BtnExportExcel_Click(object sender, RoutedEventArgs e)
        {
            var records = dgResults.ItemsSource.Cast<AlarmRecord>().ToList();
            if (!records.Any())
            {
                MessageBox.Show("没有可导出的记录", "提示", MessageBoxButton.OK, MessageBoxImage.Information);
                return;
            }

            // 复用主控件的Excel导出方法
            var tempControl = new IndustrialAlarmLog();
            tempControl.ExportToExcel(records);
        }

        private void BtnExportPdf_Click(object sender, RoutedEventArgs e)
        {
            var records = dgResults.ItemsSource.Cast<AlarmRecord>().ToList();
            if (!records.Any())
            {
                MessageBox.Show("没有可导出的记录", "提示", MessageBoxButton.OK, MessageBoxImage.Information);
                return;
            }

            // 复用主控件的PDF导出方法
            var tempControl = new IndustrialAlarmLog();
            tempControl.ExportToPdf(records);
        }

        private void BtnClose_Click(object sender, RoutedEventArgs e)
        {
            this.Close();
        }
    }
}
```

## 五、最终功能演示与使用说明

### 5.1 PDF 导出功能

1. 点击 "导出 PDF" 按钮
2. 选择保存位置
3. 导出的 PDF 包含：
   - 公司和系统信息页眉
   - 报警统计摘要
   - 带颜色标记的报警记录表格
   - 页码和导出时间页脚

### 5.2 实时声音报警

- 不同级别报警播放不同的提示音
- 严重报警重复播放 3 次
- 点击 "声音" 按钮可以切换静音 / 非静音状态
- 声音异步播放，不阻塞 UI 线程

### 5.3 历史日志查询

1. 点击 "历史查询" 按钮打开查询窗口
2. 设置查询条件：时间范围、报警级别、操作员、关键词
3. 点击 "查询" 按钮获取结果
4. 支持分页浏览查询结果
5. 可以将查询结果导出为 Excel 或 PDF

## 六、工业级优化与最佳实践

### 6.1 数据库性能优化

- 为常用查询字段创建索引（时间、级别）
- 实现自动日志清理功能，避免数据库无限增长
- 批量插入大量报警记录时使用事务
- 数据库操作全部在后台线程执行，不阻塞 UI

### 6.2 声音报警优化

- 支持自定义不同级别的报警声音
- 严重报警可以配置为持续播放直到确认
- 声音文件嵌入程序集，避免文件丢失
- 异常处理：声音播放失败不影响系统运行

### 6.3 PDF 导出优化

- 支持导出查询结果而不是全部日志
- 大量记录时自动分页
- 工业标准格式，满足合规性要求
- 可配置公司信息和页眉页脚
