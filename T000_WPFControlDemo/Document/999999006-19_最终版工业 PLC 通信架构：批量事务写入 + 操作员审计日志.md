## 一、新增功能一：原子性批量事务写入

工业场景中经常需要同时写入多个寄存器，要求 "要么全部成功，要么全部失败"。我们实现了一个支持**自动回滚**的事务机制，解决了 Modbus 协议本身不支持事务的问题。

### 1.1 定义事务特性与数据结构

```c#
// IndustrialAutomation.Core/Services/ModbusTcpPlcService.cs
using IndustrialAutomation.Core.Models;

namespace IndustrialAutomation.Core.Services
{
    public class ModbusTcpPlcService : IPlcCommunicationService
    {
        // ... 原有代码保持不变 ...

        [LogExecution("批量写入保持寄存器", logParameters: false)]
        [Retry(maxRetries: 2, initialDelayMs: 100)]
        [Timeout(3000)]
        [PlcCommand(0x10, "批量写入保持寄存器")]
        [Alarm("PLC批量写入失败", AlarmLevel.Error, debounceSeconds: 60)]
        [TransactionalWrite(timeoutMs: 5000)]
        public void WriteMultipleRegisters(IEnumerable<RegisterWriteOperation> operations)
        {
            if (_modbusMaster == null || !_tcpClient.Connected)
                throw new InvalidOperationException("PLC未连接");

            var opList = operations.ToList();
            if (opList.Count == 0)
                return;

            // 优化：合并连续地址为单次写入
            var blocks = MergeContinuousAddresses(opList);
            
            foreach (var block in blocks)
            {
                _modbusMaster.WriteMultipleRegisters(
                    _slaveId, 
                    block.StartAddress, 
                    block.Values.ToArray());
            }
        }

        [LogExecution("批量读取保持寄存器", logReturnValue: false)]
        [Retry(maxRetries: 2, initialDelayMs: 100)]
        [Timeout(2000)]
        [PlcCommand(0x03, "批量读取保持寄存器")]
        public Dictionary<ushort, ushort> ReadMultipleRegisters(IEnumerable<ushort> addresses)
        {
            if (_modbusMaster == null || !_tcpClient.Connected)
                throw new InvalidOperationException("PLC未连接");

            var addrList = addresses.Distinct().OrderBy(a => a).ToList();
            var result = new Dictionary<ushort, ushort>();
            
            if (addrList.Count == 0)
                return result;

            // 优化：合并连续地址为单次读取
            var blocks = MergeContinuousAddresses(addrList);
            
            foreach (var block in blocks)
            {
                var values = _modbusMaster.ReadHoldingRegisters(
                    _slaveId, 
                    block.StartAddress, 
                    block.Length);
                
                for (int i = 0; i < values.Length; i++)
                {
                    result[(ushort)(block.StartAddress + i)] = values[i];
                }
            }

            return result;
        }

        #region 辅助方法
        /// <summary>
        /// 合并连续地址为写入块
        /// </summary>
        private List<RegisterBlock> MergeContinuousAddresses(List<RegisterWriteOperation> operations)
        {
            if (operations.Count == 0)
                return new List<RegisterBlock>();

            var sortedOps = operations.OrderBy(o => o.Address).ToList();
            var blocks = new List<RegisterBlock>();
            var currentBlock = new RegisterBlock
            {
                StartAddress = sortedOps[0].Address,
                Values = new List<ushort> { sortedOps[0].Value }
            };

            for (int i = 1; i < sortedOps.Count; i++)
            {
                if (sortedOps[i].Address == currentBlock.StartAddress + currentBlock.Values.Count)
                {
                    currentBlock.Values.Add(sortedOps[i].Value);
                }
                else
                {
                    blocks.Add(currentBlock);
                    currentBlock = new RegisterBlock
                    {
                        StartAddress = sortedOps[i].Address,
                        Values = new List<ushort> { sortedOps[i].Value }
                    };
                }
            }

            blocks.Add(currentBlock);
            return blocks;
        }

        /// <summary>
        /// 合并连续地址为读取块
        /// </summary>
        private List<ReadBlock> MergeContinuousAddresses(List<ushort> addresses)
        {
            if (addresses.Count == 0)
                return new List<ReadBlock>();

            var blocks = new List<ReadBlock>();
            var currentBlock = new ReadBlock
            {
                StartAddress = addresses[0],
                Length = 1
            };

            for (int i = 1; i < addresses.Count; i++)
            {
                if (addresses[i] == addresses[i - 1] + 1)
                {
                    currentBlock.Length++;
                }
                else
                {
                    blocks.Add(currentBlock);
                    currentBlock = new ReadBlock
                    {
                        StartAddress = addresses[i],
                        Length = 1
                    };
                }
            }

            blocks.Add(currentBlock);
            return blocks;
        }

        private class RegisterBlock
        {
            public ushort StartAddress { get; set; }
            public List<ushort> Values { get; set; }
        }

        private class ReadBlock
        {
            public ushort StartAddress { get; set; }
            public ushort Length { get; set; }
        }
        #endregion
    }
}
```

### 1.3 实现事务装饰器

csharp:

```c#
// IndustrialAutomation.Core/Decorators/TransactionDecorator.cs
using System;
using System.Collections.Generic;
using System.Linq;
using System.Reflection;
using System.Threading;
using IndustrialAutomation.Core.Attributes;
using IndustrialAutomation.Core.Models;
using Microsoft.Extensions.Logging;

namespace IndustrialAutomation.Core.Decorators
{
    /// <summary>
    /// 事务写入装饰器：提供原子性写入保证，失败自动回滚
    /// </summary>
    public class TransactionDecorator : PlcServiceDecoratorBase
    {
        private readonly ILogger<TransactionDecorator> _logger;

        public TransactionDecorator(IPlcCommunicationService innerService, 
            ILogger<TransactionDecorator> logger) 
            : base(innerService)
        {
            _logger = logger;
        }

        public override void WriteMultipleRegisters(IEnumerable<RegisterWriteOperation> operations)
        {
            var attribute = GetAttribute<TransactionalWriteAttribute>(nameof(WriteMultipleRegisters));
            if (attribute == null)
            {
                base.WriteMultipleRegisters(operations);
                return;
            }

            var opList = operations.ToList();
            if (opList.Count == 0)
                return;

            // 步骤1：读取所有要写入的寄存器的原始值（用于回滚）
            _logger.LogDebug("开始事务写入，准备读取原始值用于回滚");
            var addresses = opList.Select(o => o.Address).ToList();
            var originalValues = base.ReadMultipleRegisters(addresses);

            // 填充原始值到操作列表
            foreach (var op in opList)
            {
                if (originalValues.TryGetValue(op.Address, out var value))
                {
                    op.OriginalValue = value;
                }
                else
                {
                    _logger.LogWarning("无法读取寄存器 {Address} 的原始值，回滚可能不完整", op.Address);
                }
            }

            // 步骤2：执行写入操作
            using var cts = new CancellationTokenSource(attribute.TimeoutMs);
            try
            {
                _logger.LogDebug("执行事务写入操作");
                base.WriteMultipleRegisters(opList);
                _logger.LogInformation("事务写入成功，共写入 {Count} 个寄存器", opList.Count);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "事务写入失败，准备回滚");

                if (attribute.EnableRollback)
                {
                    // 步骤3：回滚到原始值
                    try
                    {
                        _logger.LogWarning("开始回滚事务");
                        var rollbackOps = opList.Select(o => 
                            new RegisterWriteOperation(o.Address, o.OriginalValue)).ToList();
                        
                        using var rollbackCts = new CancellationTokenSource(attribute.RollbackTimeoutMs);
                        base.WriteMultipleRegisters(rollbackOps);
                        _logger.LogInformation("事务回滚成功");
                    }
                    catch (Exception rollbackEx)
                    {
                        _logger.LogCritical(rollbackEx, "事务回滚失败！系统可能处于不一致状态");
                        throw new InvalidOperationException("事务写入失败且回滚失败，系统数据可能不一致", ex);
                    }
                }

                throw new InvalidOperationException("事务写入失败，已回滚到原始状态", ex);
            }
        }
    }
}
```

```c#
    {
        public string LogDirectory { get; set; } = "AuditLogs";
        public string FileNameFormat { get; set; } = "audit_{0:yyyyMMdd}.log";
        public int RetentionDays { get; set; } = 90;
    }

    /// 文件审计日志存储实现
    /// </summary>
    public class FileAuditLogStore : IAuditLogStore
    {

        private readonly object _lock = new();

        public FileAuditLogStore(IOptions<FileAuditLogOptions> options)
        {
            _options = options.Value;
            Directory.CreateDirectory(_options.LogDirectory);
        }

        public async Task SaveLogAsync(AuditLogEntry logEntry)
        {
            var json = JsonSerializer.Serialize(logEntry, new JsonSerializerOptions
            {
                WriteIndented = false
            });

            var fileName = string.Format(_options.FileNameFormat, DateTime.Now);
            var filePath = Path.Combine(_options.LogDirectory, fileName);

            // 线程安全写入
            lock (_lock)
            {
                File.AppendAllText(filePath, json + Environment.NewLine);
            }

            await Task.CompletedTask;
        }
    }
```

### 2.3 实现审计装饰器

csharp:

```c#
// IndustrialAutomation.Core/Decorators/AuditDecorator.cs
using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Linq;
using System.Reflection;
using System.Text.Json;
using System.Threading.Tasks;
using IndustrialAutomation.Core.Attributes;
using IndustrialAutomation.Core.Audit;
using IndustrialAutomation.Core.Models;
using Microsoft.AspNetCore.Http;
using Microsoft.Extensions.Logging;

namespace IndustrialAutomation.Core.Decorators
{
    /// <summary>
    /// 审计装饰器：自动记录操作审计日志
    /// </summary>
    public class AuditDecorator : PlcServiceDecoratorBase
    {
        private readonly ILogger<AuditDecorator> _logger;
        private readonly IAuditLogStore _auditLogStore;
        private readonly IHttpContextAccessor _httpContextAccessor;

        public AuditDecorator(IPlcCommunicationService innerService,
            ILogger<AuditDecorator> logger,
            IAuditLogStore auditLogStore,
            IHttpContextAccessor httpContextAccessor)
            : base(innerService)
        {
            _logger = logger;
            _auditLogStore = auditLogStore;
            _httpContextAccessor = httpContextAccessor;
        }

        public override void WriteSingleRegister(ushort address, ushort value)
        {
            var attribute = GetAttribute<AuditAttribute>(nameof(WriteSingleRegister));
            if (attribute == null)
            {
                base.WriteSingleRegister(address, value);
                return;
            }

            ExecuteWithAudit(
                () => base.WriteSingleRegister(address, value),
                attribute,
                nameof(WriteSingleRegister),
                new object[] { address, value });
        }

        public override void WriteMultipleRegisters(IEnumerable<RegisterWriteOperation> operations)
        {
            var attribute = GetAttribute<AuditAttribute>(nameof(WriteMultipleRegisters));
            if (attribute == null)
            {
                base.WriteMultipleRegisters(operations);
                return;
            }

            var opList = operations.ToList();
            ExecuteWithAudit(
                () => base.WriteMultipleRegisters(opList),
                attribute,
                nameof(WriteMultipleRegisters),
                new object[] { opList });
        }

        /// <summary>
        /// 执行带审计的操作
        /// </summary>
        private void ExecuteWithAudit(Action operation, AuditAttribute attribute, 
            string methodName, object[] parameters)
        {
            var stopwatch = Stopwatch.StartNew();
            var logEntry = new AuditLogEntry
            {
                OperationName = attribute.OperationName,
                MethodName = methodName,
                OperatorId = GetCurrentUserId(),
                OperatorName = GetCurrentUserName(),
                ClientIpAddress = GetClientIpAddress()
            };

            try
            {
                // 记录参数
                if (attribute.LogParameters)
                {
                    logEntry.Parameters = JsonSerializer.Serialize(parameters);
                }

                // 记录原始值
                if (attribute.LogValueChanges)
                {
                    logEntry.OriginalValues = GetOriginalValues(methodName, parameters);
                }

                // 执行操作
                operation();

                logEntry.Success = true;
                _logger.LogInformation("操作 {OperationName} 成功", attribute.OperationName);
            }
            catch (Exception ex)
            {
                logEntry.Success = false;
                logEntry.ErrorMessage = ex.Message;
                _logger.LogError(ex, "操作 {OperationName} 失败", attribute.OperationName);
                throw;
            }
            finally
            {
                stopwatch.Stop();
                logEntry.DurationMs = stopwatch.ElapsedMilliseconds;

                // 异步保存审计日志（不阻塞主线程）
                _ = SaveAuditLogAsync(logEntry);
            }
        }

        #region 辅助方法
        private string GetCurrentUserId()
        {
            return _httpContextAccessor.HttpContext?.User?.Identity?.Name ?? "System";
        }

        private string GetCurrentUserName()
        {
            return _httpContextAccessor.HttpContext?.User?.Identity?.Name ?? "系统";
        }

        private string GetClientIpAddress()
        {
            return _httpContextAccessor.HttpContext?.Connection?.RemoteIpAddress?.ToString() ?? "Unknown";
        }

        private string GetOriginalValues(string methodName, object[] parameters)
        {
            try
            {
                return methodName switch
                {
                    nameof(WriteSingleRegister) => GetOriginalValuesForSingleWrite(parameters),
                    nameof(WriteMultipleRegisters) => GetOriginalValuesForMultipleWrite(parameters),
                    _ => null
                };
            }
            catch (Exception ex)
            {
                _logger.LogWarning(ex, "获取原始值失败");
                return null;
            }
        }

        private string GetOriginalValuesForSingleWrite(object[] parameters)
        {
            var address = (ushort)parameters[0];
            var originalValue = base.ReadHoldingRegisters(address, 1)[0];
            return JsonSerializer.Serialize(new Dictionary<ushort, ushort> { { address, originalValue } });
        }

        private string GetOriginalValuesForMultipleWrite(object[] parameters)
        {
            var operations = (IEnumerable<RegisterWriteOperation>)parameters[0];
            var addresses = operations.Select(o => o.Address).ToList();
            var originalValues = base.ReadMultipleRegisters(addresses);
            return JsonSerializer.Serialize(originalValues);
        }

        private async Task SaveAuditLogAsync(AuditLogEntry logEntry)
        {
            try
            {
                await _auditLogStore.SaveLogAsync(logEntry);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "保存审计日志失败");
            }
        }
        #endregion
    }
}
```

## 六、核心设计要点

### 6.1 事务机制设计

- **先读后写**：写入前先读取所有要修改的寄存器原始值
- **自动回滚**：写入失败时自动恢复到原始状态
- **回滚保护**：回滚过程有独立的超时控制
- **部分失败处理**：即使部分回滚失败也会明确记录

### 6.2 审计日志设计

- **自动记录**：无需手动编写审计代码
- **完整信息**：记录操作员、时间、IP、参数、原始值、新值、耗时
- **异步保存**：日志保存不阻塞业务操作
- **可扩展存储**：可以轻松替换为数据库存储
- **合规性**：满足 ISO 9001、GMP 等工业标准要求

### 6.3 装饰器最终执行顺序

plaintext:

```tex
调用方 → AlarmDecorator → AuditDecorator → LoggingDecorator → RetryDecorator → TransactionDecorator → TimeoutDecorator → 实际服务
```

- **AlarmDecorator 在最外层**：捕获所有异常并发送报警
- **AuditDecorator 在 AlarmDecorator 内层**：记录所有操作，包括失败的操作
- **TransactionDecorator 在 RetryDecorator 内层**：确保每次重试都是一个独立的事务
- **TimeoutDecorator 在最内层**：直接控制实际通信操作的超时
