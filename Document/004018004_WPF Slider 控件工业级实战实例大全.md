```c#
// ViewModel代码
public class TemperatureViewModel : INotifyPropertyChanged
{
    private double _targetTemperature = 25.0;
    public double TargetTemperature
    {
        get => _targetTemperature;
        set
        {
            if (_targetTemperature != value)
            {
                _targetTemperature = value;
                OnPropertyChanged();
                OnPropertyChanged(nameof(TemperatureStatus));
                
                // 温度报警
                if (value > 180)
                {
                    AlarmService.Instance.TriggerAlarm(
                        AlarmType.HighTemperature, 
                        $"加热炉温度过高：{value:F1}℃");
                }
                else if (value < 10)
                {
                    AlarmService.Instance.TriggerAlarm(
                        AlarmType.LowTemperature, 
                        $"加热炉温度过低：{value:F1}℃");
                }
                
                // 更新设备
                DeviceService.Instance.SetTemperature(value);
            }
        }
    }

    public TemperatureStatus TemperatureStatus
    {
        get
        {
            if (TargetTemperature > 180) return TemperatureStatus.High;
            if (TargetTemperature < 10) return TemperatureStatus.Low;
            return TemperatureStatus.Normal;
        }
    }

    public event PropertyChangedEventHandler PropertyChanged;
    protected virtual void OnPropertyChanged([CallerMemberName] string propertyName = null)
    {
        PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(propertyName));
    }
}

// 温度状态转换器
public class TemperatureStatusToColorConverter : IValueConverter
{
    public object Convert(object value, Type targetType, object parameter, CultureInfo culture)
    {
        if (value is TemperatureStatus status)
        {
            switch (status)
            {
                case TemperatureStatus.Normal:
                    return new SolidColorBrush(Color.FromRgb(0, 122, 204));
                case TemperatureStatus.High:
                    return new SolidColorBrush(Color.FromRgb(232, 17, 35));
                case TemperatureStatus.Low:
                    return new SolidColorBrush(Color.FromRgb(0, 150, 136));
                default:
                    return new SolidColorBrush(Color.FromRgb(45, 45, 48));
            }
        }
        return new SolidColorBrush(Color.FromRgb(45, 45, 48));
    }

    public object ConvertBack(object value, Type targetType, object parameter, CultureInfo culture)
    {
        throw new NotImplementedException();
    }
}

public enum TemperatureStatus
{
    Normal,
    High,
    Low
}
```

### 4.1 双滑块范围调节器

**应用场景**：温度上下限、压力范围、时间区间等需要同时调节两个值的场景

```c#
/// <summary>
/// 工业双滑块范围调节器
/// </summary>
public class RangeSlider : Control
{
    // 依赖属性
    public static readonly DependencyProperty MinimumProperty =
        DependencyProperty.Register(nameof(Minimum), typeof(double), typeof(RangeSlider),
            new PropertyMetadata(0.0, OnRangeChanged));

    public static readonly DependencyProperty MaximumProperty =
        DependencyProperty.Register(nameof(Maximum), typeof(double), typeof(RangeSlider),
            new PropertyMetadata(100.0, OnRangeChanged));

    public static readonly DependencyProperty LowerValueProperty =
        DependencyProperty.Register(nameof(LowerValue), typeof(double), typeof(RangeSlider),
            new PropertyMetadata(0.0, OnLowerValueChanged));

    public static readonly DependencyProperty UpperValueProperty =
        DependencyProperty.Register(nameof(UpperValue), typeof(double), typeof(RangeSlider),
            new PropertyMetadata(100.0, OnUpperValueChanged));

    public static readonly DependencyProperty TickFrequencyProperty =
        DependencyProperty.Register(nameof(TickFrequency), typeof(double), typeof(RangeSlider),
            new PropertyMetadata(10.0));

    // 属性
    public double Minimum
    {
        get => (double)GetValue(MinimumProperty);
        set => SetValue(MinimumProperty, value);
    }

    public double Maximum
    {
        get => (double)GetValue(MaximumProperty);
        set => SetValue(MaximumProperty, value);
    }

    public double LowerValue
    {
        get => (double)GetValue(LowerValueProperty);
        set => SetValue(LowerValueProperty, value);
    }

    public double UpperValue
    {
        get => (double)GetValue(UpperValueProperty);
        set => SetValue(UpperValueProperty, value);
    }

    public double TickFrequency
    {
        get => (double)GetValue(TickFrequencyProperty);
        set => SetValue(TickFrequencyProperty, value);
    }

    // 值变更回调
    private static void OnRangeChanged(DependencyObject d, DependencyPropertyChangedEventArgs e)
    {
        var slider = (RangeSlider)d;
        slider.UpdateVisuals();
    }

    private static void OnLowerValueChanged(DependencyObject d, DependencyPropertyChangedEventArgs e)
    {
        var slider = (RangeSlider)d;
        if (slider.LowerValue > slider.UpperValue)
        {
            slider.LowerValue = slider.UpperValue;
        }
        slider.UpdateVisuals();
    }

    private static void OnUpperValueChanged(DependencyObject d, DependencyPropertyChangedEventArgs e)
    {
        var slider = (RangeSlider)d;
        if (slider.UpperValue < slider.LowerValue)
        {
            slider.UpperValue = slider.LowerValue;
        }
        slider.UpdateVisuals();
    }

    static RangeSlider()
    {
        DefaultStyleKeyProperty.OverrideMetadata(typeof(RangeSlider),
            new FrameworkPropertyMetadata(typeof(RangeSlider)));
    }

    public override void OnApplyTemplate()
    {
        base.OnApplyTemplate();
        UpdateVisuals();
    }

    private void UpdateVisuals()
    {
        // 更新滑块位置和选中范围
        if (Template.FindName("LowerThumb", this) is Thumb lowerThumb)
        {
            double position = (LowerValue - Minimum) / (Maximum - Minimum) * ActualWidth;
            Canvas.SetLeft(lowerThumb, position - 10);
        }

        if (Template.FindName("UpperThumb", this) is Thumb upperThumb)
        {
            double position = (UpperValue - Minimum) / (Maximum - Minimum) * ActualWidth;
            Canvas.SetLeft(upperThumb, position - 10);
        }

        if (Template.FindName("SelectionRange", this) is Border selectionRange)
        {
            double left = (LowerValue - Minimum) / (Maximum - Minimum) * ActualWidth;
            double width = (UpperValue - LowerValue) / (Maximum - Minimum) * ActualWidth;
            selectionRange.Margin = new Thickness(left, 0, 0, 0);
            selectionRange.Width = width;
        }
    }
}
```

## 五、工业开发最佳实践

1. **永远开启`IsSnapToTickEnabled`**：确保工艺参数只能取允许的刻度值，避免无效参数
2. **合理设置步长**：
   - `SmallChange`：精细调节步长（如温度 0.1℃，压力 0.01MPa）
   - `LargeChange`：快速调节步长（如温度 10℃，压力 0.1MPa）
3. **使用自动提示**：拖动时显示当前值，提升操作效率
4. **触摸屏优化**：滑块尺寸至少 40x40px，增大触摸区域
5. **节流处理**：对`ValueChanged`事件添加节流，避免频繁设备通信
6. **参数越界保护**：利用`RangeBase`的自动值强制特性，无需手动校验
7. **支持键盘操作**：为常用操作添加键盘快捷键，提升操作效率
8. **统一全局样式**：整个应用使用相同的 Slider 样式，保持工业界面一致性
