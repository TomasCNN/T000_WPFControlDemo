### 1.3 实现 Excel 导出器

csharp:

```c#
// IndustrialAutomation.Core/Audit/ExcelAuditLogExporter.cs
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Text;
using System.Threading.Tasks;
using IndustrialAutomation.Core.Models;
using Microsoft.Extensions.Options;
using OfficeOpenXml;
using OfficeOpenXml.Style;

namespace IndustrialAutomation.Core.Audit
{
    /// <summary>
    /// Excel审计日志导出器
    /// </summary>
    public class ExcelAuditLogExporter : IAuditLogExporter
    {
        private readonly IAuditLogStore _auditLogStore;
        private readonly ExcelExportOptions _options;

        public ExcelAuditLogExporter(IAuditLogStore auditLogStore, 
            IOptions<ExcelExportOptions> options)
        {
            _auditLogStore = auditLogStore;
            _options = options.Value;
            
            // 设置EPPlus非商业用途许可
            ExcelPackage.LicenseContext = LicenseContext.NonCommercial;
        }

        public async Task<byte[]> ExportToExcelAsync(AuditLogQuery query)
        {
            // 临时修改分页参数以获取所有符合条件的数据
            var originalPageSize = query.PageSize;
            query.PageSize = _options.MaxExportRows;
            
            var result = await _auditLogStore.QueryLogsAsync(query);
            
            using var package = new ExcelPackage();
            var worksheet = package.Workbook.Worksheets.Add(_options.SheetName);
            
            // 设置列标题
            var headers = new[] 
            {
                "日志ID", "操作时间", "操作员ID", "操作员名称", "操作名称", 
                "方法名", "是否成功", "客户端IP", "耗时(ms)", "错误信息",
                "参数", "原始值", "新值"
            };
            
            for (int i = 0; i < headers.Length; i++)
            {
                worksheet.Cells[1, i + 1].Value = headers[i];
                worksheet.Cells[1, i + 1].Style.Font.Bold = true;
                worksheet.Cells[1, i + 1].Style.Fill.PatternType = ExcelFillStyle.Solid;
                worksheet.Cells[1, i + 1].Style.Fill.BackgroundColor.SetColor(System.Drawing.Color.LightGray);
            }
            
            // 填充数据
            for (int i = 0; i < result.Items.Count; i++)
            {
                var log = result.Items[i];
                int row = i + 2;
                
                worksheet.Cells[row, 1].Value = log.Id.ToString();
                worksheet.Cells[row, 2].Value = log.Timestamp.ToString("yyyy-MM-dd HH:mm:ss.fff");
                worksheet.Cells[row, 3].Value = log.OperatorId;
                worksheet.Cells[row, 4].Value = log.OperatorName;
                worksheet.Cells[row, 5].Value = log.OperationName;
                worksheet.Cells[row, 6].Value = log.MethodName;
                worksheet.Cells[row, 7].Value = log.Success ? "成功" : "失败";
                worksheet.Cells[row, 8].Value = log.ClientIpAddress;
                worksheet.Cells[row, 9].Value = log.DurationMs;
                worksheet.Cells[row, 10].Value = log.ErrorMessage;
                worksheet.Cells[row, 11].Value = log.Parameters;
                worksheet.Cells[row, 12].Value = log.OriginalValues;
                worksheet.Cells[row, 13].Value = log.NewValues;
                
                // 设置失败行的背景色
                if (!log.Success)
                {
                    worksheet.Cells[row, 1, row, 13].Style.Fill.PatternType = ExcelFillStyle.Solid;
                    worksheet.Cells[row, 1, row, 13].Style.Fill.BackgroundColor.SetColor(System.Drawing.Color.LightPink);
                }
            }
            
            // 自动调整列宽
            worksheet.Cells.AutoFitColumns();
            
            // 设置边框
            using var range = worksheet.Cells[1, 1, result.Items.Count + 1, headers.Length];
            range.Style.Border.Top.Style = ExcelBorderStyle.Thin;
            range.Style.Border.Bottom.Style = ExcelBorderStyle.Thin;
            range.Style.Border.Left.Style = ExcelBorderStyle.Thin;
            range.Style.Border.Right.Style = ExcelBorderStyle.Thin;
            
            // 冻结首行
            worksheet.View.FreezePanes(2, 1);
            
            return package.GetAsByteArray();
        }

        public async Task<byte[]> ExportToCsvAsync(AuditLogQuery query)
        {
            var originalPageSize = query.PageSize;
            query.PageSize = _options.MaxExportRows;
            
            var result = await _auditLogStore.QueryLogsAsync(query);
            var sb = new StringBuilder();
            
            // 写入标题行
            sb.AppendLine("日志ID,操作时间,操作员ID,操作员名称,操作名称,方法名,是否成功,客户端IP,耗时(ms),错误信息");
            
            // 写入数据行
            foreach (var log in result.Items)
            {
                sb.AppendLine($"{log.Id},{log.Timestamp:yyyy-MM-dd HH:mm:ss.fff},{log.OperatorId},{log.OperatorName},{log.OperationName},{log.MethodName},{(log.Success ? "成功" : "失败")},{log.ClientIpAddress},{log.DurationMs},{log.ErrorMessage}");
            }
            
            return Encoding.UTF8.GetBytes(sb.ToString());
        }
    }
}
```

### 2.2 实现后台清理服务

csharp:

```c#
// IndustrialAutomation.Core/Services/AuditLogCleanupService.cs
using System;
using System.Linq;
using System.Threading;
using System.Threading.Tasks;
using IndustrialAutomation.Core.Audit;
using IndustrialAutomation.Core.Data;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Hosting;
using Microsoft.Extensions.Logging;
using Microsoft.Extensions.Options;

namespace IndustrialAutomation.Core.Services
{
    /// <summary>
    /// 审计日志自动清理后台服务
    /// </summary>
    public class AuditLogCleanupService : BackgroundService
    {
        private readonly IServiceProvider _serviceProvider;
        private readonly ILogger<AuditLogCleanupService> _logger;
        private readonly AuditLogCleanupOptions _options;

        public AuditLogCleanupService(IServiceProvider serviceProvider,
            ILogger<AuditLogCleanupService> logger,
            IOptions<AuditLogCleanupOptions> options)
        {
            _serviceProvider = serviceProvider;
            _logger = logger;
            _options = options.Value;
        }

        protected override async Task ExecuteAsync(CancellationToken stoppingToken)
        {
            if (!_options.EnableAutoCleanup)
            {
                _logger.LogInformation("审计日志自动清理已禁用");
                return;
            }

            _logger.LogInformation("审计日志自动清理服务已启动，保留天数：{RetentionDays}天，清理间隔：{Interval}小时",
                _options.RetentionDays, _options.CleanupIntervalHours);

            while (!stoppingToken.IsCancellationRequested)
            {
                try
                {
                    await CleanupOldLogsAsync(stoppingToken);
                }
                catch (Exception ex)
                {
                    _logger.LogError(ex, "执行审计日志清理时发生错误");
                }

                // 等待下一次清理时间
                await Task.Delay(TimeSpan.FromHours(_options.CleanupIntervalHours), stoppingToken);
            }

            _logger.LogInformation("审计日志自动清理服务已停止");
        }

        private async Task CleanupOldLogsAsync(CancellationToken stoppingToken)
        {
            var cutoffDate = DateTime.UtcNow.AddDays(-_options.RetentionDays);
            _logger.LogInformation("开始清理 {CutoffDate} 之前的审计日志", cutoffDate);

            using var scope = _serviceProvider.CreateScope();
            var dbContext = scope.ServiceProvider.GetRequiredService<AuditDbContext>();

            int totalDeleted = 0;
            int batchSize = _options.MaxRowsPerCleanup;

            while (!stoppingToken.IsCancellationRequested)
            {
                // 分批删除，避免一次性删除过多数据导致数据库锁表
                var oldLogs = await dbContext.AuditLogs
                    .Where(e => e.TimestampUtc < cutoffDate)
                    .OrderBy(e => e.TimestampUtc)
                    .Take(batchSize)
                    .ToListAsync(stoppingToken);

                if (oldLogs.Count == 0)
                    break;

                dbContext.AuditLogs.RemoveRange(oldLogs);
                await dbContext.SaveChangesAsync(stoppingToken);

                totalDeleted += oldLogs.Count;
                _logger.LogDebug("已删除 {Count} 条审计日志", oldLogs.Count);

                // 短暂延迟，减少数据库压力
                await Task.Delay(100, stoppingToken);
            }

            if (totalDeleted > 0)
            {
                _logger.LogInformation("审计日志清理完成，共删除 {TotalDeleted} 条记录", totalDeleted);
            }
            else
            {
                _logger.LogInformation("没有需要清理的审计日志");
            }
        }
    }
}
```

## 五、核心设计要点

### 5.1 Excel 导出设计要点

1. **性能优化**：限制最大导出行数，避免导出过多数据导致内存溢出
2. **格式美化**：自动添加表头、边框、颜色区分失败操作
3. **用户友好**：自动调整列宽，冻结首行，便于查看
4. **多格式支持**：同时支持 Excel 和 CSV 两种导出格式
5. **许可合规**：使用非商业用途免费的 EPPlus 版本

### 5.2 自动清理设计要点

1. **分批删除**：每次最多删除 10000 条，避免锁表和数据库压力
2. **可配置**：保留天数、清理间隔、批量大小都可通过配置文件调整
3. **异常处理**：清理失败不会影响系统正常运行，并有详细日志记录
4. **后台执行**：使用[ASP.NET](https://link.wtturl.cn/?target=https%3A%2F%2FASP.NET&scene=im&aid=582478&lang=zh) Core 后台服务，不阻塞主线程
5. **安全终止**：支持优雅关闭，清理过程中可以安全停止服务

## 六、扩展建议

1. **添加清理前备份功能**：在删除日志前自动导出并备份到文件或对象存储
2. **添加归档功能**：将超过保留期限的日志归档到冷存储，而不是直接删除
3. **添加清理通知**：清理完成后发送邮件通知管理员
4. **添加手动清理 API**：允许管理员手动触发清理操作
5. **支持分布式部署**：使用分布式锁确保在多实例部署时只有一个实例执行清理
