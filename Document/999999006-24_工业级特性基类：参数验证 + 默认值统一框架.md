# 999999006-24_工业级特性基类：参数验证 + 默认值统一框架

**专门面向工业自动化场景的特性基类**，封装了所有常见的参数验证逻辑和默认值处理机制。使用这个基类可以大幅简化自定义特性的开发，消除重复代码，同时提供统一、清晰的错误信息。

## 一、特性基类完整实现

csharp:

```c#
using System;
using System.Linq;
using System.Reflection;

namespace IndustrialAutomation.Core.Attributes
{
    /// <summary>
    /// 工业级特性基类：提供统一的参数验证和默认值处理
    /// </summary>
    public abstract class ValidationAttributeBase : Attribute
    {
        /// <summary>
        /// 特性名称（用于错误信息）
        /// </summary>
        protected virtual string AttributeName => GetType().Name.Replace("Attribute", "");

        #region 核心验证方法

        /// <summary>
        /// 验证对象不为null
        /// </summary>
        /// <param name="value">要验证的值</param>
        /// <param name="parameterName">参数名称</param>
        /// <exception cref="ArgumentNullException">值为null时抛出</exception>
        protected void ValidateNotNull(object value, string parameterName)
        {
            if (value == null)
            {
                throw new ArgumentNullException(
                    paramName: parameterName,
                    message: $"[{AttributeName}] 特性参数 '{parameterName}' 不能为null");
            }
        }

        /// <summary>
        /// 验证字符串不为空或空白
        /// </summary>
        /// <param name="value">要验证的字符串</param>
        /// <param name="parameterName">参数名称</param>
        /// <param name="allowEmpty">是否允许空字符串</param>
        /// <exception cref="ArgumentException">字符串为空或空白时抛出</exception>
        protected void ValidateNotEmpty(string value, string parameterName, bool allowEmpty = false)
        {
            ValidateNotNull(value, parameterName);

            if (!allowEmpty && string.IsNullOrWhiteSpace(value))
            {
                throw new ArgumentException(
                    message: $"[{AttributeName}] 特性参数 '{parameterName}' 不能为空白字符串",
                    paramName: parameterName);
            }
        }

        /// <summary>
        /// 验证数值在指定范围内
        /// </summary>
        /// <typeparam name="T">数值类型</typeparam>
        /// <param name="value">要验证的值</param>
        /// <param name="minValue">最小值（包含）</param>
        /// <param name="maxValue">最大值（包含）</param>
        /// <param name="parameterName">参数名称</param>
        /// <exception cref="ArgumentOutOfRangeException">值超出范围时抛出</exception>
        protected void ValidateRange<T>(T value, T minValue, T maxValue, string parameterName)
            where T : IComparable<T>
        {
            if (value.CompareTo(minValue) < 0 || value.CompareTo(maxValue) > 0)
            {
                throw new ArgumentOutOfRangeException(
                    paramName: parameterName,
                    actualValue: value,
                    message: $"[{AttributeName}] 特性参数 '{parameterName}' 必须在 [{minValue}, {maxValue}] 范围内");
            }
        }

        /// <summary>
        /// 验证数值大于等于指定值
        /// </summary>
        protected void ValidateGreaterThanOrEqual<T>(T value, T minValue, string parameterName)
            where T : IComparable<T>
        {
            if (value.CompareTo(minValue) < 0)
            {
                throw new ArgumentOutOfRangeException(
                    paramName: parameterName,
                    actualValue: value,
                    message: $"[{AttributeName}] 特性参数 '{parameterName}' 必须大于等于 {minValue}");
            }
        }

        /// <summary>
        /// 验证数值小于等于指定值
        /// </summary>
        protected void ValidateLessThanOrEqual<T>(T value, T maxValue, string parameterName)
            where T : IComparable<T>
        {
            if (value.CompareTo(maxValue) > 0)
            {
                throw new ArgumentOutOfRangeException(
                    paramName: parameterName,
                    actualValue: value,
                    message: $"[{AttributeName}] 特性参数 '{parameterName}' 必须小于等于 {maxValue}");
            }
        }

        /// <summary>
        /// 验证数组不为null且至少包含一个元素
        /// </summary>
        /// <typeparam name="T">数组元素类型</typeparam>
        /// <param name="array">要验证的数组</param>
        /// <param name="parameterName">参数名称</param>
        /// <param name="allowEmpty">是否允许空数组</param>
        protected void ValidateArrayNotEmpty<T>(T[] array, string parameterName, bool allowEmpty = false)
        {
            ValidateNotNull(array, parameterName);

            if (!allowEmpty && array.Length == 0)
            {
                throw new ArgumentException(
                    message: $"[{AttributeName}] 特性参数 '{parameterName}' 数组不能为空",
                    paramName: parameterName);
            }
        }

        /// <summary>
        /// 验证数组中所有元素都不为null
        /// </summary>
        protected void ValidateArrayElementsNotNull<T>(T[] array, string parameterName)
            where T : class
        {
            ValidateArrayNotEmpty(array, parameterName);

            if (array.Any(item => item == null))
            {
                throw new ArgumentException(
                    message: $"[{AttributeName}] 特性参数 '{parameterName}' 数组中不能包含null元素",
                    paramName: parameterName);
            }
        }

        /// <summary>
        /// 验证枚举值是有效的
        /// </summary>
        /// <typeparam name="TEnum">枚举类型</typeparam>
        /// <param name="value">要验证的枚举值</param>
        /// <param name="parameterName">参数名称</param>
        protected void ValidateEnumValue<TEnum>(TEnum value, string parameterName)
            where TEnum : struct, Enum
        {
            if (!Enum.IsDefined(typeof(TEnum), value))
            {
                throw new ArgumentException(
                    message: $"[{AttributeName}] 特性参数 '{parameterName}' 不是有效的 {typeof(TEnum).Name} 枚举值",
                    paramName: parameterName);
            }
        }

        /// <summary>
        /// 验证类型实现了指定接口
        /// </summary>
        /// <param name="type">要验证的类型</param>
        /// <param name="interfaceType">接口类型</param>
        /// <param name="parameterName">参数名称</param>
        protected void ValidateImplementsInterface(Type type, Type interfaceType, string parameterName)
        {
            ValidateNotNull(type, parameterName);
            ValidateNotNull(interfaceType, nameof(interfaceType));

            if (!interfaceType.IsAssignableFrom(type))
            {
                throw new ArgumentException(
                    message: $"[{AttributeName}] 特性参数 '{parameterName}' 类型 {type.Name} 必须实现 {interfaceType.Name} 接口",
                    paramName: parameterName);
            }
        }

        #endregion

        #region 默认值处理

        /// <summary>
        /// 如果值为null则返回默认值
        /// </summary>
        protected T GetValueOrDefault<T>(T value, T defaultValue)
        {
            return value == null ? defaultValue : value;
        }

        /// <summary>
        /// 如果字符串为空或空白则返回默认值
        /// </summary>
        protected string GetStringOrDefault(string value, string defaultValue)
        {
            return string.IsNullOrWhiteSpace(value) ? defaultValue : value;
        }

        /// <summary>
        /// 如果数组为null则返回空数组
        /// </summary>
        protected T[] GetArrayOrDefault<T>(T[] array)
        {
            return array ?? Array.Empty<T>();
        }

        #endregion

        #region 辅助方法

        /// <summary>
        /// 获取当前特性应用到的代码元素（用于高级验证）
        /// </summary>
        /// <typeparam name="T">代码元素类型</typeparam>
        /// <returns>应用特性的代码元素</returns>
        protected T GetAppliedElement<T>() where T : MemberInfo
        {
            // 注意：这个方法只能在特性被反射读取后调用
            // 用于需要根据应用目标进行验证的场景
            return null;
        }

        #endregion
    }
}
```

## 二、使用基类重构工业场景特性

下面展示如何使用这个基类重构我们之前定义的所有工业特性，你会发现代码变得极其简洁和统一。

### 2.1 Modbus 寄存器映射特性

csharp:

```c#
[AttributeUsage(AttributeTargets.Property, Inherited = false)]
public sealed class ModbusRegisterAttribute : ValidationAttributeBase
{
    // 构造函数：必填参数
    public ModbusRegisterAttribute(ushort address, RegisterType registerType)
    {
        // 参数验证（一行代码搞定）
        ValidateRange(address, (ushort)0, (ushort)65535, nameof(address));
        ValidateEnumValue(registerType, nameof(registerType));

        Address = address;
        RegisterType = registerType;
    }

    // 必填参数
    public ushort Address { get; }
    public RegisterType RegisterType { get; }

    // 可选参数（带默认值）
    public int Length { get; set; } = 2;
    public Endianness Endianness { get; set; } = Endianness.BigEndian;
    public double ScaleFactor { get; set; } = 1.0;
    public int BitPosition { get; set; } = -1;
}
```

### 2.2 设备驱动特性

csharp:

[AttributeUsage(AttributeTargets.Class, Inherited = false)]
public sealed class DeviceDriverAttribute : ValidationAttributeBase
{
    public DeviceDriverAttribute(string deviceType, string manufacturer, 
        Type protocol, params string[] supportedModels)
    {
        // 参数验证
        ValidateNotEmpty(deviceType, nameof(deviceType));
        ValidateNotEmpty(manufacturer, nameof(manufacturer));
        ValidateImplementsInterface(protocol, typeof(ICommunicationProtocol), nameof(protocol));
        ValidateArrayNotEmpty(supportedModels, nameof(supportedModels));
        ValidateArrayElementsNotNull(supportedModels, nameof(supportedModels));

        DeviceType = deviceType;
        Manufacturer = manufacturer;
        Protocol = protocol;
        SupportedModels = supportedModels;
    }

    // 必填参数
    public string DeviceType { get; }
    public string Manufacturer { get; }
    public Type Protocol { get; }
    public string[] SupportedModels { get; }

    // 可选参数

    public bool SupportsHotSwap { get; set; } = false;
    public int DefaultTimeoutMs { get; set; } = 1000;
    public int MaxRetries { get; set; } = 3;
}

## 三、高级特性：条件验证与依赖验证

基类还支持更复杂的**条件验证**和**参数依赖验证**，满足更复杂的特性设计需求。

### 3.1 条件验证示例

csharp:

```c#
[AttributeUsage(AttributeTargets.Method, Inherited = false)]
public sealed class TimeoutAttribute : ValidationAttributeBase
{
    public TimeoutAttribute(int timeoutMs)
    {
        ValidateGreaterThanOrEqual(timeoutMs, 1, nameof(timeoutMs));

        TimeoutMs = timeoutMs;
    }

    public int TimeoutMs { get; }

    private bool _enableCancellation;
    public bool EnableCancellation
    {
        get => _enableCancellation;
        set
        {
            _enableCancellation = value;
            
            // 条件验证：如果启用取消，则必须提供取消令牌参数
            if (value && CancellationTokenParameterIndex < 0)
            {
                throw new InvalidOperationException(
                    $"[{AttributeName}] 启用取消功能时必须指定 CancellationTokenParameterIndex");
            }
        }
    }

    public int CancellationTokenParameterIndex { get; set; } = -1;
}
```

### 3.2 参数依赖验证示例

csharp:

```c#
[AttributeUsage(AttributeTargets.Class, Inherited = false)]
public sealed class PlcDataModelAttribute : ValidationAttributeBase
{
    public PlcDataModelAttribute(byte defaultSlaveId = 1)
    {
        ValidateRange(defaultSlaveId, (byte)1, (byte)247, nameof(defaultSlaveId));

        DefaultSlaveId = defaultSlaveId;
    }

    public byte DefaultSlaveId { get; }

    private int _timeoutMs = 1000;
    public int TimeoutMs
    {
        get => _timeoutMs;
        set
        {
            ValidateGreaterThanOrEqual(value, 1, nameof(TimeoutMs));
            
            // 依赖验证：超时时间必须大于重试间隔
            if (value <= RetryDelayMs)
            {
                throw new ArgumentException(
                    $"[{AttributeName}] 超时时间必须大于重试间隔时间", 
                    nameof(TimeoutMs));
            }

            _timeoutMs = value;
        }
    }

    private int _retryDelayMs = 100;
    public int RetryDelayMs
    {
        get => _retryDelayMs;
        set
        {
            ValidateGreaterThanOrEqual(value, 0, nameof(RetryDelayMs));
            
            // 依赖验证：重试间隔必须小于超时时间
            if (value >= TimeoutMs)
            {
                throw new ArgumentException(
                    $"[{AttributeName}] 重试间隔时间必须小于超时时间", 
                    nameof(RetryDelayMs));
            }

            _retryDelayMs = value;
        }
    }

    public int RetryCount { get; set; } = 3;
}
```

## 五、基类的核心优势

### 5.1 消除重复代码

所有参数验证逻辑都封装在基类中，子类只需要一行代码即可完成验证，消除了 90% 以上的重复代码。

### 5.2 统一错误格式

所有验证失败的异常都使用统一的格式，包含特性名称、参数名称和错误原因，便于调试和问题定位。

### 5.3 提高开发效率

开发新特性时，只需要关注业务逻辑，不需要重复编写参数验证代码，大幅提高开发效率。

### 5.4 提高代码质量

统一的验证机制确保了所有特性都有一致的参数验证，避免了因忘记验证参数导致的潜在 bug。

### 5.5 易于扩展

基类采用模块化设计，可以轻松添加新的验证方法，满足不同场景的需求。

## 六、最佳实践

### 6.1 验证时机

所有参数验证都应该在**构造函数**中执行，确保特性实例创建时就处于有效状态。

### 6.2 必填与可选

- 将**必须提供的核心参数**放在构造函数中
- 将**可选的配置参数**作为公共属性，并提供合理的默认值

### 6.3 验证粒度

- 对每个参数进行独立验证
- 提供精确的错误信息，指出哪个参数出了什么问题

### 6.4 性能考虑

- 验证逻辑应该是轻量级的，避免复杂计算
- 不要在特性构造函数中执行 IO 操作或网络请求

### 6.5 异常处理

在读取特性的地方捕获验证异常，并提供友好的错误提示：

csharp:

```c#
try
{
    var attribute = property.GetCustomAttribute<ModbusRegisterAttribute>();
}
catch (ArgumentException ex)
{
    Console.WriteLine($"特性配置错误：{ex.Message}");
    // 记录日志或采取其他处理措施
}
```
