csharp:

```c#
// ViewModel 部分
public class MainViewModel : INotifyPropertyChanged
{
    private double _conveyorSpeed = 1.2;
    public double ConveyorSpeed
    {
        get => _conveyorSpeed;
        set { _conveyorSpeed = value; OnPropertyChanged(); }
    }

    private double _accelerationTime = 0.5;
    public double AccelerationTime
    {
        get => _accelerationTime;
        set { _accelerationTime = value; OnPropertyChanged(); }
    }

    private double _decelerationTime = 0.5;
    public double DecelerationTime
    {
        get => _decelerationTime;
        set { _decelerationTime = value; OnPropertyChanged(); }
    }

    // 自动调整位置回调（解决超出屏幕问题）
    public CustomPopupPlacementCallback AutoAdjustPlacement => (popupSize, targetSize, offset) =>
    {
        var screen = Screen.FromPoint(System.Windows.Forms.Cursor.Position);
        var screenBounds = new Rect(screen.WorkingArea.Left, screen.WorkingArea.Top,
                                    screen.WorkingArea.Width, screen.WorkingArea.Height);
        
        Point targetPos = paramBtn.PointToScreen(new Point(0, 0));
        double x = targetPos.X;
        double y = targetPos.Y + targetSize.Height + 5;

        // 超出下边界则显示在上方
        if (y + popupSize.Height > screenBounds.Bottom)
            y = targetPos.Y - popupSize.Height - 5;

        return new[] { new CustomPopupPlacement(new Point(x - targetPos.X, y - targetPos.Y), PopupPrimaryAxis.Vertical) };
    };

    public ICommand ApplyParamsCommand => new RelayCommand(() =>
    {
        // 写入参数到PLC
        PlcClient.Write("DB1.DBD0", ConveyorSpeed);
        PlcClient.Write("DB1.DBD4", AccelerationTime);
        PlcClient.Write("DB1.DBD8", DecelerationTime);
        
        // 自动关闭Popup
        ClosePopupCommand.Execute(null);
    });

    public ICommand ClosePopupCommand => new RelayCommand(() =>
    {
        // 关闭Popup
        paramPopup.IsOpen = false;
    });
}
```

### 核心特点

- 点击外部自动关闭（`StaysOpen="False"`）
- 自动调整位置避免超出屏幕
- 支持 MVVM 数据绑定
- 工业风格设计，大按钮高对比度

csharp:

```c#
// 后台代码（处理鼠标悬停逻辑）
private readonly DispatcherTimer _hideTimer;

public MainWindow()
{
    InitializeComponent();
    DataContext = new MainViewModel();
    
    // 初始化隐藏计时器（延迟300ms关闭，防止误触发）
    _hideTimer = new DispatcherTimer();
    _hideTimer.Interval = TimeSpan.FromMilliseconds(300);
    _hideTimer.Tick += (s, e) =>
    {
        deviceDetailPopup.IsOpen = false;
        _hideTimer.Stop();
    };

    // 鼠标进入显示
    deviceIndicator.MouseEnter += (s, e) =>
    {
        _hideTimer.Stop();
        // 实时更新设备数据
        ((MainViewModel)DataContext).UpdateDeviceData();
        deviceDetailPopup.IsOpen = true;
    };

    // 鼠标离开延迟关闭
    deviceIndicator.MouseLeave += (s, e) =>
    {
        _hideTimer.Start();
    };

    // 鼠标进入Popup时取消关闭
    deviceDetailPopup.MouseEnter += (s, e) =>
    {
        _hideTimer.Stop();
    };

    // 鼠标离开Popup时延迟关闭
    deviceDetailPopup.MouseLeave += (s, e) =>
    {
        _hideTimer.Start();
    };
}
```

### 核心特点

- 延迟关闭防止误触发
- 鼠标进入 Popup 时保持打开
- 深色工业风格，高对比度
- 实时更新设备数据

```c#
// 拖动功能实现
private Point _lastMousePos;
private bool _isDragging;

private void Toolbar_MouseLeftButtonDown(object sender, MouseButtonEventArgs e)
{
    _lastMousePos = e.GetPosition(this);
    _isDragging = true;
    Mouse.Capture((IInputElement)sender);
}

protected override void OnMouseMove(MouseEventArgs e)
{
    base.OnMouseMove(e);
    
    if (_isDragging)
    {
        Point currentPos = e.GetPosition(this);
        double deltaX = currentPos.X - _lastMousePos.X;
        double deltaY = currentPos.Y - _lastMousePos.Y;

        // 更新Popup位置
        floatingToolbar.HorizontalOffset += deltaX;
        floatingToolbar.VerticalOffset += deltaY;

        _lastMousePos = currentPos;
    }
}

protected override void OnMouseLeftButtonUp(MouseButtonEventArgs e)
{
    base.OnMouseLeftButtonUp(e);
    
    if (_isDragging)
    {
        _isDragging = false;
        Mouse.Capture(null);
    }
}
```

### 核心特点

- 可自由拖动位置
- 一直保持打开状态
- 不占用主界面空间

## 七、工业场景最佳实践

1. **永远使用`CustomPopupPlacementCallback`**：确保 Popup 在多显示器环境下不会超出屏幕边界
2. **重要弹窗设置`Topmost="True"`**：报警弹窗必须置顶，防止被其他窗口遮挡
3. **复用 Popup 实例**：不要频繁创建和销毁 Popup，复用同一个实例提升性能
4. **及时释放资源**：关闭 Popup 时设置`Child = null`，防止内存泄漏
5. **避免嵌套 Popup**：Popup 内部不要再放 Popup，会导致焦点和位置问题
6. **合理使用`StaysOpen`**：
   - 临时弹窗（参数设置、右键菜单）：`StaysOpen="False"`
   - 重要弹窗（报警、确认）：`StaysOpen="True"`
