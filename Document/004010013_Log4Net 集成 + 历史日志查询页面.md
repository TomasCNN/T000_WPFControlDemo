## 一、TextBlock 工具类与 Log4Net 深度集成

### 集成优势

- **统一日志接口**：业务代码只需要用 `ILog` 记录日志，自动实现「UI 实时显示 + 文件持久化 + 数据库存储」
- **Log4Net 强大能力**：支持按日期 / 大小自动分割文件、多级别过滤、多输出源（文件 / 数据库 / 邮件）
- **保留原有优势**：TextBlock 工具类的线程安全、自动裁剪、节流更新、颜色分级全部保留

### 步骤 4：实现自定义 TextBlockAppender

在 `Industrial.Controls` 命名空间下新建 `TextBlockAppender.cs`：

csharp:

```c#
using log4net.Appender;
using log4net.Core;
using System;
using System.Windows;
using System.Windows.Controls;

namespace Industrial.Controls
{
    /// <summary>
    /// Log4Net 自定义 Appender：将日志转发到 TextBlock 显示
    /// </summary>
    public class TextBlockAppender : AppenderSkeleton
    {
        // 全局事件：所有日志都会通过这个事件广播
        public static event Action<LogLevel, string> LogReceived;

        protected override void Append(LoggingEvent loggingEvent)
        {
            // 将 Log4Net 级别转换为我们的日志级别
            LogLevel level = ConvertLogLevel(loggingEvent.Level);
            string message = RenderLoggingEvent(loggingEvent);

            // 在 UI 线程触发事件
            Application.Current.Dispatcher.InvokeAsync(() =>
            {
                LogReceived?.Invoke(level, message);
            });
        }

        // Log4Net 级别转换
        private LogLevel ConvertLogLevel(Level level)
        {
            if (level == Level.Debug) return LogLevel.Debug;
            if (level == Level.Info) return LogLevel.Info;
            if (level == Level.Warn) return LogLevel.Warning;
            if (level == Level.Error) return LogLevel.Error;
            if (level == Level.Fatal) return LogLevel.Fatal;
            return LogLevel.Info;
        }
    }
}
```

### 步骤 5：修改 TextBlock 工具类，支持订阅全局日志

在 `TextBlockLogExtensions.cs` 中添加以下代码：

csharp:

```c#
public static class TextBlockLogExtensions
{
    // 在类顶部添加静态构造函数，订阅全局日志事件
    static TextBlockLogExtensions()
    {
        // 订阅 Log4Net 广播的所有日志
        TextBlockAppender.LogReceived += (level, message) =>
        {
            // 广播给所有订阅了全局日志的 TextBlock
            GlobalLogReceived?.Invoke(level, message);
        };
    }

    // 全局日志广播事件
    public static event Action<LogLevel, string> GlobalLogReceived;

    // 添加订阅全局日志的扩展方法
    public static void SubscribeGlobalLog(this TextBlock textBlock)
    {
        if (!_managers.TryGetValue(textBlock, out var manager))
        {
            textBlock.InitLog();
            manager = _managers[textBlock];
        }

        GlobalLogReceived += manager.AddLog;
    }

    // 在 LogManager 类中添加对应的 AddLog 重载
    private class LogManager : IDisposable
    {
        // 添加这个重载方法
        public void AddLog(LogLevel level, string message)
        {
            AddLog(level, message);
        }

        // 原有代码不变...
    }
}
```

### 2. 后台代码实现

csharp:

```c#
using Industrial.Controls;
using Microsoft.Win32;
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Text;
using System.Windows;
using System.Windows.Controls;

namespace Industrial.Views
{
    public partial class LogQueryPage : Page
    {
        private const int PAGE_SIZE = 100; // 每页显示100条
        private List<LogEntry> _allLogs = new List<LogEntry>();
        private List<LogEntry> _filteredLogs = new List<LogEntry>();
        private int _currentPage = 1;

        public LogQueryPage()
        {
            InitializeComponent();
            // 默认查询最近7天的日志
            dpStartDate.SelectedDate = DateTime.Now.AddDays(-7);
            dpEndDate.SelectedDate = DateTime.Now;
        }

        // 查询按钮
        private void BtnQuery_Click(object sender, RoutedEventArgs e)
        {
            try
            {
                Mouse.OverrideCursor = Cursors.Wait;
                
                // 1. 读取所有日志文件
                _allLogs = LogFileReader.ReadLogs(dpStartDate.SelectedDate.Value, dpEndDate.SelectedDate.Value);
                
                // 2. 应用过滤条件
                ApplyFilters();
                
                // 3. 显示第一页
                _currentPage = 1;
                UpdatePageDisplay();
            }
            catch (Exception ex)
            {
                MessageBox.Show($"查询失败：{ex.Message}", "错误", MessageBoxButton.OK, MessageBoxImage.Error);
            }
            finally
            {
                Mouse.OverrideCursor = null;
            }
        }

        // 应用过滤条件
        private void ApplyFilters()
        {
            var query = _allLogs.AsEnumerable();

            // 级别过滤
            if (!chkDebug.IsChecked.Value) query = query.Where(l => l.Level != LogLevel.Debug);
            if (!chkInfo.IsChecked.Value) query = query.Where(l => l.Level != LogLevel.Info);
            if (!chkWarning.IsChecked.Value) query = query.Where(l => l.Level != LogLevel.Warning);
            if (!chkError.IsChecked.Value) query = query.Where(l => l.Level != LogLevel.Error);
            if (!chkFatal.IsChecked.Value) query = query.Where(l => l.Level != LogLevel.Fatal);

            // 关键字过滤
            string keyword = txtKeyword.Text.Trim();
            if (!string.IsNullOrEmpty(keyword))
            {
                query = query.Where(l => l.Message.Contains(keyword, StringComparison.OrdinalIgnoreCase));
            }

            _filteredLogs = query.ToList();
        }

        // 更新分页显示
        private void UpdatePageDisplay()
        {
            int totalPages = Math.Max(1, (int)Math.Ceiling(_filteredLogs.Count / (double)PAGE_SIZE));
            _currentPage = Math.Clamp(_currentPage, 1, totalPages);

            // 获取当前页数据
            var pageData = _filteredLogs
                .Skip((_currentPage - 1) * PAGE_SIZE)
                .Take(PAGE_SIZE)
                .ToList();

            dgLogs.ItemsSource = pageData;
            txtPageInfo.Text = $"第 {_currentPage} 页 / 共 {totalPages} 页（总计 {_filteredLogs.Count} 条）";

            // 更新按钮状态
            btnPrev.IsEnabled = _currentPage > 1;
            btnNext.IsEnabled = _currentPage < totalPages;
        }

        // 导出CSV
        private void BtnExport_Click(object sender, RoutedEventArgs e)
        {
            if (_filteredLogs.Count == 0)
            {
                MessageBox.Show("没有可导出的日志", "提示", MessageBoxButton.OK, MessageBoxImage.Information);
                return;
            }

            var saveDialog = new SaveFileDialog
            {
                Filter = "CSV文件 (*.csv)|*.csv|所有文件 (*.*)|*.*",
                FileName = $"日志_{DateTime.Now:yyyyMMddHHmmss}.csv"
            };

            if (saveDialog.ShowDialog() == true)
            {
                try
                {
                    Mouse.OverrideCursor = Cursors.Wait;
                    
                    var sb = new StringBuilder();
                    sb.AppendLine("时间,级别,线程,消息");
                    
                    foreach (var log in _filteredLogs)
                    {
                        sb.AppendLine($"\"{log.Time:yyyy-MM-dd HH:mm:ss.fff}\",\"{log.Level}\",\"{log.ThreadId}\",\"{log.Message.Replace("\"", "\"\"")}\"");
                    }

                    File.WriteAllText(saveDialog.FileName, sb.ToString(), Encoding.UTF8);
                    MessageBox.Show("导出成功", "提示", MessageBoxButton.OK, MessageBoxImage.Information);
                }
                catch (Exception ex)
                {
                    MessageBox.Show($"导出失败：{ex.Message}", "错误", MessageBoxButton.OK, MessageBoxImage.Error);
                }
                finally
                {
                    Mouse.OverrideCursor = null;
                }
            }
        }

        // 清空日志
        private void BtnClear_Click(object sender, RoutedEventArgs e)
        {
            if (MessageBox.Show("确定要清空所有日志吗？此操作不可恢复", "确认", MessageBoxButton.YesNo, MessageBoxImage.Warning) == MessageBoxResult.Yes)
            {
                try
                {
                    // 删除所有日志文件
                    string logDir = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "Logs");
                    if (Directory.Exists(logDir))
                    {
                        foreach (var file in Directory.GetFiles(logDir, "*.log"))
                        {
                            File.Delete(file);
                        }
                    }

                    _allLogs.Clear();
                    _filteredLogs.Clear();
                    UpdatePageDisplay();
                    
                    MessageBox.Show("日志已清空", "提示", MessageBoxButton.OK, MessageBoxImage.Information);
                }
                catch (Exception ex)
                {
                    MessageBox.Show($"清空失败：{ex.Message}", "错误", MessageBoxButton.OK, MessageBoxImage.Error);
                }
            }
        }

        // 上一页
        private void BtnPrev_Click(object sender, RoutedEventArgs e)
        {
            _currentPage--;
            UpdatePageDisplay();
        }

        // 下一页
        private void BtnNext_Click(object sender, RoutedEventArgs e)
        {
            _currentPage++;
            UpdatePageDisplay();
        }
    }
}
```

### 3. 日志文件读取器实现

在 `Industrial.Controls` 命名空间下新建 `LogFileReader.cs`：

csharp:

```c#
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Text.RegularExpressions;

namespace Industrial.Controls
{
    /// <summary>
    /// Log4Net 日志文件读取器
    /// </summary>
    public static class LogFileReader
    {
        // 日志格式正则表达式（与 log4net.config 中的 conversionPattern 对应）
        private static readonly Regex _logRegex = new Regex(
            @"^(?<Time>\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}\.\d{3}) \[T(?<Thread>\d+)\] \[(?<Level>\w+)\] (?<Message>.*)$",
            RegexOptions.Compiled);

        /// <summary>
        /// 读取指定日期范围内的所有日志
        /// </summary>
        public static List<LogEntry> ReadLogs(DateTime startDate, DateTime endDate)
        {
            var logs = new List<LogEntry>();
            string logDir = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "Logs");

            if (!Directory.Exists(logDir))
                return logs;

            // 获取日期范围内的所有日志文件
            var logFiles = Directory.GetFiles(logDir, "*.log")
                .Where(f =>
                {
                    var fileName = Path.GetFileNameWithoutExtension(f);
                    if (DateTime.TryParseExact(fileName, "yyyyMMdd", null, System.Globalization.DateTimeStyles.None, out var fileDate))
                    {
                        return fileDate.Date >= startDate.Date && fileDate.Date <= endDate.Date;
                    }
                    return false;
                })
                .OrderBy(f => f)
                .ToList();

            // 逐行读取每个文件
            foreach (var file in logFiles)
            {
                using (var reader = new StreamReader(file, Encoding.UTF8))
                {
                    string line;
                    while ((line = reader.ReadLine()) != null)
                    {
                        var match = _logRegex.Match(line);
                        if (match.Success)
                        {
                            var log = new LogEntry
                            {
                                Time = DateTime.ParseExact(match.Groups["Time"].Value, "yyyy-MM-dd HH:mm:ss.fff", null),
                                Level = ParseLogLevel(match.Groups["Level"].Value),
                                ThreadId = int.Parse(match.Groups["Thread"].Value),
                                Message = match.Groups["Message"].Value
                            };

                            // 精确时间过滤
                            if (log.Time >= startDate && log.Time <= endDate)
                            {
                                logs.Add(log);
                            }
                        }
                    }
                }
            }

            return logs;
        }

        // 解析日志级别
        private static LogLevel ParseLogLevel(string level)
        {
            if (Enum.TryParse<LogLevel>(level, true, out var result))
                return result;
            return LogLevel.Info;
        }
    }
}
```

## 四、工业项目最佳实践

1. **生产环境配置**：
   - Log4Net 全局级别设为 `INFO`，过滤 Debug 日志
   - 单个日志文件最大 100MB，保留 7 天
   - TextBlock 最大保留 1000 行，更新间隔 150ms
2. **异常处理**：
   - 所有日志相关异常内部消化，不影响主程序运行
   - 日志写入失败时，不抛出异常
3. **性能优化**：
   - 历史日志查询使用虚拟化 DataGrid，支持 10 万 + 条日志
   - 大文件读取使用 StreamReader 逐行读取，不一次性加载到内存
4. **安全考虑**：
   - 日志文件不要包含敏感信息（密码、密钥等）
   - 定期备份日志文件到服务器
