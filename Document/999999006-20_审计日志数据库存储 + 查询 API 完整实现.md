### 1.4 实现数据库版审计日志存储

csharp:

```c#
// IndustrialAutomation.Core/Audit/DatabaseAuditLogStore.cs
using System;
using System.Collections.Generic;
using System.Linq;
using System.Threading.Tasks;
using IndustrialAutomation.Core.Data;
using IndustrialAutomation.Core.Entities;
using IndustrialAutomation.Core.Models;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Logging;

namespace IndustrialAutomation.Core.Audit
{
    /// <summary>
    /// 数据库审计日志存储实现
    /// </summary>
    public class DatabaseAuditLogStore : IAuditLogStore
    {
        private readonly AuditDbContext _dbContext;
        private readonly ILogger<DatabaseAuditLogStore> _logger;
        private readonly int _batchSize = 100;

        public DatabaseAuditLogStore(AuditDbContext dbContext, 
            ILogger<DatabaseAuditLogStore> logger)
        {
            _dbContext = dbContext;
            _logger = logger;
        }

        /// <summary>
        /// 保存单条审计日志
        /// </summary>
        public async Task SaveLogAsync(AuditLogEntry logEntry)
        {
            try
            {
                var entity = AuditLogEntity.FromModel(logEntry);
                _dbContext.AuditLogs.Add(entity);
                await _dbContext.SaveChangesAsync();
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "保存审计日志到数据库失败");
                // 数据库失败时回退到文件存储作为备份
                await FallbackToFileStorage(logEntry);
            }
        }

        /// <summary>
        /// 批量保存审计日志（高性能）
        /// </summary>
        public async Task SaveLogsAsync(IEnumerable<AuditLogEntry> logEntries)
        {
            try
            {
                var entities = logEntries.Select(AuditLogEntity.FromModel).ToList();
                
                // 分批插入，避免单次插入过大
                for (int i = 0; i < entities.Count; i += _batchSize)
                {
                    var batch = entities.Skip(i).Take(_batchSize).ToList();
                    _dbContext.AuditLogs.AddRange(batch);
                    await _dbContext.SaveChangesAsync();
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "批量保存审计日志到数据库失败");
                // 数据库失败时回退到文件存储
                foreach (var entry in logEntries)
                {
                    await FallbackToFileStorage(entry);
                }
            }
        }

        /// <summary>
        /// 分页查询审计日志
        /// </summary>
        public async Task<PagedResult<AuditLogEntry>> QueryLogsAsync(AuditLogQuery query)
        {
            var dbQuery = _dbContext.AuditLogs.AsQueryable();

            // 应用过滤条件
            if (query.StartTime.HasValue)
            {
                dbQuery = dbQuery.Where(e => e.TimestampUtc >= query.StartTime.Value.ToUniversalTime());
            }

            if (query.EndTime.HasValue)
            {
                dbQuery = dbQuery.Where(e => e.TimestampUtc <= query.EndTime.Value.ToUniversalTime());
            }

            if (!string.IsNullOrEmpty(query.OperatorId))
            {
                dbQuery = dbQuery.Where(e => e.OperatorId == query.OperatorId);
            }

            if (!string.IsNullOrEmpty(query.OperationName))
            {
                dbQuery = dbQuery.Where(e => e.OperationName.Contains(query.OperationName));
            }

            if (query.Success.HasValue)
            {
                dbQuery = dbQuery.Where(e => e.Success == query.Success.Value);
            }

            // 排序
            dbQuery = query.SortDescending 
                ? dbQuery.OrderByDescending(e => e.TimestampUtc) 
                : dbQuery.OrderBy(e => e.TimestampUtc);

            // 计算总数
            var totalCount = await dbQuery.CountAsync();

            // 分页
            var items = await dbQuery
                .Skip((query.PageNumber - 1) * query.PageSize)
                .Take(query.PageSize)
                .ToListAsync();

            // 转换为业务模型
            var resultItems = items.Select(e => e.ToModel()).ToList();

            return new PagedResult<AuditLogEntry>
            {
                Items = resultItems,
                TotalCount = totalCount,
                PageNumber = query.PageNumber,
                PageSize = query.PageSize
            };
        }

        /// <summary>
        /// 数据库失败时回退到文件存储
        /// </summary>
        private async Task FallbackToFileStorage(AuditLogEntry logEntry)
        {
            try
            {
                var fileStore = new FileAuditLogStore(
                    Microsoft.Extensions.Options.Options.Create(new FileAuditLogOptions()));
                await fileStore.SaveLogAsync(logEntry);
                _logger.LogWarning("已将审计日志回退保存到文件");
            }
            catch (Exception ex)
            {
                _logger.LogCritical(ex, "审计日志保存彻底失败！日志丢失");
            }
        }
    }

    /// <summary>
    /// 审计日志查询参数
    /// </summary>
    public class AuditLogQuery
    {
        public DateTime? StartTime { get; set; }
        public DateTime? EndTime { get; set; }
        public string OperatorId { get; set; }
        public string OperationName { get; set; }
        public bool? Success { get; set; }
        public int PageNumber { get; set; } = 1;
        public int PageSize { get; set; } = 20;
        public bool SortDescending { get; set; } = true;
    }

    /// <summary>
    /// 分页结果
    /// </summary>
    /// <typeparam name="T"></typeparam>
    public class PagedResult<T>
    {
        public List<T> Items { get; set; } = new();
        public int TotalCount { get; set; }
        public int PageNumber { get; set; }
        public int PageSize { get; set; }
        public int TotalPages => (int)Math.Ceiling((double)TotalCount / PageSize);
        public bool HasPreviousPage => PageNumber > 1;
        public bool HasNextPage => PageNumber < TotalPages;
    }
}
```

### 2.1 创建审计日志控制器

csharp:

```c#
// IndustrialAutomation.Web/Controllers/AuditLogsController.cs
using System.Threading.Tasks;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using IndustrialAutomation.Core.Audit;

namespace IndustrialAutomation.Web.Controllers
{
    /// <summary>
    /// 审计日志查询API
    /// </summary>
    [ApiController]
    [Route("api/[controller]")]
    [Authorize(Roles = "Admin, Auditor")] // 只有管理员和审计员可以访问
    public class AuditLogsController : ControllerBase
    {
        private readonly IAuditLogStore _auditLogStore;

        public AuditLogsController(IAuditLogStore auditLogStore)
        {
            _auditLogStore = auditLogStore;
        }

        /// <summary>
        /// 分页查询审计日志
        /// </summary>
        /// <param name="query">查询参数</param>
        /// <returns>分页审计日志结果</returns>
        [HttpGet]
        public async Task<ActionResult<PagedResult<AuditLogEntry>>> GetLogs(
            [FromQuery] AuditLogQuery query)
        {
            var result = await _auditLogStore.QueryLogsAsync(query);
            return Ok(result);
        }

        /// <summary>
        /// 根据ID获取单条审计日志详情
        /// </summary>
        /// <param name="id">日志ID</param>
        /// <returns>审计日志详情</returns>
        [HttpGet("{id}")]
        public async Task<ActionResult<AuditLogEntry>> GetLogById(Guid id)
        {
            // 这里可以扩展根据ID查询的方法
            // 为了简化示例，我们先查询最近100条然后过滤
            var query = new AuditLogQuery { PageSize = 100 };
            var result = await _auditLogStore.QueryLogsAsync(query);
            var log = result.Items.FirstOrDefault(l => l.Id == id);
            
            if (log == null)
            {
                return NotFound();
            }
            
            return Ok(log);
        }
    }
}
```

## 六、核心优势与设计要点

### 6.1 数据库存储相比文件存储的优势

1. **高性能查询**：支持复杂的条件查询和排序，毫秒级响应
2. **数据安全**：支持事务、备份和恢复，防止数据丢失
3. **可扩展性**：可以轻松处理百万级以上的日志数据
4. **集中管理**：所有系统的审计日志可以集中存储在同一个数据库
5. **合规性**：满足 ISO 9001、GMP 等标准对审计数据的存储要求

### 6.2 关键设计要点

1. **UTC 时间存储**：所有时间都以 UTC 格式存储，避免时区问题
2. **索引优化**：为常用查询字段创建索引，大幅提升查询性能
3. **分批插入**：批量插入时分批处理，避免单次插入过大导致的性能问题
4. **故障回退**：数据库不可用时自动回退到文件存储，保证日志不丢失
5. **权限控制**：审计日志 API 默认启用授权，只有管理员和审计员可以访问
6. **分页查询**：所有查询都支持分页，避免返回大量数据导致的性能问题
