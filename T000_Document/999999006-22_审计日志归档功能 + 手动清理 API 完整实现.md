### 1.2 实现 ZIP 压缩归档器

csharp:

```c#
// IndustrialAutomation.Core/Audit/ZipAuditLogArchiver.cs
using System;
using System.Collections.Generic;
using System.IO;
using System.IO.Compression;
using System.Linq;
using System.Text;
using System.Threading.Tasks;
using IndustrialAutomation.Core.Models;
using Microsoft.Extensions.Logging;
using Microsoft.Extensions.Options;

namespace IndustrialAutomation.Core.Audit
{
    /// <summary>
    /// ZIP压缩审计日志归档器
    /// </summary>
    public class ZipAuditLogArchiver : IAuditLogArchiver
    {
        private readonly IAuditLogStore _auditLogStore;
        private readonly ILogger<ZipAuditLogArchiver> _logger;
        private readonly AuditLogArchiveOptions _options;

        public ZipAuditLogArchiver(IAuditLogStore auditLogStore,
            ILogger<ZipAuditLogArchiver> logger,
            IOptions<AuditLogArchiveOptions> options)
        {
            _auditLogStore = auditLogStore;
            _logger = logger;
            _options = options.Value;
            
            // 确保归档目录存在
            Directory.CreateDirectory(_options.ArchiveDirectory);
        }

        public async Task<string> ArchiveLogsAsync(DateTime startTime, DateTime endTime)
        {
            if (!_options.EnableArchive)
            {
                _logger.LogInformation("审计日志归档已禁用");
                return null;
            }

            _logger.LogInformation("开始归档 {StartTime} 至 {EndTime} 的审计日志", 
                startTime.ToString("yyyy-MM-dd HH:mm:ss"), 
                endTime.ToString("yyyy-MM-dd HH:mm:ss"));

            // 查询所有需要归档的日志
            var query = new AuditLogQuery
            {
                StartTime = startTime,
                EndTime = endTime,
                PageSize = _options.MaxRowsPerArchive,
                SortDescending = false
            };

            var allLogs = new List<AuditLogEntry>();
            PagedResult<AuditLogEntry> result;
            
            do
            {
                result = await _auditLogStore.QueryLogsAsync(query);
                allLogs.AddRange(result.Items);
                query.PageNumber++;
            } while (result.HasNextPage);

            if (allLogs.Count == 0)
            {
                _logger.LogInformation("没有需要归档的审计日志");
                return null;
            }

            return await ArchiveLogsAsync(allLogs);
        }

        public async Task<string> ArchiveLogsAsync(IEnumerable<AuditLogEntry> logs)
        {
            if (!_options.EnableArchive)
                return null;

            var logList = logs.ToList();
            if (logList.Count == 0)
                return null;

            // 生成归档文件名
            var minTime = logList.Min(l => l.Timestamp);
            var maxTime = logList.Max(l => l.Timestamp);
            var fileName = string.Format(_options.FileNameFormat, minTime, maxTime);
            var filePath = Path.Combine(_options.ArchiveDirectory, fileName);

            try
            {
                // 生成CSV内容
                var csvContent = GenerateCsvContent(logList);

                if (_options.EnableCompression)
                {
                    // 压缩为ZIP文件
                    using var memoryStream = new MemoryStream();
                    using (var archive = new ZipArchive(memoryStream, ZipArchiveMode.Create, true))
                    {
                        var entry = archive.CreateEntry(Path.ChangeExtension(fileName, ".csv"), CompressionLevel.Optimal);
                        using var entryStream = entry.Open();
                        using var writer = new StreamWriter(entryStream, Encoding.UTF8);
                        await writer.WriteAsync(csvContent);
                    }

                    memoryStream.Seek(0, SeekOrigin.Begin);
                    await File.WriteAllBytesAsync(Path.ChangeExtension(filePath, ".zip"), memoryStream.ToArray());
                    filePath = Path.ChangeExtension(filePath, ".zip");
                }
                else
                {
                    // 直接保存为CSV文件
                    await File.WriteAllTextAsync(Path.ChangeExtension(filePath, ".csv"), csvContent, Encoding.UTF8);
                    filePath = Path.ChangeExtension(filePath, ".csv");
                }

                _logger.LogInformation("审计日志归档成功，共归档 {Count} 条记录，文件：{FilePath}", 
                    logList.Count, filePath);

                // 清理过期的归档文件
                CleanupOldArchives();

                return filePath;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "审计日志归档失败");
                throw new InvalidOperationException("审计日志归档失败，已取消清理操作", ex);
            }
        }

        /// <summary>
        /// 生成CSV格式的日志内容
        /// </summary>
        private string GenerateCsvContent(IEnumerable<AuditLogEntry> logs)
        {
            var sb = new StringBuilder();
            
            // 写入标题行
            sb.AppendLine("日志ID,操作时间,操作员ID,操作员名称,操作名称,方法名,是否成功,客户端IP,耗时(ms),错误信息,参数,原始值,新值");
            
            // 写入数据行
            foreach (var log in logs)
            {
                // 处理CSV中的逗号和引号
                string EscapeCsv(string value)
                {
                    if (string.IsNullOrEmpty(value))
                        return string.Empty;
                    if (value.Contains(',') || value.Contains('"') || value.Contains('\n'))
                        return $"\"{value.Replace("\"", "\"\"")}\"";
                    return value;
                }

                sb.AppendLine($"{log.Id},{log.Timestamp:yyyy-MM-dd HH:mm:ss.fff},{EscapeCsv(log.OperatorId)},{EscapeCsv(log.OperatorName)},{EscapeCsv(log.OperationName)},{EscapeCsv(log.MethodName)},{(log.Success ? "成功" : "失败")},{EscapeCsv(log.ClientIpAddress)},{log.DurationMs},{EscapeCsv(log.ErrorMessage)},{EscapeCsv(log.Parameters)},{EscapeCsv(log.OriginalValues)},{EscapeCsv(log.NewValues)}");
            }

            return sb.ToString();
        }

        /// <summary>
        /// 清理过期的归档文件
        /// </summary>
        private void CleanupOldArchives()
        {
            if (_options.ArchiveRetentionDays <= 0)
                return;

            try
            {
                var cutoffDate = DateTime.Now.AddDays(-_options.ArchiveRetentionDays);
                var files = Directory.GetFiles(_options.ArchiveDirectory);

                foreach (var file in files)
                {
                    var fileInfo = new FileInfo(file);
                    if (fileInfo.CreationTime < cutoffDate)
                    {
                        fileInfo.Delete();
                        _logger.LogInformation("已删除过期归档文件：{File}", file);
                    }
                }
            }
            catch (Exception ex)
            {
                _logger.LogWarning(ex, "清理过期归档文件失败");
            }
        }
    }
}
```

### 1.3 更新自动清理服务集成归档功能

csharp:

```c#
// IndustrialAutomation.Core/Services/AuditLogCleanupService.cs
using System;
using System.Linq;
using System.Threading;
using System.Threading.Tasks;
using IndustrialAutomation.Core.Audit;
using IndustrialAutomation.Core.Data;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Hosting;
using Microsoft.Extensions.Logging;
using Microsoft.Extensions.Options;

namespace IndustrialAutomation.Core.Services
{
    public class AuditLogCleanupService : BackgroundService
    {
        private readonly IServiceProvider _serviceProvider;
        private readonly ILogger<AuditLogCleanupService> _logger;
        private readonly AuditLogCleanupOptions _cleanupOptions;
        private readonly AuditLogArchiveOptions _archiveOptions;

        public AuditLogCleanupService(IServiceProvider serviceProvider,
            ILogger<AuditLogCleanupService> logger,
            IOptions<AuditLogCleanupOptions> cleanupOptions,
            IOptions<AuditLogArchiveOptions> archiveOptions)
        {
            _serviceProvider = serviceProvider;
            _logger = logger;
            _cleanupOptions = cleanupOptions.Value;
            _archiveOptions = archiveOptions.Value;
        }

        protected override async Task ExecuteAsync(CancellationToken stoppingToken)
        {
            if (!_cleanupOptions.EnableAutoCleanup)
            {
                _logger.LogInformation("审计日志自动清理已禁用");
                return;
            }

            _logger.LogInformation("审计日志自动清理服务已启动");
            _logger.LogInformation("日志保留天数：{RetentionDays}天", _cleanupOptions.RetentionDays);
            _logger.LogInformation("清理间隔：{Interval}小时", _cleanupOptions.CleanupIntervalHours);
            _logger.LogInformation("归档功能：{ArchiveStatus}", _archiveOptions.EnableArchive ? "已启用" : "已禁用");

            while (!stoppingToken.IsCancellationRequested)
            {
                try
                {
                    await CleanupOldLogsAsync(stoppingToken);
                }
                catch (Exception ex)
                {
                    _logger.LogError(ex, "执行审计日志清理时发生错误");
                }

                // 等待下一次清理时间
                await Task.Delay(TimeSpan.FromHours(_cleanupOptions.CleanupIntervalHours), stoppingToken);
            }

            _logger.LogInformation("审计日志自动清理服务已停止");
        }

        private async Task CleanupOldLogsAsync(CancellationToken stoppingToken)
        {
            var cutoffDate = DateTime.UtcNow.AddDays(-_cleanupOptions.RetentionDays);
            _logger.LogInformation("开始清理 {CutoffDate} 之前的审计日志", cutoffDate);

            using var scope = _serviceProvider.CreateScope();
            var dbContext = scope.ServiceProvider.GetRequiredService<AuditDbContext>();
            var archiver = scope.ServiceProvider.GetRequiredService<IAuditLogArchiver>();

            // 步骤1：先归档需要清理的日志
            string archivePath = null;
            if (_archiveOptions.EnableArchive)
            {
                try
                {
                    archivePath = await archiver.ArchiveLogsAsync(DateTime.MinValue, cutoffDate);
                }
                catch (Exception ex)
                {
                    _logger.LogCritical(ex, "审计日志归档失败，已取消清理操作");
                    return;
                }
            }

            // 步骤2：删除已归档的日志
            int totalDeleted = 0;
            int batchSize = _cleanupOptions.MaxRowsPerCleanup;

            while (!stoppingToken.IsCancellationRequested)
            {
                var oldLogs = await dbContext.AuditLogs
                    .Where(e => e.TimestampUtc < cutoffDate)
                    .OrderBy(e => e.TimestampUtc)
                    .Take(batchSize)
                    .ToListAsync(stoppingToken);

                if (oldLogs.Count == 0)
                    break;

                dbContext.AuditLogs.RemoveRange(oldLogs);
                await dbContext.SaveChangesAsync(stoppingToken);

                totalDeleted += oldLogs.Count;
                _logger.LogDebug("已删除 {Count} 条审计日志", oldLogs.Count);

                await Task.Delay(100, stoppingToken);
            }

            if (totalDeleted > 0)
            {
                _logger.LogInformation("审计日志清理完成");
                _logger.LogInformation("归档文件：{ArchivePath}", archivePath ?? "未归档");
                _logger.LogInformation("共删除 {TotalDeleted} 条记录", totalDeleted);
            }
            else
            {
                _logger.LogInformation("没有需要清理的审计日志");
            }
        }

        /// <summary>
        /// 手动清理指定时间之前的日志
        /// </summary>
        public async Task<CleanupResult> ManualCleanupAsync(DateTime cutoffDate, bool forceDelete = false)
        {
            using var scope = _serviceProvider.CreateScope();
            var dbContext = scope.ServiceProvider.GetRequiredService<AuditDbContext>();
            var archiver = scope.ServiceProvider.GetRequiredService<IAuditLogArchiver>();

            var result = new CleanupResult();
            result.CutoffDate = cutoffDate;

            // 统计将要清理的记录数
            result.TotalRecordsToCleanup = await dbContext.AuditLogs
                .CountAsync(e => e.TimestampUtc < cutoffDate);

            if (result.TotalRecordsToCleanup == 0)
            {
                result.Message = "没有需要清理的审计日志";
                return result;
            }

            // 归档
            if (_archiveOptions.EnableArchive && !forceDelete)
            {
                try
                {
                    result.ArchiveFilePath = await archiver.ArchiveLogsAsync(DateTime.MinValue, cutoffDate);
                    result.ArchiveSuccess = true;
                }
                catch (Exception ex)
                {
                    result.ArchiveSuccess = false;
                    result.ErrorMessage = $"归档失败：{ex.Message}";
                    
                    if (!forceDelete)
                    {
                        result.Success = false;
                        result.Message = "归档失败，已取消清理操作";
                        return result;
                    }
                }
            }

            // 删除
            int totalDeleted = 0;
            int batchSize = _cleanupOptions.MaxRowsPerCleanup;

            while (true)
            {
                var oldLogs = await dbContext.AuditLogs
                    .Where(e => e.TimestampUtc < cutoffDate)
                    .OrderBy(e => e.TimestampUtc)
                    .Take(batchSize)
                    .ToListAsync();

                if (oldLogs.Count == 0)
                    break;

                dbContext.AuditLogs.RemoveRange(oldLogs);
                await dbContext.SaveChangesAsync();

                totalDeleted += oldLogs.Count;
            }

            result.TotalRecordsDeleted = totalDeleted;
            result.Success = true;
            result.Message = forceDelete 
                ? $"强制清理成功，共删除 {totalDeleted} 条记录（未归档）" 
                : $"清理成功，共删除 {totalDeleted} 条记录";

            return result;
        }
    }

    /// <summary>
    /// 清理操作结果
    /// </summary>
    public class CleanupResult
    {
        public bool Success { get; set; }
        public string Message { get; set; }
        public DateTime CutoffDate { get; set; }
        public int TotalRecordsToCleanup { get; set; }
        public int TotalRecordsDeleted { get; set; }
        public bool ArchiveSuccess { get; set; }
        public string ArchiveFilePath { get; set; }
        public string ErrorMessage { get; set; }
    }
}
```

## 五、核心设计要点

### 5.1 归档功能设计要点

1. **先归档后删除**：确保数据在删除前已经安全保存
2. **原子性操作**：归档失败时不执行删除，避免数据丢失
3. **压缩存储**：默认使用 ZIP 压缩，节省 80% 以上的存储空间
4. **CSV 格式**：使用通用的 CSV 格式，便于导入到其他工具进行分析
5. **自动清理归档**：支持自动清理过期的归档文件
6. **错误处理**：归档失败时有详细的错误日志，并停止清理操作

### 5.2 手动清理设计要点

1. **权限控制**：只有管理员可以执行手动清理操作
2. **灵活配置**：可以指定任意清理截止日期
3. **强制删除选项**：在紧急情况下可以跳过归档直接删除
4. **详细结果**：返回清理的详细信息，包括记录数和归档文件路径
