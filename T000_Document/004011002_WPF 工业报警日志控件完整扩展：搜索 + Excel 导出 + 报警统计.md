### 2.2 搜索功能核心实现

csharp:

```c#
// IndustrialAlarmLog.xaml.cs
using System.Windows.Media;
using System.Windows.Documents;

public partial class IndustrialAlarmLog : UserControl
{
    // 原有字段保持不变
    private TextPointer _lastSearchPosition;
    private string _currentSearchText;
    private readonly Brush _highlightBrush = Brushes.Yellow;

    /// <summary>
    /// 查找指定文本
    /// </summary>
    private bool FindText(string searchText, bool searchForward = true)
    {
        if (string.IsNullOrEmpty(searchText))
            return false;

        _currentSearchText = searchText;
        TextPointer startPosition;

        // 确定搜索起始位置
        if (_lastSearchPosition == null)
        {
            startPosition = searchForward 
                ? richTextBox.Document.ContentStart 
                : richTextBox.Document.ContentEnd;
        }
        else
        {
            startPosition = searchForward 
                ? _lastSearchPosition.GetPositionAtOffset(1) 
                : _lastSearchPosition.GetPositionAtOffset(-1);
        }

        // 执行搜索
        TextRange foundRange = FindTextInRange(startPosition, 
            searchForward ? richTextBox.Document.ContentEnd : richTextBox.Document.ContentStart,
            searchText);

        if (foundRange != null)
        {
            // 清除之前的高亮
            ClearHighlight();
            
            // 高亮找到的文本
            foundRange.ApplyPropertyValue(TextElement.BackgroundProperty, _highlightBrush);
            
            // 滚动到找到的位置
            richTextBox.Selection.Select(foundRange.Start, foundRange.End);
            richTextBox.ScrollToVerticalOffset(richTextBox.VerticalOffset);
            
            _lastSearchPosition = foundRange.End;
            return true;
        }
        else
        {
            // 循环搜索：到达末尾后从头开始
            if (MessageBox.Show("已到达文档末尾，是否从头开始搜索？", "搜索", 
                MessageBoxButton.YesNo, MessageBoxImage.Question) == MessageBoxResult.Yes)
            {
                _lastSearchPosition = null;
                return FindText(searchText, searchForward);
            }
            return false;
        }
    }

    /// <summary>
    /// 在指定范围内查找文本
    /// </summary>
    private TextRange FindTextInRange(TextPointer start, TextPointer end, string searchText)
    {
        TextPointer current = start;
        while (current != null && current.CompareTo(end) < 0)
        {
            if (current.GetPointerContext(LogicalDirection.Forward) == TextPointerContext.Text)
            {
                string textRun = current.GetTextInRun(LogicalDirection.Forward);
                int index = textRun.IndexOf(searchText, StringComparison.OrdinalIgnoreCase);
                
                if (index >= 0)
                {
                    TextPointer startPointer = current.GetPositionAtOffset(index);
                    TextPointer endPointer = startPointer.GetPositionAtOffset(searchText.Length);
                    return new TextRange(startPointer, endPointer);
                }
            }
            current = current.GetNextContextPosition(LogicalDirection.Forward);
        }
        return null;
    }

    /// <summary>
    /// 清除所有高亮
    /// </summary>
    private void ClearHighlight()
    {
        TextRange wholeDocument = new TextRange(richTextBox.Document.ContentStart, richTextBox.Document.ContentEnd);
        wholeDocument.ApplyPropertyValue(TextElement.BackgroundProperty, null);
        _lastSearchPosition = null;
    }

    // 事件处理
    private void TxtSearch_KeyDown(object sender, KeyEventArgs e)
    {
        if (e.Key == Key.Enter)
        {
            BtnFindNext_Click(sender, e);
        }
    }

    private void BtnFindNext_Click(object sender, RoutedEventArgs e)
    {
        if (!FindText(txtSearch.Text, true))
        {
            MessageBox.Show("未找到匹配内容", "搜索", MessageBoxButton.OK, MessageBoxImage.Information);
        }
    }

    private void BtnFindPrev_Click(object sender, RoutedEventArgs e)
    {
        if (!FindText(txtSearch.Text, false))
        {
            MessageBox.Show("未找到匹配内容", "搜索", MessageBoxButton.OK, MessageBoxImage.Information);
        }
    }

    private void BtnClearHighlight_Click(object sender, RoutedEventArgs e)
    {
        ClearHighlight();
        txtSearch.Clear();
    }
}
```

```c#
// IndustrialAlarmLog.xaml.cs
using System.IO;
using System.Linq;
using Microsoft.Win32;
using OfficeOpenXml;
using OfficeOpenXml.Style;

public partial class IndustrialAlarmLog : UserControl
{
    /// <summary>
    /// 导出报警记录到Excel
    /// </summary>
    private void ExportToExcel()
    {
        if (_alarmRecords.Count == 0)
        {
            MessageBox.Show("没有可导出的报警记录", "提示", MessageBoxButton.OK, MessageBoxImage.Information);
            return;
        }

        var saveDialog = new SaveFileDialog
        {
            Filter = "Excel文件(*.xlsx)|*.xlsx",
            FileName = $"报警日志_{DateTime.Now:yyyyMMddHHmmss}"
        };

        if (saveDialog.ShowDialog() != true)
            return;

        try
        {
            // 设置EPPlus非商业许可
            ExcelPackage.LicenseContext = LicenseContext.NonCommercial;

            using (var package = new ExcelPackage())
            {
                var worksheet = package.Workbook.Worksheets.Add("报警日志");

                // 1. 设置表头
                worksheet.Cells[1, 1].Value = "序号";
                worksheet.Cells[1, 2].Value = "报警时间";
                worksheet.Cells[1, 3].Value = "报警级别";
                worksheet.Cells[1, 4].Value = "操作员";
                worksheet.Cells[1, 5].Value = "报警信息";

                // 2. 设置表头样式
                using (var range = worksheet.Cells[1, 1, 1, 5])
                {
                    range.Style.Font.Bold = true;
                    range.Style.Fill.PatternType = ExcelFillStyle.Solid;
                    range.Style.Fill.BackgroundColor.SetColor(Color.LightBlue);
                    range.Style.HorizontalAlignment = ExcelHorizontalAlignment.Center;
                    range.Style.VerticalAlignment = ExcelVerticalAlignment.Center;
                    range.Style.Border.Top.Style = ExcelBorderStyle.Thin;
                    range.Style.Border.Bottom.Style = ExcelBorderStyle.Thin;
                    range.Style.Border.Left.Style = ExcelBorderStyle.Thin;
                    range.Style.Border.Right.Style = ExcelBorderStyle.Thin;
                }

                // 3. 写入数据
                for (int i = 0; i < _alarmRecords.Count; i++)
                {
                    var record = _alarmRecords[i];
                    int row = i + 2;

                    worksheet.Cells[row, 1].Value = i + 1;
                    worksheet.Cells[row, 2].Value = record.Timestamp.ToString("yyyy-MM-dd HH:mm:ss.fff");
                    worksheet.Cells[row, 3].Value = record.Level.ToString();
                    worksheet.Cells[row, 4].Value = record.OperatorName;
                    worksheet.Cells[row, 5].Value = record.Message;

                    // 4. 根据报警级别设置行颜色
                    var rowRange = worksheet.Cells[row, 1, row, 5];
                    rowRange.Style.Border.Top.Style = ExcelBorderStyle.Thin;
                    rowRange.Style.Border.Bottom.Style = ExcelBorderStyle.Thin;
                    rowRange.Style.Border.Left.Style = ExcelBorderStyle.Thin;
                    rowRange.Style.Border.Right.Style = ExcelBorderStyle.Thin;

                    switch (record.Level)
                    {
                        case AlarmLevel.Warning:
                            rowRange.Style.Font.Color.SetColor(Color.Orange);
                            break;
                        case AlarmLevel.Error:
                            rowRange.Style.Font.Color.SetColor(Color.Red);
                            break;
                        case AlarmLevel.Critical:
                            rowRange.Style.Font.Color.SetColor(Color.White);
                            rowRange.Style.Fill.PatternType = ExcelFillStyle.Solid;
                            rowRange.Style.Fill.BackgroundColor.SetColor(Color.Red);
                            break;
                    }
                }

                // 5. 自动调整列宽
                worksheet.Cells.AutoFitColumns();
                worksheet.Column(2).Width = 25; // 时间列加宽
                worksheet.Column(5).Width = 50; // 信息列加宽

                // 6. 添加统计信息
                int infoCount = _alarmRecords.Count(r => r.Level == AlarmLevel.Info);
                int warningCount = _alarmRecords.Count(r => r.Level == AlarmLevel.Warning);
                int errorCount = _alarmRecords.Count(r => r.Level == AlarmLevel.Error);
                int criticalCount = _alarmRecords.Count(r => r.Level == AlarmLevel.Critical);

                int statsRow = _alarmRecords.Count + 3;
                worksheet.Cells[statsRow, 1].Value = "统计信息";
                worksheet.Cells[statsRow, 1].Style.Font.Bold = true;
                worksheet.Cells[statsRow + 1, 1].Value = $"信息：{infoCount}条";
                worksheet.Cells[statsRow + 2, 1].Value = $"警告：{warningCount}条";
                worksheet.Cells[statsRow + 3, 1].Value = $"错误：{errorCount}条";
                worksheet.Cells[statsRow + 4, 1].Value = $"严重：{criticalCount}条";
                worksheet.Cells[statsRow + 5, 1].Value = $"总计：{_alarmRecords.Count}条";

                // 7. 保存文件
                package.SaveAs(new FileInfo(saveDialog.FileName));
            }

            MessageBox.Show($"成功导出 {_alarmRecords.Count} 条记录！\n文件：{saveDialog.FileName}", 
                "导出成功", MessageBoxButton.OK, MessageBoxImage.Information);
        }
        catch (Exception ex)
        {
            MessageBox.Show($"导出失败：{ex.Message}", "错误", MessageBoxButton.OK, MessageBoxImage.Error);
        }
    }

    private void BtnExportExcel_Click(object sender, RoutedEventArgs e)
    {
        ExportToExcel();
    }
}
```

```c#
// Models/AlarmStatistics.cs
public class AlarmStatistics
{
    public int TotalCount { get; set; }
    public int InfoCount { get; set; }
    public int WarningCount { get; set; }
    public int ErrorCount { get; set; }
    public int CriticalCount { get; set; }
    
    public double InfoPercentage => TotalCount > 0 ? Math.Round((double)InfoCount / TotalCount * 100, 1) : 0;
    public double WarningPercentage => TotalCount > 0 ? Math.Round((double)WarningCount / TotalCount * 100, 1) : 0;
    public double ErrorPercentage => TotalCount > 0 ? Math.Round((double)ErrorCount / TotalCount * 100, 1) : 0;
    public double CriticalPercentage => TotalCount > 0 ? Math.Round((double)CriticalCount / TotalCount * 100, 1) : 0;
    
    public DateTime StartTime { get; set; }
    public DateTime EndTime { get; set; }
}
```

```c#
// IndustrialAlarmLog.xaml.cs
/// <summary>
/// 生成报警统计
/// </summary>
private AlarmStatistics GenerateStatistics()
{
    var stats = new AlarmStatistics
    {
        TotalCount = _alarmRecords.Count,
        InfoCount = _alarmRecords.Count(r => r.Level == AlarmLevel.Info),
        WarningCount = _alarmRecords.Count(r => r.Level == AlarmLevel.Warning),
        ErrorCount = _alarmRecords.Count(r => r.Level == AlarmLevel.Error),
        CriticalCount = _alarmRecords.Count(r => r.Level == AlarmLevel.Critical),
        StartTime = _alarmRecords.Count > 0 ? _alarmRecords.Min(r => r.Timestamp) : DateTime.Now,
        EndTime = _alarmRecords.Count > 0 ? _alarmRecords.Max(r => r.Timestamp) : DateTime.Now
    };
    return stats;
}

private void BtnStatistics_Click(object sender, RoutedEventArgs e)
{
    if (_alarmRecords.Count == 0)
    {
        MessageBox.Show("没有可统计的报警记录", "提示", MessageBoxButton.OK, MessageBoxImage.Information);
        return;
    }

    var statistics = GenerateStatistics();
    var statsWindow = new AlarmStatisticsWindow(statistics);
    statsWindow.Owner = Window.GetWindow(this);
    statsWindow.ShowDialog();
}
```

## 六、性能优化与最佳实践

### 6.1 搜索性能优化

- 对于超过 10 万行的文档，建议使用索引搜索而不是逐行遍历
- 限制单次搜索的最大匹配数量，避免 UI 卡顿
- 异步执行搜索操作，保持 UI 响应

### 6.2 Excel 导出优化

- 分批写入数据，避免内存溢出
- 对于超过 10 万条的记录，自动分工作表
- 添加导出进度提示

### 6.3 统计功能优化

- 缓存统计结果，避免重复计算
- 对于大数据集，使用 LINQ 查询优化统计计算
- 定期自动更新统计数据
