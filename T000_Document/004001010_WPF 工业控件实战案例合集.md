```c#
// StatusIndicator.cs
public class StatusIndicator : Control
{
    static StatusIndicator()
    {
        DefaultStyleKeyProperty.OverrideMetadata(typeof(StatusIndicator), 
            new FrameworkPropertyMetadata(typeof(StatusIndicator)));
    }

    // 状态颜色依赖属性
    public static readonly DependencyProperty StatusColorProperty = DependencyProperty.Register(
        nameof(StatusColor), typeof(Brush), typeof(StatusIndicator), 
        new PropertyMetadata(Brushes.Gray, OnStatusColorChanged));

    public Brush StatusColor
    {
        get => (Brush)GetValue(StatusColorProperty);
        set => SetValue(StatusColorProperty, value);
    }

    // 是否闪烁
    public static readonly DependencyProperty IsFlashingProperty = DependencyProperty.Register(
        nameof(IsFlashing), typeof(bool), typeof(StatusIndicator), 
        new PropertyMetadata(false));

    public bool IsFlashing
    {
        get => (bool)GetValue(IsFlashingProperty);
        set => SetValue(IsFlashingProperty, value);
    }

    // 发光颜色
    public static readonly DependencyProperty GlowColorProperty = DependencyProperty.Register(
        nameof(GlowColor), typeof(Brush), typeof(StatusIndicator), 
        new PropertyMetadata(Brushes.Gray));

    public Brush GlowColor
    {
        get => (Brush)GetValue(GlowColorProperty);
        set => SetValue(GlowColorProperty, value);
    }

    private static void OnStatusColorChanged(DependencyObject d, DependencyPropertyChangedEventArgs e)
    {
        var indicator = (StatusIndicator)d;
        indicator.Background = e.NewValue as Brush;
        indicator.GlowColor = e.NewValue as Brush;
    }
}
```

### 2.1 带范围验证的数值输入框

工业参数输入必备，自动验证数值范围，支持小数点位数控制。

csharp:

```c#
public class NumericTextBox : TextBox
{
    static NumericTextBox()
    {
        DefaultStyleKeyProperty.OverrideMetadata(typeof(NumericTextBox), 
            new FrameworkPropertyMetadata(typeof(NumericTextBox)));
    }

    public NumericTextBox()
    {
        PreviewTextInput += NumericTextBox_PreviewTextInput;
        DataObject.AddPastingHandler(this, OnPasting);
    }

    // 最小值
    public static readonly DependencyProperty MinValueProperty = DependencyProperty.Register(
        nameof(MinValue), typeof(double), typeof(NumericTextBox), 
        new PropertyMetadata(double.MinValue));

    public double MinValue
    {
        get => (double)GetValue(MinValueProperty);
        set => SetValue(MinValueProperty, value);
    }

    // 最大值
    public static readonly DependencyProperty MaxValueProperty = DependencyProperty.Register(
        nameof(MaxValue), typeof(double), typeof(NumericTextBox), 
        new PropertyMetadata(double.MaxValue));

    public double MaxValue
    {
        get => (double)GetValue(MaxValueProperty);
        set => SetValue(MaxValueProperty, value);
    }

    // 小数位数
    public static readonly DependencyProperty DecimalPlacesProperty = DependencyProperty.Register(
        nameof(DecimalPlaces), typeof(int), typeof(NumericTextBox), 
        new PropertyMetadata(2));

    public int DecimalPlaces
    {
        get => (int)GetValue(DecimalPlacesProperty);
        set => SetValue(DecimalPlacesProperty, value);
    }

    // 当前数值
    public static readonly DependencyProperty ValueProperty = DependencyProperty.Register(
        nameof(Value), typeof(double), typeof(NumericTextBox), 
        new FrameworkPropertyMetadata(0.0, FrameworkPropertyMetadataOptions.BindsTwoWayByDefault, OnValueChanged));

    public double Value
    {
        get => (double)GetValue(ValueProperty);
        set => SetValue(ValueProperty, value);
    }

    private static void OnValueChanged(DependencyObject d, DependencyPropertyChangedEventArgs e)
    {
        var textBox = (NumericTextBox)d;
        textBox.Text = Math.Round((double)e.NewValue, textBox.DecimalPlaces).ToString();
    }

    protected override void OnLostFocus(RoutedEventArgs e)
    {
        base.OnLostFocus(e);
        
        if (double.TryParse(Text, out double value))
        {
            // 范围验证
            value = Math.Clamp(value, MinValue, MaxValue);
            Value = Math.Round(value, DecimalPlaces);
        }
        else
        {
            // 输入无效时恢复原值
            Text = Math.Round(Value, DecimalPlaces).ToString();
        }
    }

    private void NumericTextBox_PreviewTextInput(object sender, System.Windows.Input.TextCompositionEventArgs e)
    {
        // 只允许数字和一个小数点
        e.Handled = !IsValidInput(Text + e.Text);
    }

    private void OnPasting(object sender, DataObjectPastingEventArgs e)
    {
        if (e.DataObject.GetDataPresent(typeof(string)))
        {
            string text = (string)e.DataObject.GetData(typeof(string));
            if (!IsValidInput(text))
            {
                e.CancelCommand();
            }
        }
        else
        {
            e.CancelCommand();
        }
    }

    private bool IsValidInput(string text)
    {
        // 允许负号
        if (text == "-" && MinValue < 0)
            return true;
        
        // 允许一个小数点
        if (text == "." && !Text.Contains("."))
            return true;
        
        return double.TryParse(text, out _);
    }
}
```

csharp:

```c#
public class AlarmBar : Control
{
    static AlarmBar()
    {
        DefaultStyleKeyProperty.OverrideMetadata(typeof(AlarmBar), 
            new FrameworkPropertyMetadata(typeof(AlarmBar)));
    }

    // 报警级别
    public static readonly DependencyProperty AlarmLevelProperty = DependencyProperty.Register(
        nameof(AlarmLevel), typeof(AlarmLevel), typeof(AlarmBar), 
        new PropertyMetadata(AlarmLevel.Info, OnAlarmLevelChanged));

    public AlarmLevel AlarmLevel
    {
        get => (AlarmLevel)GetValue(AlarmLevelProperty);
        set => SetValue(AlarmLevelProperty, value);
    }

    // 报警信息
    public static readonly DependencyProperty MessageProperty = DependencyProperty.Register(
        nameof(Message), typeof(string), typeof(AlarmBar), 
        new PropertyMetadata(""));

    public string Message
    {
        get => (string)GetValue(MessageProperty);
        set => SetValue(MessageProperty, value);
    }

    // 自动关闭时间（毫秒）
    public static readonly DependencyProperty AutoCloseMsProperty = DependencyProperty.Register(
        nameof(AutoCloseMs), typeof(int), typeof(AlarmBar), 
        new PropertyMetadata(5000));

    public int AutoCloseMs
    {
        get => (int)GetValue(AutoCloseMsProperty);
        set => SetValue(AutoCloseMsProperty, value);
    }

    // 关闭命令
    public static readonly DependencyProperty CloseCommandProperty = DependencyProperty.Register(
        nameof(CloseCommand), typeof(ICommand), typeof(AlarmBar), 
        new PropertyMetadata(null));

    public ICommand CloseCommand
    {
        get => (ICommand)GetValue(CloseCommandProperty);
        set => SetValue(CloseCommandProperty, value);
    }

    private static void OnAlarmLevelChanged(DependencyObject d, DependencyPropertyChangedEventArgs e)
    {
        var bar = (AlarmBar)d;
        var level = (AlarmLevel)e.NewValue;
        
        // 根据报警级别设置背景色
        bar.Background = level switch
        {
            AlarmLevel.Info => Brushes.Blue,
            AlarmLevel.Warning => Brushes.Orange,
            AlarmLevel.Error => Brushes.Red,
            AlarmLevel.Critical => Brushes.DarkRed,
            _ => Brushes.Gray
        };
    }

    public override void OnApplyTemplate()
    {
        base.OnApplyTemplate();
        
        if (GetTemplateChild("PART_Close") is Button closeButton)
        {
            closeButton.Click += (s, e) => CloseCommand?.Execute(null);
        }

        // 自动关闭
        if (AutoCloseMs > 0)
        {
            _ = Task.Delay(AutoCloseMs).ContinueWith(_ => 
                Dispatcher.Invoke(() => CloseCommand?.Execute(null)));
        }
    }
}

public enum AlarmLevel
{
    Info,
    Warning,
    Error,
    Critical
}
```

```c#
public partial class RealTimeTrendChart : UserControl
{
    private readonly LineSeries _series;
    private readonly Queue<double> _values = new Queue<double>();
    private readonly int _maxPoints = 100;

    public RealTimeTrendChart()
    {
        InitializeComponent();
        
        _series = new LineSeries
        {
            Values = new ChartValues<double>(),
            StrokeThickness = 2,
            PointGeometry = null,
            Fill = Brushes.Transparent
        };
        
        chart.Series.Add(_series);
        DataContext = this;
    }

    // Y轴标题
    public static readonly DependencyProperty YTitleProperty = DependencyProperty.Register(
        nameof(YTitle), typeof(string), typeof(RealTimeTrendChart), 
        new PropertyMetadata("数值"));

    public string YTitle
    {
        get => (string)GetValue(YTitleProperty);
        set => SetValue(YTitleProperty, value);
    }

    // 线条颜色
    public static readonly DependencyProperty LineColorProperty = DependencyProperty.Register(
        nameof(LineColor), typeof(Brush), typeof(RealTimeTrendChart), 
        new PropertyMetadata(Brushes.Blue, OnLineColorChanged));

    public Brush LineColor
    {
        get => (Brush)GetValue(LineColorProperty);
        set => SetValue(LineColorProperty, value);
    }

    private static void OnLineColorChanged(DependencyObject d, DependencyPropertyChangedEventArgs e)
    {
        var chart = (RealTimeTrendChart)d;
        chart._series.Stroke = e.NewValue as Brush;
    }

    // 添加数据点（线程安全）
    public void AddValue(double value)
    {
        if (!Dispatcher.CheckAccess())
        {
            Dispatcher.Invoke(() => AddValue(value));
            return;
        }

        _values.Enqueue(value);
        if (_values.Count > _maxPoints)
        {
            _values.Dequeue();
        }

        _series.Values.Clear();
        _series.Values.AddRange(_values);
    }

    // X轴格式化器
    public Func<double, string> XFormatter => value => 
        $"{Math.Round(value * 1, 0)}s";

    // Y轴格式化器
    public Func<double, string> YFormatter => value => 
        $"{Math.Round(value, 1)}";
}
```

## 四、工业控件设计原则总结

所有工业控件都应该遵循以下设计原则：

1. **线程安全**：所有 UI 更新必须通过 Dispatcher 调度，支持后台线程直接调用
2. **高可靠性**：添加完善的异常处理，避免控件崩溃导致整个系统不稳定
3. **低内存占用**：及时释放资源，避免内存泄漏
4. **高 DPI 支持**：使用矢量图形和相对布局，在不同 DPI 下都能清晰显示
5. **MVVM 友好**：使用依赖属性和命令，支持数据绑定
6. **样式与逻辑分离**：所有外观都通过 ControlTemplate 定义，便于自定义主题
7. **工业风格**：简洁、清晰、高对比度，避免花哨的效果，重点突出信息
