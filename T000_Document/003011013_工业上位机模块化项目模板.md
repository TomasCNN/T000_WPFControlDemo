```c#
        public bool StartCapture()

            try
            {
                // 开始采集代码
                _logger.Info("埃科相机开始采集");
                return true;
            }
            catch (Exception ex)
            {
                _logger.Error("埃科相机开始采集失败", ex);
                return false;
            }
        }

        public bool StopCapture()
        {
            try
            {
                // 停止采集代码
                _logger.Info("埃科相机停止采集");
                return true;
            }
            catch (Exception ex)
            {
                _logger.Error("埃科相机停止采集失败", ex);
                return false;
            }
        }

        public Bitmap GetBitmap(int timeoutMs = 3000)
        {
            try
            {
                // 清空旧图
                lock (typeof(IkCamera))
                {
                    _lineScanImage?.Dispose();
                    _lineScanImage = null;
                }

                // 分级等待：前100ms自旋，之后睡眠
                var sw = Stopwatch.StartNew();
                var spinWait = new SpinWait();

                while (sw.ElapsedMilliseconds < timeoutMs)
                {
                    if (_lineScanImage != null)
                        break;

                    if (sw.ElapsedMilliseconds < 100)
                    {
                        spinWait.SpinOnce();
                    }
                    else
                    {
                        Thread.Sleep(5);
                    }
                }

                if (_lineScanImage == null)
                {
                    _logger.Error($"相机取图超时，等待了{sw.ElapsedMilliseconds}ms");
                    return null;
                }

                // 克隆图像
                var newBitmap = new Bitmap(_lineScanImage);

                // 释放原图
                lock (typeof(IkCamera))
                {
                    _lineScanImage.Dispose();
                    _lineScanImage = null;
                }

                _logger.Debug($"相机取图成功，耗时{sw.ElapsedMilliseconds}ms");
                return newBitmap;
            }
            catch (Exception ex)
            {
                _logger.Error("相机取图异常", ex);
                return null;
            }
        }

        public bool SetExposureTime(int exposureUs)
        {
            // 设置曝光时间代码
            _logger.Info($"设置曝光时间：{exposureUs}us");
            return true;
        }

        public bool SetGain(double gain)
        {
            // 设置增益代码
            _logger.Info($"设置增益：{gain}");
            return true;
        }

        public bool TriggerOnce()
        {
            // 软触发代码
            _logger.Info("软触发一次");
            return true;
        }

        public void Dispose()
        {
            StopCapture();
            lock (typeof(IkCamera))
            {
                _lineScanImage?.Dispose();
                _lineScanImage = null;
            }
            _logger.Info("埃科相机资源已释放");
```

#### 03_Services/PlcService.cs

csharp:

```c#
using System.Threading;
using IndustrialVisionTemplate.Abstractions;

namespace IndustrialVisionTemplate.Services
{
    /// <summary>
    /// PLC服务
    /// 封装所有PLC操作，提供强类型方法
    /// </summary>
    public class PlcService
    {
        private readonly IPlc _plc;
        private readonly ILogger _logger;

        public PlcService(IPlc plc, ILogger logger)
        {
            _plc = plc;
            _logger = logger;
        }

        /// <summary>
        /// 连接PLC
        /// </summary>
        public bool Connect(string ip, int port)
        {
            return _plc.Connect(ip, port);
        }

        /// <summary>
        /// 等待起点触发信号
        /// </summary>
        public bool WaitForStartTrigger(int timeoutMs = 30000)
        {
            _logger.Info("等待PLC起点触发信号");
            var sw = Stopwatch.StartNew();
            var spinWait = new SpinWait();

            while (sw.ElapsedMilliseconds < timeoutMs)
            {
                if (_plc.ReadBool("M100")) // 起点触发地址
                {
                    _logger.Info($"收到起点触发信号，等待耗时{sw.ElapsedMilliseconds}ms");
                    _plc.WriteBool("M100", false); // 复位信号
                    return true;
                }

                if (sw.ElapsedMilliseconds < 100)
                {
                    spinWait.SpinOnce();
                }
                else
                {
                    Thread.Sleep(5);
                }
            }

            _logger.Error($"等待起点触发信号超时，已等待{sw.ElapsedMilliseconds}ms");
            return false;
        }

        /// <summary>
        /// 发送终点触发信号
        /// </summary>
        public void SendEndTrigger(bool isOk)
        {
            short value = (short)(isOk ? 3 : 999);
            _plc.WriteShort("D200", value); // 终点触发地址
            _logger.Info($"发送终点触发信号：{value}");
        }

        /// <summary>
        /// 发送报警信号
        /// </summary>
        public void SendAlarm(string message)
        {
            _plc.WriteBool("M200", true); // 报警地址
            _logger.Error($"设备报警：{message}");
        }

        /// <summary>
        /// 清除报警
        /// </summary>
        public void ClearAlarm()
        {
            _plc.WriteBool("M200", false);
            _logger.Info("报警已清除");
        }
    }
}
```

### 4. 业务逻辑层

#### 04_Business/DetectionProcess.cs（检测主流程）

csharp:

```c#
using System.Drawing;
using IndustrialVisionTemplate.Abstractions;
using IndustrialVisionTemplate.Services;

namespace IndustrialVisionTemplate.Business
{
    /// <summary>
    /// 检测主流程
    /// 核心业务逻辑，编排所有服务
    /// </summary>
    public class DetectionProcess
    {
        private readonly CameraService _cameraService;
        private readonly PlcService _plcService;
        private readonly AlgorithmService _algorithmService;
        private readonly DataService _dataService;
        private readonly ILogger _logger;

        public DetectionProcess(
            CameraService cameraService,
            PlcService plcService,
            AlgorithmService algorithmService,
            DataService dataService,
            ILogger logger)
        {
            _cameraService = cameraService;
            _plcService = plcService;
            _algorithmService = algorithmService;
            _dataService = dataService;
            _logger = logger;
        }

        /// <summary>
        /// 执行一次完整的检测流程
        /// </summary>
        public async Task<bool> RunAsync()
        {
            try
            {
                _logger.Info("开始检测流程");
                var totalSw = Stopwatch.StartNew();

                // 1. 等待PLC起点触发
                if (!await Task.Run(() => _plcService.WaitForStartTrigger()))
                {
                    return false;
                }

                // 2. 取图
                var sw = Stopwatch.StartNew();
                var bitmap = await Task.Run(() => _cameraService.GetBitmap());
                if (bitmap == null)
                {
                    _plcService.SendAlarm("相机取图失败");
                    return false;
                }
                sw.Stop();
                _logger.Info($"取图耗时：{sw.ElapsedMilliseconds}ms");

                // 3. 立刻发送PLC结束信号（CT优化核心）
                // 设备和电脑并行工作，不要等算法处理完
                _plcService.SendEndTrigger(true);

                // 4. 图像处理和缺陷检测
                sw.Restart();
                var defects = await Task.Run(() => _algorithmService.DetectDefects(bitmap));
                sw.Stop();
                _logger.Info($"算法处理耗时：{sw.ElapsedMilliseconds}ms，检测到{defects.Count}个缺陷");

                // 5. 保存数据和图像
                var record = new ProductRecord
                {
                    Barcode = DateTime.Now.ToString("yyyyMMddHHmmssfff"),
                    IsOk = defects.Count == 0,
                    Defects = defects,
                    CreateTime = DateTime.Now
                };
                await Task.Run(() => _dataService.SaveRecord(record, bitmap));

                totalSw.Stop();
                _logger.Info($"检测流程完成，总耗时：{totalSw.ElapsedMilliseconds}ms，结果：{(record.IsOk ? "OK" : "NG")}");

                return record.IsOk;
            }
            catch (Exception ex)
            {
                _logger.Error("检测流程异常", ex);
                _plcService.SendAlarm("检测流程异常");
                return false;
            }
        }
    }
}
```

```c#
using Prism.Commands;
using Prism.Events;
using Prism.Mvvm;
using IndustrialVisionTemplate.Business;
using IndustrialVisionTemplate.Services;

namespace IndustrialVisionTemplate.ViewModels
{
    /// <summary>
    /// 主界面ViewModel
    /// </summary>
    public class MainViewModel : BindableBase
    {
        private readonly DetectionProcess _detectionProcess;
        private readonly IEventAggregator _eventAggregator;
        private readonly ILogger _logger;

        private bool _isRunning;
        private string _statusMessage = "系统就绪";
        private int _totalCount;
        private int _okCount;
        private int _ngCount;

        public bool IsRunning
        {
            get => _isRunning;
            set => SetProperty(ref _isRunning, value);
        }

        public string StatusMessage
        {
            get => _statusMessage;
            set => SetProperty(ref _statusMessage, value);
        }

        public int TotalCount
        {
            get => _totalCount;
            set => SetProperty(ref _totalCount, value);
        }

        public int OkCount
        {
            get => _okCount;
            set => SetProperty(ref _okCount, value);
        }

        public int NgCount
        {
            get => _ngCount;
            set => SetProperty(ref _ngCount, value);
        }

        public DelegateCommand StartCommand { get; }
        public DelegateCommand StopCommand { get; }

        public MainViewModel(
            DetectionProcess detectionProcess,
            IEventAggregator eventAggregator,
            ILogger logger)
        {
            _detectionProcess = detectionProcess;
            _eventAggregator = eventAggregator;
            _logger = logger;

            StartCommand = new DelegateCommand(StartProcess, () => !IsRunning);
            StopCommand = new DelegateCommand(StopProcess, () => IsRunning);
        }

        private async void StartProcess()
        {
            IsRunning = true;
            StatusMessage = "系统运行中";
            _logger.Info("系统开始运行");

            // 后台循环检测
            await Task.Run(async () =>
            {
                while (IsRunning)
                {
                    var result = await _detectionProcess.RunAsync();

                    // 更新统计
                    TotalCount++;
                    if (result)
                        OkCount++;
                    else
                        NgCount++;
                }
            });

            StatusMessage = "系统已停止";
            _logger.Info("系统已停止");
        }

        private void StopProcess()
        {
            IsRunning = false;
        }
    }
}
```

## 四、快速上手指南

1. **替换硬件实现**：根据你实际使用的相机和 PLC，修改`02_Implementations`文件夹下的代码
2. **修改 PLC 地址**：在`PlcService.cs`中修改实际的 PLC 信号地址
3. **添加算法逻辑**：在`AlgorithmService.cs`中实现你的缺陷检测算法
4. **修改界面**：在`06_Views`文件夹中修改界面布局，适配你的设备
5. **配置数据库**：在`DatabaseContext.cs`中配置你的数据库连接

## 五、模板优势

- ✅ **完全模块化**：换硬件只需要改实现类，上层代码一行不动
- ✅ **CT 最优**：集成了所有工业级优化（分级等待、提前发 PLC 信号）
- ✅ **生产级稳定**：完善的异常处理、资源释放、日志记录
- ✅ **团队友好**：标准架构，新人上手快
- ✅ **可扩展性强**：添加新功能只需要加新模块，不影响现有代码
