# 999999006-5_C# 特性 (Attribute) 的 10 大核心应用场景与实战案例

特性是 C# 中最强大的声明式编程工具，它通过将元数据与代码元素关联，实现了 "代码即配置" 的编程范式。结合反射机制，特性可以在不修改原有代码逻辑的情况下，为程序添加各种横切关注点功能。

以下是特性在实际开发中最常见、最有价值的应用场景，每个场景都配有具体的代码示例和最佳实践。

```c#
[AttributeUsage(AttributeTargets.Class)]
public class DeviceDriverAttribute : Attribute
{
    public string Manufacturer { get; }
    public string Model { get; }
    public string Protocol { get; }

    public DeviceDriverAttribute(string manufacturer, string model, string protocol)
    {
        Manufacturer = manufacturer;
        Model = model;
        Protocol = protocol;
    }
}

[DeviceDriver("IK", "E200", "TCP/IP")]
public class IKCameraDriver : ICameraDriver
{
    // 埃科相机驱动实现
}

[DeviceDriver("Basler", "acA2040-180km", "GigE Vision")]
public class BaslerCameraDriver : ICameraDriver
{
    // Basler相机驱动实现
}

// 自动发现并注册所有设备驱动
public class DeviceDriverFactory
{
    public static ICameraDriver CreateDriver(string manufacturer, string model)
    {
        var driverType = Assembly.GetExecutingAssembly()
            .GetTypes()
            .Where(t => typeof(ICameraDriver).IsAssignableFrom(t) && !t.IsAbstract)
            .Select(t => new
            {
                Type = t,
                Attr = t.GetCustomAttribute<DeviceDriverAttribute>()
            })
            .FirstOrDefault(x => x.Attr?.Manufacturer == manufacturer && x.Attr?.Model == model)
            ?.Type;

        if (driverType == null)
            throw new NotSupportedException($"不支持的设备：{manufacturer} {model}");

        return (ICameraDriver)Activator.CreateInstance(driverType);
    }
}
```

## 四、AOP 与面向切面编程

特性是实现 AOP（面向切面编程）的核心工具，可以在不修改原有业务代码的情况下，添加日志、事务、异常处理等横切关注点。

### 4.1 事务特性

自动管理数据库事务：

csharp:

```c#
[AttributeUsage(AttributeTargets.Method)]
public class TransactionalAttribute : Attribute
{
    public IsolationLevel IsolationLevel { get; set; } = IsolationLevel.ReadCommitted;
}

// 拦截器实现（使用Castle DynamicProxy）
public class TransactionInterceptor : IInterceptor
{
    private readonly IDbContext _dbContext;

    public TransactionInterceptor(IDbContext dbContext)
    {
        _dbContext = dbContext;
    }

    public void Intercept(IInvocation invocation)
    {
        var attribute = invocation.Method.GetCustomAttribute<TransactionalAttribute>();
        if (attribute == null)
        {
            invocation.Proceed();
            return;
        }

        using var transaction = _dbContext.Database.BeginTransaction(attribute.IsolationLevel);
        try
        {
            invocation.Proceed();
            transaction.Commit();
        }
        catch
        {
            transaction.Rollback();
            throw;
        }
    }
}

// 使用
public class OrderService
{
    [Transactional]
    public void CreateOrder(Order order)
    {
        // 业务逻辑：创建订单、扣减库存、更新用户积分
        // 所有操作在同一个事务中执行
    }
}
```

```c#
[AttributeUsage(AttributeTargets.Method)]
public class LogExecutionTimeAttribute : Attribute
{
    public string OperationName { get; }

    public LogExecutionTimeAttribute(string operationName)
    {
        OperationName = operationName;
    }
}

public class LoggingInterceptor : IInterceptor
{
    private readonly ILogger _logger;

    public LoggingInterceptor(ILogger logger)
    {
        _logger = logger;
    }

    public void Intercept(IInvocation invocation)
    {
        var attribute = invocation.Method.GetCustomAttribute<LogExecutionTimeAttribute>();
        if (attribute == null)
        {
            invocation.Proceed();
            return;
        }

        var stopwatch = Stopwatch.StartNew();
        try
        {
            invocation.Proceed();
            _logger.LogInformation("{Operation} 执行成功，耗时：{Elapsed}ms", 
                attribute.OperationName, stopwatch.ElapsedMilliseconds);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "{Operation} 执行失败，耗时：{Elapsed}ms", 
                attribute.OperationName, stopwatch.ElapsedMilliseconds);
            throw;
        }
    }
}

// 使用
public class ProductService
{
    [LogExecutionTime("获取产品列表")]
    public List<Product> GetAllProducts()
    {
        // 业务逻辑
    }
}
```

## 五、依赖注入与服务注册

特性可以自动发现并注册服务，替代手动注册的繁琐过程。

csharp:

```c#
[AttributeUsage(AttributeTargets.Class)]
public class ServiceAttribute : Attribute
{
    public ServiceLifetime Lifetime { get; }
    public Type ServiceType { get; }

    public ServiceAttribute(ServiceLifetime lifetime = ServiceLifetime.Scoped, Type serviceType = null)
    {
        Lifetime = lifetime;
        ServiceType = serviceType;
    }
}

// 自动注册服务扩展方法
public static class ServiceCollectionExtensions
{
    public static IServiceCollection AddServicesByAttribute(this IServiceCollection services, Assembly assembly)
    {
        var serviceTypes = assembly.GetTypes()
            .Where(t => !t.IsAbstract && !t.IsInterface)
            .Select(t => new
            {
                ImplementationType = t,
                Attribute = t.GetCustomAttribute<ServiceAttribute>()
            })
            .Where(x => x.Attribute != null);

        foreach (var item in serviceTypes)
        {
            var serviceType = item.Attribute.ServiceType ?? item.ImplementationType;
            services.Add(new ServiceDescriptor(serviceType, item.ImplementationType, item.Attribute.Lifetime));
        }

        return services;
    }
}

// 使用
[Service(ServiceLifetime.Scoped, typeof(IProductService))]
public class ProductService : IProductService { }

[Service(ServiceLifetime.Singleton)]
public class ConfigurationService { }

// 在Program.cs中
builder.Services.AddServicesByAttribute(Assembly.GetExecutingAssembly());
```

## 六、权限与安全控制

特性可以集中管理权限控制逻辑，实现声明式授权。

csharp:

```c#
[AttributeUsage(AttributeTargets.Method | AttributeTargets.Class)]
public class PermissionAttribute : Attribute
{
    public string[] Permissions { get; }

    public PermissionAttribute(params string[] permissions)
    {
        Permissions = permissions;
    }
}

// ASP.NET Core授权过滤器实现
public class PermissionFilter : IAsyncAuthorizationFilter
{
    private readonly IUserService _userService;

    public PermissionFilter(IUserService userService)
    {
        _userService = userService;
    }

    public async Task OnAuthorizationAsync(AuthorizationFilterContext context)
    {
        var attribute = context.ActionDescriptor.EndpointMetadata
            .OfType<PermissionAttribute>()
            .FirstOrDefault();

        if (attribute == null)
            return;

        var userId = context.HttpContext.User.FindFirstValue(ClaimTypes.NameIdentifier);
        if (userId == null)
        {
            context.Result = new UnauthorizedResult();
            return;
        }

        var userPermissions = await _userService.GetUserPermissions(int.Parse(userId));
        if (!attribute.Permissions.All(p => userPermissions.Contains(p)))
        {
            context.Result = new ForbidResult();
            return;
        }
    }
}

// 使用
[Permission("product:view")]
public IActionResult GetProducts() { }

[Permission("product:create", "product:edit")]
public IActionResult CreateProduct() { }
```

```c#
// 插件接口
public interface IPlugin
{
    string Name { get; }
    void Execute();

// 插件特性
[AttributeUsage(AttributeTargets.Class)]
public class PluginAttribute : Attribute
{
    public string Name { get; }
    public string Version { get; }
    public string Description { get; }

    public PluginAttribute(string name, string version, string description)
    {
        Name = name;
        Version = version;
        Description = description;
    }
}

// 插件加载器
public class PluginLoader
{
    public List<IPlugin> LoadPlugins(string pluginDirectory)
    {
        var plugins = new List<IPlugin>();

        foreach (var file in Directory.GetFiles(pluginDirectory, "*.dll"))
        {
            var assembly = Assembly.LoadFrom(file);
            var pluginTypes = assembly.GetTypes()
                .Where(t => typeof(IPlugin).IsAssignableFrom(t) && !t.IsAbstract && !t.IsInterface);

            foreach (var type in pluginTypes)
            {
                var attribute = type.GetCustomAttribute<PluginAttribute>();
                if (attribute != null)
                {
                    var plugin = (IPlugin)Activator.CreateInstance(type);
                    plugins.Add(plugin);
                    Console.WriteLine($"加载插件：{attribute.Name} v{attribute.Version}");
                }
            }
        }

        return plugins;
    }
}
```

## 特性使用的最佳实践与注意事项

### 10.1 什么时候应该使用特性

- 当你需要为代码元素添加声明性元数据时
- 当你需要实现横切关注点（日志、事务、权限）时
- 当你需要减少重复代码和配置时
- 当你需要实现框架级别的扩展能力时

### 10.2 什么时候不应该使用特性

- 当逻辑非常复杂且变化频繁时（特性应该是静态的）
- 当性能要求极高且无法接受反射开销时
- 当需要强类型编译时检查时（特性是运行时的）

### 10.3 性能最佳实践

- 优先使用`IsDefined`检查特性是否存在
- 手动缓存特性实例，避免重复反射
- 避免在特性构造函数中执行复杂逻辑
- 在应用启动时预加载关键特性

## 总结

特性是 C# 中最强大的元数据编程工具，它的核心价值在于：

1. **声明式编程**：将配置与代码逻辑分离
2. **横切关注点**：在不修改业务代码的情况下添加通用功能
3. **自动化**：通过反射自动发现和处理代码元素
4. **可扩展性**：为框架和库提供强大的扩展能力

从简单的枚举描述到复杂的 AOP 框架，特性几乎贯穿了.NET 开发的方方面面。掌握特性的使用，能够让你写出更加简洁、灵活和可维护的代码。
